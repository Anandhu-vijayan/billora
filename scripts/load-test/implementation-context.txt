export async function GET(req: Request) {
  const { searchParams } = new URL(req.url);
  const mode = searchParams.get('hub.mode');
  const receivedToken = searchParams.get('hub.verify_token');
  const expectedToken = process.env.WHATSAPP_VERIFY_TOKEN || '';
  // Minimal operational log: outcome only, no token material or comparison diagnostics.
  const verified = mode === 'subscribe' && !!expectedToken && !!receivedToken
    && safeTokenEqual(receivedToken, expectedToken);
  logger.info('[WhatsApp] verify attempt', { mode: mode ?? null, verified });
  if (verified) {
    return new Response(searchParams.get('hub.challenge') || '', {
      status: 200, headers: { 'Content-Type': 'text/plain' },
    });
  }
  return new Response('Forbidden', { status: 403 });
}

/**
 * Humanize a single editor field's validation failure (from the shared
 * parseReviewCorrections) so the reviewer learns exactly WHICH field and WHY it
 * was rejected — never a silent no-op. The codes mirror lib/invoices/review.ts.
 */
function describeFieldErrors(label: string, errors: Array<{ field: string; code: string }>): string {
  const reason = (code: string): string =>
    code === 'TOO_LONG' ? 'is too long'
      : code === 'INVALID_AMOUNT' ? 'is not a readable amount'
        : code === 'NEGATIVE_NOT_ALLOWED' ? 'cannot be negative'
          : code === 'INVALID_TYPE' ? 'was not a valid value'
            : 'was not accepted';
  return `I could not save that change — ${label} ${reason(errors[0]?.code ?? '')}.`;
}

/**
 * How long a sent upload acknowledgment suppresses further acks (§user: one
 * confirmation per burst, not one per file). A multi-file send arrives as
 * several near-simultaneous webhooks; the atomic claim in sessions.claimUploadAck
 * lets only the first through, and this window re-arms the slot so a genuinely
 * later batch still gets exactly one ack. Five minutes comfortably spans a burst
 * of forwarded PDFs without going quiet for a fresh working session.
 */
const UPLOAD_ACK_COALESCE_MS = 5 * 60 * 1000;

/**

 * an approve-then-Done flow often has the workbook a few seconds from ready; waiting
 * briefly lets the SAME reply attach the Excel instead of deferring to the next
 * inbound. Capped well under Meta's webhook budget so a slow/failed job still falls
 * back to the truthful "still working" reply (delivered later via the outbox flush).
 */

/**
 * Provisioning + conditional persist FIRST, outbound sends SECOND.
 * Ordering guarantee: the session row is the source of truth. A failed
 * conditional save (concurrent delivery won the race) throws STALE_SESSION so
 * the caller returns retryable 503 and sends nothing — no contradictory replies.
 * Rule 5 — processing success is separate from message delivery: a failed
 * send NEVER rewinds the persisted state or re-triggers intake. Transient
 * failures are deferred (Phase 6 outbound queue resends them); credential
 * blocks are alerted in send.ts without provoking a Meta retry loop.
 */
async function handleOnboardingSideEffects(
  db: Db, hmac: string, label: string, to: string, providerMessageId: string,
  snapshot: SessionSnapshot, result: TransitionResult,
): Promise<{ workspaceId: string | null; clientId: string | null }> {
  let workspaceId: string | null = snapshot.workspaceId;
  let clientId: string | null = snapshot.clientId;
  let pendingClientId: string | null = snapshot.pendingClientId;
  let assignedNow = 0;
  // Result of a WhatsApp-native review confirmation, delivered as the reply.
  let confirmOutcome: ConfirmOutcome | null = null;
  // The invoice currently open in the in-chat editor. Persisted (survives a
  // restart / webhook retry) and cleared when the editor sub-state is exited.
  let editInvoiceId: string | null = snapshot.reviewEditInvoiceId ?? null;
  // The single text reply for an editor turn (open / correct / approve / help).
  // Computed during side effects (which run BEFORE persist+send) so a DB write is
  // never reordered ahead of its persistence guard.
  let editReplyBody: string | null = null;
  // The reply for an optional own-GSTIN capture turn (§P0-1b), computed in the
  // side-effects block after the tenant-scoped, migration-safe write attempt.
  let gstinReplyBody: string | null = null;
  // The reply for a proactive opt-in/opt-out turn (§P1-6), computed in the
  // side-effects block after the consent write (which seals/clears the wa_id).
  let proactiveReplyBody: string | null = null;
  // The reply for a bulk "approve all" turn (§P1-7): the honest counts of what was
  // filed vs. what still needs a specific correction (never a forced approval).
  let approveAllSummary: string | null = null;

  // ---- provisioning + pending resolution (may rewrite nextState) ----
  if (result.note === 'individual-selected') {
    workspaceId = await ensureIndividualWorkspace(db, hmac);
    result.workspaceId = workspaceId;
    // Individuals carry no client ambiguity: file held uploads immediately.
    assignedNow = await assignPendingUploads(db, hmac, {
      workspaceId, clientId: null, limit: getEnv().WHATSAPP_MAX_INVOICES_PER_DAY,
    });
  } else if (result.note.startsWith('org-name:')) {
    const ensured = await ensureOrganizationWorkspace(db, hmac, result.note.slice(9));
    workspaceId = ensured.workspaceId;
    result.workspaceId = workspaceId;
  } else if (result.note.startsWith('client-name:')) {
    if (!workspaceId) throw new Error('ONBOARDING_FAULT: no workspace for client creation');
    clientId = await createClient(db, workspaceId, result.note.slice(12));
    result.clientId = clientId;
  } else if (result.note === 'client-selected' && result.clientId) {
    if (!workspaceId || !(await verifyClientInWorkspace(db, workspaceId, result.clientId))) {
      logger.info('[WhatsApp] unauthorized client selection rejected', { sender: label });
      result.nextState = 'awaiting_client_select';
      result.clientId = undefined;
      result.outbound = { kind: 'none' };
      result.note = 'client-selection-rejected';
    } else {
      clientId = result.clientId;
    }
  } else if (result.note === 'pending-confirm-yes') {
    if (!workspaceId || !pendingClientId) {
      // The question outlived its candidate (client deleted, stale retry):
      // degrade to plain collection, held files stay held.
      pendingClientId = null;
      result.note = 'pending-confirm-stale';
    } else {
      clientId = pendingClientId;
      result.clientId = clientId;
      assignedNow = await assignPendingUploads(db, hmac, {
        workspaceId, clientId, limit: getEnv().WHATSAPP_MAX_INVOICES_PER_DAY,
      });
      pendingClientId = null;
    }
  } else if (result.note === 'pending-confirm-no') {
    // Client stays active for FUTURE uploads; held files are NOT reassigned.
    pendingClientId = null;
  } else if (result.note.startsWith('gstin-set:') || result.note === 'gstin-clear') {
    // Optional own-GSTIN capture (§P0-1b). Persist to the ACTIVE client, tenant-
    // scoped and migration-safe, and never change the state (this is a side note,
    // not a conversation turn). A near-miss value never reaches here: the pure
    // parser only emits `gstin-set:<VALUE>` for a checksum-valid GSTIN.
    const isSet = result.note.startsWith('gstin-set:')
    const value = isSet ? result.note.slice('gstin-set:'.length) : null
    if (!workspaceId || !clientId) {
      gstinReplyBody = 'Select a client first (send Change client), then send "My GSTIN <number>" so I attach it to the right client.'
    } else {
      const r = await setClientOwnGstin(db, workspaceId, clientId, value)
      gstinReplyBody = r.ok
        ? (isSet
            ? `Saved. I will now treat ${value} as your own (buyer) GSTIN and never file it as a supplier.`
            : 'Cleared — no own GSTIN is saved for this client.')
        : r.reason === 'NOT_MIGRATED'
          ? 'I could not save that GSTIN yet — this workspace needs a one-time update first. Your invoices are still processed, just without the own-GSTIN cross-check.'
          : r.reason === 'NOT_FOUND'
            ? 'That client is no longer available — send Change client to pick one, then retry.'
            : 'I could not save that GSTIN right now — please try again in a moment.'
    }
    result.note = isSet ? 'gstin-set' : 'gstin-clear'
  } else if (result.note === 'proactive-enable' || result.note === 'proactive-disable') {
    // Opt-in proactive delivery (§P1-6). A side note — never changes state. Consent
    // is only ACTED ON (the number sealed + stored) when the operator has switched
    // the feature on AND configured the separate key; otherwise we acknowledge the
    // intent and store NOTHING. Opting out always clears consent + the sealed value.
    const enable = result.note === 'proactive-enable';
    if (!workspaceId) {
      proactiveReplyBody = 'Finish setup first — send Hi, then reply "Proactive on".';
    } else if (enable && !(proactiveDeliveryEnabled() && proactiveKeyConfigured())) {
      proactiveReplyBody = 'Proactive delivery is not switched on for this workspace yet, so I can only send your register when you message. Nothing was stored — reply "Proactive on" once it is enabled.';
    } else {
      const r = await setProactiveConsent(db, hmac, { enabled: enable, waId: enable ? to : null });
      proactiveReplyBody = r.ok
        ? (enable
            ? 'Done — I can now push your finished register to this number without waiting for you to message (only while WhatsApp\'s 24-hour delivery window is open). Reply "Proactive off" anytime to stop.'
            : 'OK — I have turned off proactive delivery and removed your number from the push list. Your register will be sent when you message. Reply "Proactive on" to allow pushes again anytime.')
        : r.reason === 'NOT_MIGRATED'
          ? 'I noted that, but proactive delivery needs a one-time update here before it can work — for now I\'ll still send your register when you message.'
          : r.reason === 'NO_KEY'
            ? 'Proactive delivery is on but not fully configured, so I can\'t store your number safely yet. For now I\'ll send your register when you message.'
            : 'I could not update proactive delivery right now — please try again in a moment.';
    }
  } else if (result.note === 'review-confirm' && result.invoiceId) {
    // A tapped Confirm row on the Review list: approve that invoice through the
    // SHARED review path, scoped to the sender's own workspace. The action runs
    // before the reply is sent; it is idempotent, so a redelivered webhook that
    // re-confirms never duplicates a batch or a workbook job.
    if (!workspaceId) {
      confirmOutcome = { ok: false, error: 'NO_WORKSPACE' };
    } else {
      try {
        confirmOutcome = await confirmInvoice(db, { workspaceId, invoiceId: result.invoiceId });
      } catch (e) {
        logger.warn('[WhatsApp] review confirm failed', {
          sender: label, error: e instanceof Error ? e.message : String(e),
        });
        confirmOutcome = { ok: false, error: 'FAULT' };
      }
    }
  } else if (result.note === 'review-approve-all') {
    // Bulk approval (§P1-7). Every row goes through confirmInvoice — the SAME shared,
    // re-validating path a single tap uses — so "approve all" never bypasses a
    // validator: an invoice that still fails to reconcile stays needs_review and is
    // reported honestly, never force-filed. Bounded + idempotent (safe on redelivery).
    if (!workspaceId) {
      approveAllSummary = "You're not set up yet — send Hi to begin, then Review to approve invoices.";
    } else {
      try {
        const r = await approveAllPending(db, workspaceId, 25);
        const parts: string[] = [];
        if (r.approved > 0) parts.push(`${r.approved} invoice(s) approved and filed for the register.`);
        if (r.stillPending > 0) {
          // §UX: name the EXACT field to correct so "approve all" never looks like it
          // did nothing. Descriptive only — a row that fails a validator is still
          // never force-filed; the reviewer fixes the field, then replies APPROVE.
          const fix = r.blockers.length > 0
            ? ` To fix, tap an invoice below and correct: ${r.blockers.join('; ')}. Then reply APPROVE.`
            : ' Tap an invoice below, fix the flagged field, then reply APPROVE.';
          parts.push(`${r.stillPending} could NOT be approved as-is — a value did not add up, so nothing was forced through.${fix}`);
        }
        if (r.failed > 0) parts.push(`${r.failed} could not be processed right now — please try again in a moment.`);
        approveAllSummary = parts.length > 0
          ? `Approve all: ${parts.join(' ')} Your Excel register is generated and sent here automatically once every invoice is approved — nothing is filed until then.`
          : 'Nothing was ready for bulk approval — send Review, tap an invoice, fix the field it names, then reply APPROVE.';
      } catch (e) {
        logger.warn('[WhatsApp] approve-all failed', { sender: label, error: e instanceof Error ? e.message : String(e) });
        approveAllSummary = 'I could not complete the bulk approval right now — please try again in a moment.';
      }
    }
  } else if (result.note.startsWith('review-edit')) {
    // The in-chat field editor. Every action here is scoped to the sender's own
    // workspace and the ONE invoice the editor is open on. The cardinal rule (§7):
    // a field CORRECTION persists + re-validates but NEVER approves or batches —
    // only an explicit APPROVE (the shared confirm path) can file an invoice.
    if (result.note === 'review-edit-invalid') {
      editReplyBody = 'That edit request is no longer valid — send Review to see the current list.';
    } else if (!workspaceId || !result.invoiceId) {
      editInvoiceId = null;
      editReplyBody = "You're not set up yet — send Hi to begin, then Review to correct an invoice.";
    } else {
      const cmd = result.editCommand;
      const invoiceId = result.invoiceId;
      if (result.note === 'review-edit') {
        // ENTER the editor (tapped an Edit row): read-only, shows the stored row.
        editInvoiceId = invoiceId;
        const load = await loadInvoiceForEdit(db, { workspaceId, invoiceId });
        if (load.ok) editReplyBody = composeReviewEditorMessage(load.row, { reasons: load.reasons, classification: load.classification });
        else if (load.error === 'NOT_PENDING') editReplyBody = 'That invoice is no longer pending review — send Review to see the current list.';
        else if (load.error === 'NOT_FOUND') editReplyBody = 'That invoice was not found — send Review to see the current list.';
        else editReplyBody = 'I could not open that invoice right now — please try again in a moment.';
      } else if (cmd && cmd.action === 'correct') {
        const outcome = await applyInvoiceEdit(db, { workspaceId, invoiceId, fieldKey: cmd.field.key, rawValue: cmd.value });
        if (outcome.ok && outcome.row) {
          const note = outcome.approvable
            ? 'Saved. Everything checks out now — reply APPROVE to file it for the register.'
            : 'Saved. It is still flagged because something does not add up — correct another field, or reply APPROVE only if you are sure.';
          editReplyBody = composeReviewEditorMessage(outcome.row, { note, reasons: outcome.reasons, classification: outcome.classification ?? null });
        } else if (outcome.fieldErrors && outcome.fieldErrors.length > 0) {
          // Never silently drop a malformed value: name the field + why.
          editReplyBody = `${describeFieldErrors(cmd.field.label, outcome.fieldErrors)}\n\n${REVIEW_EDIT_HELP}`;
        } else if (outcome.error === 'NOT_PENDING') {
          editInvoiceId = null; result.nextState = 'collecting';
          editReplyBody = 'That invoice is no longer pending review — send Review to see the current list.';
        } else {
          editReplyBody = 'I could not save that change right now — please try again in a moment.';
        }
      } else if (cmd && cmd.action === 'approve') {
        // The ONLY editor action that may file + batch, via the shared confirm
        // path. confirmInvoice echoes the stored number, so a genuine approval is
        // honored but a still-incomplete row is NOT blind-approved (stays flagged).
        let outcome: ConfirmOutcome;
        try {
          outcome = await confirmInvoice(db, { workspaceId, invoiceId });
        } catch (e) {
          logger.warn('[WhatsApp] review edit approve failed', { sender: label, error: e instanceof Error ? e.message : String(e) });
          outcome = { ok: false, error: 'FAULT' };
        }
        editReplyBody = composeConfirmResultMessage(outcome);
        if (outcome.ok && outcome.review === 'approved') {
          editInvoiceId = null; result.nextState = 'collecting';
        } else {
          // Not approvable yet: stay in the editor and re-show it truthfully.
          const load = await loadInvoiceForEdit(db, { workspaceId, invoiceId });
          if (load.ok) {
            editReplyBody += '\n\n' + composeReviewEditorMessage(load.row, { header: 'Still needs a fix before it can be approved:', note: 'Correct a field above, or reply CANCEL to leave it flagged.', reasons: load.reasons, classification: load.classification });
          } else {
            editInvoiceId = null; result.nextState = 'collecting';
          }
        }
      } else if (cmd && cmd.action === 'cancel') {
        // conversation.ts already moved nextState to 'collecting'; just drop the target.
        editInvoiceId = null;
        editReplyBody = 'Left that invoice flagged — nothing was changed. Send Review anytime to edit or approve it.';
      } else if (cmd && cmd.action === 'help') {
        editReplyBody = REVIEW_EDIT_HELP;
      } else {
        // unknown-field / reprompt: never guess which field a stray number meant, and
        // never claim a REAL field does not exist — a bare number just needs its value.
        editReplyBody = cmd && cmd.action === 'unknown-field' && cmd.fieldNo != null
          ? `${fieldPromptForIndex(cmd.fieldNo)}\n\n${REVIEW_EDIT_HELP}`
          : REVIEW_EDIT_HELP;
      }
    }
  }

  // "Remember context, never guess a client": when a client just became
  // resolved and files are held for this sender, ask before applying it.
  if ((result.note === 'client-selected' || result.note.startsWith('client-name:')) && clientId && workspaceId) {
    const waiting = await countWaitingUploads(db, hmac);
    if (waiting > 0) {
      result.nextState = 'awaiting_pending_confirm';
      result.outbound = { kind: 'none' };
      pendingClientId = clientId;
    }
  }

  // Persist BEFORE any outbound send. A `false` return means a concurrent
  // delivery already advanced this sender: abort without sending.
  const saved = await saveSession(db, hmac, snapshot.lastProviderMessageId, {
    state: result.nextState,
    workspace_id: result.workspaceId ?? workspaceId,
    client_id: result.clientId ?? clientId,
    batch_id: result.batchId ?? snapshot.batchId,
    pending_client_id: pendingClientId,
    review_edit_invoice_id: editInvoiceId,
    last_provider_message_id: providerMessageId,
    expires_at: new Date(Date.now() + 24 * 60 * 60 * 1000).toISOString(),
  });
  if (!saved) throw new Error('STALE_SESSION: concurrent delivery advanced this sender; retry');
  logger.info('[WhatsApp] onboarding persisted', { sender: label, state: result.nextState });

  // Outbound sends happen only after durable persistence. Failures never throw:
  // the intake stays accepted and delivery is tracked separately (rule 5).
  // A failed (transient/blocked) TEXT reply is remembered and best-effort
  // resent on the sender's NEXT inbound message — never re-running a completed
  // business operation. Inbound idempotency (webhook_events) is untouched.
  let sendOutcome: SendOutcome = 'ok';
  const track = (outcome: SendOutcome): void => {
    if (outcome === 'blocked') sendOutcome = 'blocked';
    else if (outcome === 'transient' && sendOutcome !== 'blocked') sendOutcome = 'transient';
  };
  const deliverText = async (body: string): Promise<void> => {
    const o = await sendText(to, body);
    if (o === 'ok') await updateSessionDeliveryState(db, hmac, { pending_reply_body: null, last_reply_error_code: null });
    else await updateSessionDeliveryState(db, hmac, { pending_reply_body: body, last_reply_error_code: o });
    track(o);
  };
  // §one-delivery-path: Done/Review used to hand-send EVERY workbook it could
  // find at status='ready', in parallel with the durable outbox flush. That second
  // path is what buried owners in copies of the same month (a 2-invoice file, then
  // a 3-invoice one, then the real one) and shipped them with the generic caption,
  // because it never read the register's own month or row count. Both commands now
  // go through the single outbox consumer, which holds an intermediate register
  // while work is still running, sends only the newest file per (month, category),
  // captions it with its real month and invoice count, and records every attempt —
  // including why a send did not land — in workbook_deliveries. Returns how many
  // files reached this chat.
  const deliverOwedRegisters = async (): Promise<number> => {
    try {
      const delivered = await flushPendingDeliveries(db, hmac, to);
      logger.info('[WhatsApp] register outbox flush', { sender: label, delivered });
      return delivered;
    } catch (e) {
      logger.warn('[WhatsApp] register outbox flush failed', { sender: label, error: e instanceof Error ? e.message : 'unknown' });
      return 0;
    }
  };
  // Menu/list/button sends are re-derivable, so a failed one records only the
  // error code (no body resend); a failed interactive list degrades to text.
  const deliverMenu = async (workspace: boolean): Promise<void> => {
    const o = await sendMenu(to, { workspace });
    if (o === 'ok') { await updateSessionDeliveryState(db, hmac, { last_reply_error_code: null }); track(o); return }
    await updateSessionDeliveryState(db, hmac, { last_reply_error_code: o });
    await deliverText(menuText({ ...snapshot, workspaceId: workspace ? (workspaceId ?? snapshot.workspaceId) : null }));
  };
  // §P1-7: send ONE page of the confirmable review list, carrying an "Approve all"
  // control and a "Next" row when more remain. Read-only + best-effort: a failure
  // leaves the truthful text guidance as the only reply (never a broken list send).
  // The page is sized (REVIEW_PAGE_SIZE) so items + the two controls stay <= 10 rows.
  const sendReviewPage = async (page: number): Promise<void> => {
    if (!workspaceId) return;
    try {
      const pg = await listPendingReviewPage(db, workspaceId, page);
      if (pg.items.length === 0) return;
      const remaining = Math.max(0, pg.total - pg.page * pg.pageSize);
      const nextCount = pg.hasMore ? Math.min(pg.pageSize, remaining) : 0;
      const outcome = await sendReviewList(to, pg.items, { approveAllCount: pg.total, page: pg.page, nextCount });
      if (outcome !== 'ok') {
        logger.warn('[WhatsApp] review confirm list send failed', { sender: label, outcome });
      }
      track(outcome);
    } catch (e) {
      logger.warn('[WhatsApp] review confirm list unavailable', {
        sender: label, error: e instanceof Error ? e.message : String(e),
      });
    }
  };
  const countNeedsReview = async (): Promise<number> => {
    if (!workspaceId) return 0;
    const { count } = await db.from('invoices').select('id', { count: 'exact', head: true })
      .eq('workspace_id', workspaceId).eq('status', 'needs_review');
    return count ?? 0;
  };
  // "Done" is a real finalization, not just an ack. It re-affirms a register job
  // for every eligible batch, then reads the CURRENT state and sends a TRUTHFUL
  // result. It never claims the Excel is ready until a verified generated_workbooks
  // row exists (build_register inserts that row only after an encrypted upload
  // succeeded), so a ready claim always implies the stored file is available.
  // While extraction/register jobs are still running it says "still working" and
  // defers the final result to the register job's delivery. Repeated Done is
  // idempotent: scheduling is deduped and the reads never mutate processing state.
  // §immediate-register: register generation is an async worker job, so a Done
  // tapped right after approval often finds the workbook still compiling. Wait a
  // SHORT, bounded window for it to land so the common "approve then Done" path can
  // still ATTACH the Excel in this very reply, instead of only reaching the chat on
  // the next inbound. If a register is still running after the window we keep the
  // truthful "still working" reply and the outbox flush delivers on the next inbound.
  // §never-partial: the window ends on QUIESCENCE, not on "a workbook exists". Mid-burst
  // there is always some live register, but it is a PARTIAL snapshot of the month — and
  // the delivery consumer deliberately holds a flush while any job is live, so returning
  // early here attached nothing and taught the owner to type Done twice. A workspace with
  // extraction or register work still running is never "ready" for this reply.
  const waitForRegisterReady = async (scope: DeliveryScope, first: FinalizeCounts): Promise<FinalizeCounts> => {
    if (!workspaceId) return first;
    if (first.outstanding === 0 && first.compiling === 0) return first;
    // Nudge the worker so the queued register gets a tick inside this window
    // (ensureBatchRegisters only wakes when it newly scheduled one).
    wakeWorker();
    let latest = first;
    const deadline = Date.now() + REGISTER_READY_WAIT_MS;
    while (Date.now() < deadline) {
      await new Promise((r) => setTimeout(r, 1200));
      try {
        latest = await collectFinalizeState(db, workspaceId, scope);
      } catch {
        return latest;
      }
      if (latest.outstanding === 0 && latest.compiling === 0) return latest;
    }
    return latest;
  };
  // §truthful-promise: the Done reply may only promise an unprompted handover when
  // the SANCTIONED opt-in path can actually deliver it — consent + the separate
  // sealing key + inside Meta's 24h window (canProactivelyPush), with the sealed
  // number still opening. `pushOffer` is the honest middle state: the operator
  // turned the feature on but this owner has not opted in yet, so naming the
  // "Proactive on" keyword is real help instead of a dead end. Never leaks the
  // phone or the sealed value — only two booleans leave here.
  const proactivePushState = async (): Promise<{ pushOk: boolean; pushOffer: boolean }> => {
    if (!proactiveDeliveryEnabled() || !proactiveKeyConfigured()) {
      return { pushOk: false, pushOffer: false };
    }
    try {
      const id = await loadProactiveIdentity(db, hmac);
      if (!id) return { pushOk: false, pushOffer: false };
      const pushOk = canProactivelyPush({ consent: id.consent, lastActivityMs: id.lastActivityMs })
        && openWaId(id.sealed) != null;
      return { pushOk, pushOffer: !id.consent };
    } catch {
      return { pushOk: false, pushOffer: false };
    }
  };
  const deliverDone = async (): Promise<void> => {
    if (!workspaceId) { await deliverText('Nothing to finalize yet — send Hi to start, or just send an invoice.'); return; }
    // Serve ONLY this sender's tenant: individuals see client-less batches,
    // org senders see their selected client's. An older month still sitting at
    // 'ready' for another client can then never ride along on this reply.
    const scope = { clientId: clientId ?? null };
    try {
      await ensureBatchRegisters(db, workspaceId, scope);
      const counts = await waitForRegisterReady(scope, await collectFinalizeState(db, workspaceId, scope));
      const delivered = await deliverOwedRegisters();
      const base = (process.env.PUBLIC_APP_URL ?? '').replace(/\/$/, '');
      // Deterministic link: the LATEST accounting period among the scoped ready
      // registers — never an arbitrary first row (the wrong-month leak path).
      const primary = pickPrimaryWorkbook(counts.readyMonths ?? [])?.workbookId ?? null;
      const link = base && primary
        ? `${base}/api/exports/${primary}/download`
        : null;
      const { pushOk, pushOffer } = await proactivePushState();
      await deliverText(composeDoneMessage(counts, { link, delivered: delivered > 0, pushOk, pushOffer }));
    } catch (e) {
      // Finalization is read/schedule only; a failure here must never lose work or
      // rewind intake. Fall back to an honest "queued" message (no ready claim).
      logger.warn('[WhatsApp] Done finalization deferred', {
        sender: label, error: e instanceof Error ? e.message : String(e),
      });
      await deliverText('Done — your uploads are queued. I\'ll send your register here once processing finishes.');
    }
  };
  const sendStateMessage = async (): Promise<void> => {
    if (result.note === 'individual-selected') {
      const held = assignedNow > 0 ? ` I also filed the ${assignedNow} invoice file(s) you sent earlier.` : '';
      await deliverText(`Your workspace is ready. Send invoice PDFs or photos here.${held}`);
    } else if (result.note.startsWith('org-name:')) {
      const clients = await listClients(db, workspaceId as string);
      if (clients.length > 0) { track(await sendClientList(to, clients)); }
      else {
        await deliverText('Organization workspace created. No clients yet — send the first client name.');
      }
    } else if (result.note === 'reprompt-client-list' || result.note === 're-list-clients'
      || result.note === 'change-client-requested') {
      if (workspaceId) track(await sendClientList(to, await listClients(db, workspaceId)));
    } else if (result.note.startsWith('client-name:')) {
      await deliverText('Client saved. Send invoice PDFs or photos here. Optional: send "My GSTIN <number>" so I never mistake your own GSTIN for a supplier\'s.');
    } else if (result.note === 'client-selected') {
      await deliverText('Client selected. Send invoice PDFs or photos here. Optional: send "My GSTIN <number>" so I never mistake your own GSTIN for a supplier\'s.');
    } else if (result.note === 'client-selection-rejected') {
      await deliverText('That client is not available. Choose from the list below.');
      if (workspaceId) track(await sendClientList(to, await listClients(db, workspaceId)));
    } else if (result.note === 'gstin-set' || result.note === 'gstin-clear') {
      // Own-GSTIN capture reply (§P0-1b). The set/clear note is normalised by the
      // side-effects block, and the write already happened before this send.
      await deliverText(gstinReplyBody ?? 'Done.');
    } else if (result.note === 'proactive-enable' || result.note === 'proactive-disable') {
      // Proactive opt-in/opt-out reply (§P1-6). Consent was persisted (or softly
      // declined) in the side-effects block before this send.
      await deliverText(proactiveReplyBody ?? 'Done.');
    } else if (result.note === 'awaiting-pending-confirm' || result.note === 'pending-confirm-stale') {
      // noop: handled by the confirm branch below / generic text.
    } else if (result.note === 'pending-confirm-yes') {
      await deliverText(assignedNow > 0
        ? `Applied to ${assignedNow} file(s) — they're queued for processing. Send more invoices anytime.`
        : 'Applied. Nothing was held for you. Send invoice PDFs or photos here.');
    } else if (result.note === 'welcome-back') {
      const waiting = await countWaitingUploads(db, hmac);
      await deliverText(waiting > 0
        ? `Welcome back to Billora. Note: ${waiting} earlier file(s) are still held for a client — send Change client to assign them, or just send more invoices.`
        : 'Welcome back to Billora — send invoice PDFs or photos here. Send Menu for options.');
    } else if (result.note === 'done') {
      await deliverDone();
    } else if (result.note === 'status-requested') {
      const waiting = await countWaitingUploads(db, hmac).catch(() => 0);
      await deliverText(!workspaceId
        ? "You're not set up yet — send Hi to begin, or just send an invoice and I'll hold it safely."
        : `You're ready. Send invoice PDFs or photos anytime.${waiting > 0 ? ` ${waiting} file(s) are held for a client — send Change client to assign them.` : ''} Send Done when this batch is finished, or Menu for options.`);
    } else if (result.note === 'review-requested') {
      const waiting = await countWaitingUploads(db, hmac).catch(() => 0);
      let body: string;
      if (!workspaceId) {
        body = "You're not set up yet — send Hi to begin, or just send an invoice.";
      } else {
        try {
          // Same tenant scope as Done: a Review reply can only ever carry this
          // sender's own client's ready registers, latest period first.
          const scope = { clientId: clientId ?? null };
          const counts = await collectFinalizeState(db, workspaceId, scope);
          const delivered = await deliverOwedRegisters();
          const base = (process.env.PUBLIC_APP_URL ?? '').replace(/\/$/, '');
          const primary = pickPrimaryWorkbook(counts.readyMonths ?? [])?.workbookId ?? null;
          const link = base && primary
            ? `${base}/api/exports/${primary}/download`
            : null;
          body = composeReviewMessage(counts, { link, delivered: delivered > 0 });
        } catch {
          const flagged = await countNeedsReview().catch(() => 0);
          body = flagged > 0
            ? `${flagged} invoice(s) flagged for review.`
            : 'Nothing needs your attention right now — keep sending invoices.';
        }
      }
      // §P0-1c: when the active client never set its own (buyer) GSTIN, a
      // recipient GSTIN recurring on >=3 of their bills is a strong candidate —
      // but a machine must NEVER silently rewrite a fact that drives the
      // supplier==buyer guard. Surface a plain-English SUGGESTION and let the
      // reviewer confirm it (reply `My GSTIN <value>`). Best-effort: a missing
      // key, read or decrypt failure leaves the truthful reply untouched.
      if (workspaceId && clientId) {
        try {
          const { masterKeyHex: suggestKey } = resolveFileKeyConfig();
          const hint = await suggestOwnGstin(db, { workspaceId, clientId, masterKeyHex: suggestKey, minCount: 3 });
          if (hint.suggestion) {
            body += `\n\nAlso: ${hint.suggestion} looks like it might be your own GSTIN — it appears as the recipient on ${hint.count} of your invoices. If that's right, reply "My GSTIN ${hint.suggestion}" and I'll never file it as a supplier. If not, just ignore this.`;
          }
        } catch { /* suggestion is a bonus; never break the review reply */ }
      }
      const held = waiting > 0 ? ` Also: ${waiting} file(s) are held for a client — send Change client to assign them.` : '';
      await deliverText(body + held);
      // §P1-7: one confirmable page (with an "Approve all" control + a "Next" row
      // when more remain) instead of a single unbounded 10-row list. Best-effort.
      // §user #2: ONE combined Review & edit screen. The list rows now carry
      // `billora:review:edit:<id>` ids, so tapping any invoice OPENS the combined
      // editor (view every field, correct any field, then reply APPROVE) — the
      // separate edit picker is gone. `listPendingReviewPage` widens its eligibility
      // to cover EVERY needs_review invoice (including unbatched / numberless ones
      // whose only problem is a missing date or number), so nothing becomes
      // unreachable now that the second list is removed. Best-effort: a read/send
      // failure leaves the truthful text guidance above as the only reply.
      if (workspaceId) {
        await sendReviewPage(1);
      }
    } else if (result.note === 'review-page') {
      // §P1-7: page through a review list longer than the 10-row cap. No re-summary
      // (the reviewer already saw it on the first Review) — just the requested page.
      await sendReviewPage(result.page ?? 2);
    } else if (result.note === 'review-approve-all') {
      // §P1-7: bulk approval already ran (with re-validation) in the side-effects
      // block; deliver the honest counts, then re-show any row that stayed flagged.
      await deliverText(approveAllSummary ?? 'Done.');
      await sendReviewPage(1);
    } else if (result.note === 'review-confirm') {
      await deliverText(composeConfirmResultMessage(confirmOutcome ?? { ok: false, error: 'FAULT' }));
    } else if (result.note === 'review-confirm-invalid') {
      await deliverText('That confirmation is no longer valid — send Review to see the current list.');
    } else if (result.note.startsWith('review-edit')) {
      // One text reply for the whole editor turn (open / correct / approve /
      // cancel / help). Already computed during side effects, BEFORE persist.
      await deliverText(editReplyBody ?? REVIEW_EDIT_HELP);
    } else if (result.outbound.kind === 'menu_list') {
      await deliverMenu(result.outbound.workspace);
    } else if (result.outbound.kind === 'text') {
      await deliverText(result.outbound.body);
    } else if (result.outbound.kind === 'account_type_buttons') {
      track(await sendAccountTypeButtons(to));
    }
    // Held files + a just-resolved client: the explicit confirmation question.
    if (result.nextState === 'awaiting_pending_confirm' && pendingClientId && workspaceId) {
      const waiting = await countWaitingUploads(db, hmac);
      const clients = await listClients(db, workspaceId);
      const chosen = clients.find((c) => c.id === pendingClientId);
      track(await sendConfirmApply(to, waiting, chosen?.name ?? 'the selected client'));
    }
  };
  await sendStateMessage();
  if (sendOutcome !== 'ok') {
    logger.warn('[WhatsApp] outbound send deferred; processing state persisted', {
      sender: label, outcome: sendOutcome,
    });
  }
  return { workspaceId: result.workspaceId ?? workspaceId, clientId: result.clientId ?? clientId };
}

export async function POST(request: Request) {
  try {
    const body = await readRequestBodyLimited(request, WEBHOOK_BODY_MAX_BYTES);
    if (!body) {
      return NextResponse.json({ status: 'SKIPPED_EMPTY_BODY' }, { status: 200 });
    }
    const rawText = body.toString('utf8');
    if (!rawText || rawText.trim().length === 0) {
      return NextResponse.json({ status: 'SKIPPED_EMPTY_BODY' }, { status: 200 });
    }
    // Signature gate: X-Hub-Signature-256 over the RAW body, verified before any
    // intake row is created. Production fails closed when the app secret is
    // unconfigured; dev without a secret proceeds (local/smoke testing only).
    const appSecret = process.env.WHATSAPP_APP_SECRET || '';
    if (!appSecret && process.env.NODE_ENV === 'production') {
      logger.error('[WhatsApp] WHATSAPP_APP_SECRET unconfigured in production; delivery rejected');
      return new Response('Forbidden', { status: 401 });
    }
    if (appSecret && !verifyWebhookSignature(rawText, request.headers.get('x-hub-signature-256'), appSecret)) {
      logger.warn('[WhatsApp] webhook signature invalid; delivery rejected');
      return new Response('Forbidden', { status: 401 });
    }
    let payload: { entry?: Array<{ changes?: Array<{ value?: {
      messages?: IncomingMessage[]; statuses?: DeliveryStatus[];
    } }> }> };
    try { payload = JSON.parse(rawText); }
    catch {
      logger.info('[WhatsApp] non-JSON body skipped', {
        preview: rawText.slice(0, 200), contentType: request.headers.get('content-type'),
      });
      return NextResponse.json({ status: 'SKIPPED_NON_JSON_BODY' }, { status: 200 });
    }
    const changes = (payload.entry ?? []).flatMap((entry) => entry.changes ?? []);
    const values = changes.map((change) => change.value).filter((value) => value !== undefined);
    const messages = values.flatMap((value) => value?.messages ?? []);
    const statuses = values.flatMap((value) => value?.statuses ?? []);
    for (const delivery of statuses) {
      logger.info('[WhatsApp] delivery status received', {
        providerMessageIdSuffix: delivery.id?.slice(-8) ?? null,
        status: delivery.status ?? 'unknown',
        timestamp: delivery.timestamp ?? null,
        errorCodes: (delivery.errors ?? []).map((error) => error.code).filter((code) => code !== undefined),
      });
    }
    if (messages.length === 0) {
      // Delivery receipts/statuses carry no inbound message to process.
      logger.info('[WhatsApp] non-message delivery skipped', {
        hasMessages: messages.length > 0, statusCount: statuses.length,
      });
      return NextResponse.json({ status: 'SKIPPED_STATUS_UPDATE' }, { status: 200 });
    }
    const db = getSupabaseAdmin();
    // Persist every normalized message before claiming or doing synchronous
    // intake work. The worker can replay these encrypted envelopes if this
    // invocation dies and Meta does not redeliver the webhook.
    for (const message of messages) {
      if (!message?.id) continue;
      await storeReplayableWebhookMessage(db, message.id, { ...message, id: message.id });
    }
    // Process EVERY message in the delivery: a 503 makes Meta redeliver the
    // whole payload and the intake ledger skips messages already accepted.
    let retryable = false;
    for (const message of messages) {
      if (!message?.id) continue;
      if (await handleSingleMessage(db, message) === 'retryable') retryable = true;
    }

    async function readRequestBodyLimited(request: Request, maxBytes: number): Promise<Buffer | null> {
      const declaredLength = Number(request.headers.get('content-length'))
      if (Number.isFinite(declaredLength) && declaredLength > maxBytes) {
        throw new Error('WEBHOOK_BODY_TOO_LARGE')
      }
      if (!request.body) return null

      const reader = request.body.getReader()
      const chunks: Buffer[] = []
      let total = 0
      try {
        while (true) {
          const { done, value } = await reader.read()
          if (done) break
          total += value.byteLength
          if (total > maxBytes) {
            await reader.cancel('webhook body exceeds limit')
            throw new Error('WEBHOOK_BODY_TOO_LARGE')
          }
          chunks.push(Buffer.from(value))
        }
      } finally {
        reader.releaseLock()
      }
      return total === 0 ? null : Buffer.concat(chunks, total)
    }
    if (retryable) {
      return NextResponse.json({ status: 'RETRYABLE_PROCESSING_ERROR' }, { status: 503 });
    }
    return NextResponse.json({ status: 'ACCEPTED_FOR_QUEUE' }, { status: 200 });
  } catch (error: unknown) {
    const detail = error instanceof Error ? `${error.name}: ${error.message}` : String(error);
    logger.error('[Gateway Interface Router] runtime anomaly', { error: detail });
    const code = /Missing Supabase env|Invalid environment/i.test(detail) ? 'CONFIG_ERROR'
      : /webhook_events|relation .* does not exist/i.test(detail) ? 'DB_TABLE_MISSING'
      : 'SYSTEM_ROUTING_FAULT';
    const retryable = /CONFIG_ERROR|DB_TABLE_MISSING/.test(code) ? 503 : 500;
    return NextResponse.json({ status: code, detail }, { status: retryable });
  }
}

type MessageOutcome = 'processed' | 'retryable'

/**
 * Claim -> process -> durably accept exactly one provider message. Only an
 * `accepted` row suppresses reprocessing; `received`/`failed` rows reprocess
 * on Meta redelivery. Retries stay idempotent end to end.
 */
async function handleSingleMessage(db: Db, message: IncomingMessage): Promise<MessageOutcome> {
  const providerMessageId: string = message.id as string;
  const shortId = providerMessageId.slice(-8);
  logger.info('[WhatsApp] webhook received', {
    hasMessages: true, shortId, messageType: message.type ?? null, statuses: null,
  });
  let intake: Awaited<ReturnType<typeof claimIntakeRow>>;
  try {
    intake = await claimIntakeRow(db, providerMessageId);
  } catch (e) {
    const detail = e instanceof Error ? e.message : String(e);
    logger.error('[WhatsApp] intake claim failed', { shortId, error: detail });
    return 'retryable';
  }
  if (!intake.claimed) {
    if (isDurablyAccepted(intake.row)) {
      logger.info('[WhatsApp] duplicate accepted event skipped', { shortId });
      return 'processed';
    }
    // Another invocation owns the live lease. A 503 asks Meta to retry; returning
    // 200 could strand this event if its current owner crashes before acceptance.
    logger.info('[WhatsApp] duplicate delivery deferred; active claim exists', {
      shortId, priorStatus: intake.row.status, leaseExpiresAt: intake.row.claim_expires_at,
    });
    return 'retryable';
  }
  const claimToken = intake.claimToken;
  if (!claimToken) {
    logger.error('[WhatsApp] acquired intake claim has no owner token', { shortId });
    return 'retryable';
  }
  if (!intake.created) logger.info('[WhatsApp] retry acquired exclusive claim', {
    shortId, priorStatus: intake.row.status, attempt: intake.row.attempts,
  });
  let claimLost = false;
  const heartbeat = setInterval(() => {
    void renewIntakeClaim(db, providerMessageId, claimToken).then((renewed) => {
      if (!renewed) {
        claimLost = true;
        logger.error('[WhatsApp] intake claim heartbeat rejected; owner is stale', { shortId });
      }
    }).catch((heartbeatError) => {
      logger.warn('[WhatsApp] intake claim heartbeat failed', {
        shortId, error: heartbeatError instanceof Error ? heartbeatError.message : 'unknown',
      });
    });
  }, 45_000);
  heartbeat.unref?.();
  // A crash leaves a leased processing row; Meta redelivery can reclaim it after expiry.
  try {
    await processMessage(db, message, providerMessageId, shortId);
    if (claimLost) throw new Error('INTAKE_CLAIM_LOST: heartbeat rejected this invocation');
    await markIntakeAccepted(db, providerMessageId, claimToken);
  } catch (e) {
    const detail = e instanceof Error ? `${e.name}: ${e.message}` : String(e);
    const code = (e instanceof Error ? e.message : detail).split(':')[0] || 'PROCESSING_ERROR';
    if (isPermanentMediaFailure(e)) {
      // Expired/blocked credential: 503 would loop Meta redeliveries forever
      // and a retry can never succeed. Accept the event as final (audit code
      // recorded), alert the operator, and ask the sender to resend once the
      // token is reissued. Delivery itself is best-effort (rule 5).
      logger.error('[WhatsApp] MEDIA_CREDENTIAL_BLOCKED: media unrecoverable with current token; operator must reissue WHATSAPP_ACCESS_TOKEN', {
        shortId, error: code, detail: detail.slice(0, 200),
      });
      try {
        await markIntakeAccepted(db, providerMessageId, claimToken, detail.slice(0, 120));
      } catch (claimError) {
        logger.warn('[WhatsApp] permanent media failure could not settle stale claim', {
          shortId, error: claimError instanceof Error ? claimError.message : 'unknown',
        });
        return 'retryable';
      }
      if (message.from) {
        await sendText(message.from, 'I could not retrieve that file right now (a system credential expired). Please resend the invoice in a few minutes.').catch(() => undefined);
      }
      return 'processed';
    }
    if (isEncryptionConfigFailure(e)) {
      // Encryption keys are a deploy-time contract: a missing/malformed key is
      // an operator-configuration error that NO retry can fix. Returning 503
      // would make Meta redeliver the message in a loop, so accept this
      // delivery as final with a stable audit code (webhook_events stays the
      // idempotency ledger: a genuine later redelivery of an ACCEPTED event is
      // skipped, so the sender below is notified at most once per message),
      // alert the operator, and tell the sender the truth: nothing was saved.
      logger.error('[WhatsApp] ENCRYPTION_CONFIG: file encryption is not configured; nothing was stored — operator must set ENCRYPTION_MASTER_KEY + STORAGE_PATH_KEY and restart', {
        shortId, code: 'ENCRYPTION_CONFIG',
      });
      try {
        await markIntakeAccepted(db, providerMessageId, claimToken, 'ENCRYPTION_CONFIG');
      } catch (claimError) {
        logger.warn('[WhatsApp] encryption config failure could not settle stale claim', {
          shortId, error: claimError instanceof Error ? claimError.message : 'unknown',
        });
        return 'retryable';
      }
      if (message.from) {
        await sendText(message.from, 'I could not file that invoice — a required system setting is missing on our side, so nothing was saved. The account owner needs to finish setup; please resend afterwards.').catch(() => undefined);
      }
      return 'processed';
    }
    if (isStorageConfigFailure(e)) {
      // The private bucket is not yet configured to accept the encrypted
      // container's transport type (application/octet-stream). This is an
      // operator/schema-delivery gap that no retry can fix: returning 503 would
      // loop Meta redeliveries and re-download the same media forever. Accept as
      // final with a stable audit code, alert the operator, and tell the sender
      // the truth — nothing was saved.
      logger.error('[WhatsApp] STORAGE_CONFIG: private storage rejected the encrypted upload; operator must apply the invoices bucket migration', {
        shortId, code: 'STORAGE_CONFIG',
      });
      try {
        await markIntakeAccepted(db, providerMessageId, claimToken, 'STORAGE_CONFIG');
      } catch (claimError) {
        logger.warn('[WhatsApp] storage config failure could not settle stale claim', {
          shortId, error: claimError instanceof Error ? claimError.message : 'unknown',
        });
        return 'retryable';
      }
      if (message.from) {
        await sendText(message.from, 'I could not file that invoice — our private storage is mid-setup, so nothing was saved. The account owner needs to finish a configuration step; please resend afterwards.').catch(() => undefined);
      }
      return 'processed';
    }
    await markIntakeFailed(db, providerMessageId, claimToken, code).catch(() => undefined);
    logger.error('[WhatsApp] processing failed, retryable', { shortId, error: code, detail: detail.slice(0, 200) });
    return 'retryable';
  } finally {
    clearInterval(heartbeat);
  }
  logger.info('[WhatsApp] accepted for queue', { shortId });
  return 'processed';
}

/**
 * Best-effort resend of a reply that failed delivery earlier (rule 5 decouples
 * outbound from inbound). Fires at the top of a NEW inbound message and only
 * sends — it never re-runs intake or onboarding side effects, so a completed
 * business operation cannot be duplicated. Cleared on success; left stored on
 * any non-ok so the next message retries. Never throws.
 */
async function resendPendingReply(db: Db, hmac: string, to: string): Promise<void> {
  try {
    const { data } = await db.from('whatsapp_sessions').select('pending_reply_body')
      .eq('sender_hmac', hmac).maybeSingle();
    const rawBody = (data?.pending_reply_body as string | null) ?? null;
    if (!rawBody) return;
    // A register handoff carries one or more leading [[deliver:<uuid>]] tokens —
    // one per ready workbook that finished since the last inbound (a multi-month
    // burst can queue several). Deliver EACH decrypted file into the chat; the web
    // link is unreachable for a WhatsApp-only owner (no login exists). Strip every
    // internal token so any text fallback shows only human copy, and fall back to
    // the text link only if document delivery is not possible.
    const workbookIds = [...rawBody.matchAll(/\[\[deliver:([0-9a-f-]{36})\]\]/gi)].map((m) => m[1]);
    const body = rawBody.replace(/\s*\[\[deliver:[0-9a-f-]{36}\]\]/gi, '').trim();
    if (workbookIds.length > 0) {
      const outstanding: string[] = [];
      let deliveredCount = 0;
      for (const workbookId of workbookIds) {
        const r = await deliverWorkbookToWhatsApp(db, { workbookId, to });
        if (r.delivered) { deliveredCount += 1; continue }
        // NOT yet in the chat but could succeed later: keep THIS handoff so the
        // next inbound retries the document. Falling through here would send the
        // reassuring "I'll send the Excel" text and, on a successful text send,
        // clear the slot — stranding a 'ready' workbook forever with the file
        // never arriving. A 'skipped' (gone/expired/not-ready) workbook has nothing
        // to retry, so it is simply dropped from the set below.
        if (r.outcome === 'transient' || r.outcome === 'blocked') outstanding.push(workbookId);
      }
      if (outstanding.length > 0) {
        // Re-arm the slot with ONLY the still-outstanding tokens and stay silent;
        // the current message is still handled normally, and these retry next time.
        const kept = outstanding.map((id) => `[[deliver:${id}]]`).join('\n');
        await updateSessionDeliveryState(db, hmac, { pending_reply_body: `${kept}\n${body}`.trim(), last_reply_error_code: null });
        logger.warn('[WhatsApp] some deferred registers still pending; retrying next inbound', { outstanding: outstanding.length });
        return;
      }
      // Every register landed or is gone. When at least one file arrived, the
      // document IS the reply — clear the handoff and say nothing further.
      await updateSessionDeliveryState(db, hmac, { pending_reply_body: null, last_reply_error_code: null });
      if (deliveredCount > 0) {
        logger.info('[WhatsApp] deferred register(s) delivered as document', { count: deliveredCount });
        return;
      }
      // All were 'skipped' (nothing landed): fall through to the one-time text.
    }
    const o = await sendText(to, body, workbookIds[0] ? { workbookId: workbookIds[0] } : undefined);
    if (o === 'ok') {
      await updateSessionDeliveryState(db, hmac, { pending_reply_body: null, last_reply_error_code: null });
      logger.info('[WhatsApp] deferred reply resent');
    }
  } catch { /* delivery is best-effort; ignore and try again next message */ }
}

/**
 * Onboarding + upload decision engine for one message. Throws only when the
 * work is NOT durably recorded (caller marks intake failed -> Meta retries).
 * Media lifecycle: validate type -> store privately + SHA-256 -> invoice
 * (received) + job (queued) — or a pending_uploads hold when workspace/client
 * is unresolved ("receive first, ask later"). Held paths are deterministic
 * per provider media id, so webhook redelivery never duplicates a hold.
 */
async function processMessage(
  db: Db, message: IncomingMessage, providerMessageId: string, shortId: string,
): Promise<void> {
  const replyId = message.interactive?.button_reply?.id || message.interactive?.list_reply?.id || null;
  const textBody = message.text?.body ?? null;
  // Decoupled outbound retry: if a previous reply failed delivery, resend it
  // now (send only — no business operation re-runs), before normal processing.
  if (message.from) {
    try { await resendPendingReply(db, senderHmac(message.from), message.from); } catch { /* best effort */ }
    // Durable outbox flush: send every workbook this owner is currently owed via
    // the atomic-claim state machine (replaces the racy single-slot handoff). Send
    // only — never re-runs intake/onboarding. Best-effort; a failure leaves the
    // rows pending for the next inbound or the worker recovery sweep.
    try { await flushPendingDeliveries(db, senderHmac(message.from), message.from); } catch { /* best effort */ }
  }
  let onboardingWorkspaceId: string | null = null;
  let onboardingClientId: string | null = null;
  if ((message.type === 'text' || message.type === 'interactive') && message.from) {
    const hmac = senderHmac(message.from);
    const label = senderLabel(hmac);
    const stored = await loadSession(db, hmac);
    let snapshot: SessionSnapshot = isExpired(stored.updated_at)
      ? expiredSnapshot(stored, hmac)
      : { senderHmac: hmac, state: stored.state as ConversationState,
          workspaceId: stored.workspace_id, clientId: stored.client_id,
          batchId: stored.batch_id, pendingClientId: stored.pending_client_id ?? null,
          reviewEditInvoiceId: stored.review_edit_invoice_id ?? null,
          lastProviderMessageId: stored.last_provider_message_id,
          expired: false };
    // Session row lost its linkage but the identity is onboarded: backfill so
    // returning users never repeat setup.
    if (!snapshot.workspaceId) {
      const linked = await db.from('whatsapp_identities').select('workspace_id')
        .eq('sender_hmac', hmac).maybeSingle();
      const linkedWs = (linked.data?.workspace_id as string | null) ?? null;
      if (linkedWs) snapshot = { ...snapshot, workspaceId: linkedWs };
    }
    const result = transition({ session: snapshot, replyId, text: textBody, providerMessageId });
    logger.info('[WhatsApp] onboarding transition', { sender: label, note: result.note });
    const out = await handleOnboardingSideEffects(db, hmac, label, message.from, providerMessageId, snapshot, result);
    onboardingWorkspaceId = out.workspaceId;
    onboardingClientId = out.clientId;
  }
  const media = message.type === 'image' ? message.image
    : message.type === 'document' ? message.document : undefined;
  const mediaId = media?.id;
  if (!mediaId) {
    if (message.type === 'image' || message.type === 'document') {
      logger.info('[WhatsApp] media message without media id', { shortId });
      throw new Error('MEDIA_MISSING_ID: retryable media without provider id');
    }
    return;
  }
  if (!message.from) throw new Error('MEDIA_MISSING_SENDER: media without from address');
  // Encryption keys are required before ANY file handling, so validate them
  // FIRST: a misconfigured deploy must not pay for a media download it can
  // never store (fail-closed: plaintext is never written, retries never help).
  const { masterKeyHex, pathKeyHex } = resolveFileKeyConfig();
  logger.info('[WhatsApp] media intake start', { shortId });
  const mHmac = senderHmac(message.from);
  const mSession = await loadSession(db, mHmac);

  // Download, then validate the ACTUAL bytes. WhatsApp's declared mime /
  // filename / extension are hints only — the magic-byte sniff is authoritative,
  // so a mislabeled, unsupported or oversized file is rejected here: nothing is
  // stored, no DB error is raised, and the intake settles accepted (a permanent
  // format/size failure must NEVER trigger a Meta retry / re-download loop).
  let downloaded: DownloadedMedia;
  try {
    downloaded = await downloadWhatsAppMedia({
      mediaId, accessToken: process.env.WHATSAPP_ACCESS_TOKEN || '', declaredMime: media?.mime_type,
    });
  } catch (e) {
    if (isPermanentMediaFormatFailure(e)) {
      logger.info('[WhatsApp] media rejected by declared size', { shortId, reason: (e as Error).message });
      await sendText(message.from, (e as { userReply: string }).userReply);
      return;
    }
    throw e;
  }
  let validated: ValidatedMedia;
  try {
    validated = validateInvoiceMedia({
      buffer: downloaded.buffer, declaredMime: downloaded.contentType, declaredFilename: media?.filename ?? null,
    });
  } catch (e) {
    if (isPermanentMediaFormatFailure(e)) {
      logger.info('[WhatsApp] unsupported or oversized media rejected', { shortId, reason: (e as Error).message });
      await sendText(message.from, (e as { userReply: string }).userReply);
      return;
    }
    throw e;
  }
  // Persist the VALIDATED original MIME + extension (never the provider hint) so
  // extraction routes on the true format. The stored blob is always the opaque
  // octet-stream container; the original bytes stay intact inside it.
  downloaded.contentType = validated.mime;
  downloaded.extension = validated.extension;
  const sha256Hex = createHash('sha256').update(downloaded.buffer).digest('hex');
  const workspaceId = onboardingWorkspaceId ?? mSession.workspace_id;
  const clientId = onboardingClientId ?? mSession.client_id;

  // Envelope-encrypt the ORIGINAL bytes before they ever touch storage (keys
  // were validated above, before the download). The SHA-256 above stays over
  // plaintext for dedup + integrity; the stored blob is the self-describing
  // container.
  const invoiceBucket = bucketFor('invoices');

  // Deterministic private hold path: redelivery of the same provider media id
  // resolves to the same opaque object + upserts the same row — never a
  // duplicate hold, and never collides with an unrelated file.
  const holdMedia = async (): Promise<string | null> => {
    const pendingPath = deriveObjectKey('pending', mediaId, pathKeyHex);
    const { WHATSAPP_MAX_INVOICES_PER_DAY } = getEnv();
    const reservation = await reserveDailyIntake(db, {
      providerMessageId,
      workspaceId,
      senderHmac: mHmac,
      limit: WHATSAPP_MAX_INVOICES_PER_DAY,
    });
    if (!reservation.allowed) {
      logger.info('[WhatsApp] held media blocked: daily intake limit reached', {
        shortId, limit: WHATSAPP_MAX_INVOICES_PER_DAY,
      });
      await sendText(message.from!, `You've reached today's limit of ${WHATSAPP_MAX_INVOICES_PER_DAY} invoices. Please try again tomorrow.`);
      return null;
    }
    await beginStorageIntake(db, { providerMessageId, objectPath: pendingPath, kind: 'pending' });
    const enc = encryptFile(downloaded.buffer, masterKeyHex, { context: pendingPath });
    const { error: pendingUploadError } = await db.storage.from(invoiceBucket)
      .upload(pendingPath, enc.container, {
        contentType: 'application/octet-stream', upsert: true,
        metadata: { original_mime: validated.mime, original_extension: validated.extension },
      });
    if (pendingUploadError) throw new Error(`STORAGE_UPLOAD_ERROR: ${pendingUploadError.message}`);
    await markStorageIntake(db, pendingPath, 'uploaded');
    await recordPendingUpload(db, {
      senderHmac: mHmac, objectPath: pendingPath, originalFileName: media?.filename ?? null,
      contentType: downloaded.contentType, fileSha256Hex: sha256Hex,
      sizeBytes: enc.sizeBytes, providerMediaId: mediaId, providerMessageId, keyVersion: enc.keyVersion,
    });
    await commitDailyIntake(db, providerMessageId, reservation.scopeKey);
    await markStorageIntake(db, pendingPath, 'linked');
    return pendingPath;
  };

  if (!workspaceId) {
    // Account type unknown: preserve the file, then continue/ask setup.
    if (!(await holdMedia())) return;
    if (mSession.state === 'awaiting_account_type') {
      await sendText(message.from, 'I saved that invoice safely. Finish setup above — pick Individual or Organization — and I will file it automatically.');
    } else {
      await sendText(message.from, 'I saved that invoice safely. Before I file it: are you onboarding as an Individual or an Organization / CA firm?');
      await sendAccountTypeButtons(message.from);
    }
    return;
  }

  const accountType = await loadAccountType(db, mHmac);
  if (accountType === 'organization' && !clientId) {
    // Client unresolved: hold and ask. NEVER guess a client.
    if (!(await holdMedia())) return;
    await saveSession(db, mHmac, mSession.last_provider_message_id, {
      state: 'awaiting_client_select', workspace_id: workspaceId, client_id: clientId,
      batch_id: mSession.batch_id, pending_client_id: mSession.pending_client_id,
      last_provider_message_id: providerMessageId,
      expires_at: new Date(Date.now() + 24 * 60 * 60 * 1000).toISOString(),
    }).catch(() => undefined); // best effort: the file is durably held either way
    await sendText(message.from, 'I saved that invoice — but I must not guess which client it belongs to. Choose a client (or add one) below.');
    await sendClientList(message.from, await listClients(db, workspaceId));
    return;
  }

  // Assigned path is deterministic too (per provider media id): combined with
  // the invoices.file_object_path unique constraint, a redelivery can never
  // store or queue the same file twice, whatever the retry timing.
  //
  // Exact-file deduplication is status-aware. Keep the duplicate guard, but do
  // not tell customers that nothing was lost when the original row is failed,
  // awaiting review, or not yet included in a register.
  const duplicateFileInvoice = await findLiveInvoiceByFileHash(db, {
    workspaceId, clientId, fileSha256Hex: sha256Hex,
  });
  if (duplicateFileInvoice) {
    logger.info('[WhatsApp] exact-file duplicate skipped at intake', {
      shortId, invoiceId: duplicateFileInvoice.id.slice(0, 8),
      status: duplicateFileInvoice.status, hasBatch: !!duplicateFileInvoice.batchId,
    });
    const duplicateReply = duplicateFileInvoice.status === 'needs_review'
      ? 'I already received this exact file, so I did not create a duplicate. It still needs your review — send Review to correct or approve it.'
      : duplicateFileInvoice.status === 'failed'
        ? 'I already received this exact file, but its previous processing attempt failed. I have not created a duplicate. Please contact support with this message so the existing record can be recovered safely.'
        : duplicateFileInvoice.status === 'received' || duplicateFileInvoice.status === 'queued' || duplicateFileInvoice.status === 'processing'
          ? 'I already received this exact file and it is still in the processing queue. I did not create a duplicate. Send Done later to check the register.'
          : duplicateFileInvoice.status === 'completed'
            ? 'I already received and processed this exact file, so I did not add it again. If it is missing from your Excel register, send Done to refresh the register; otherwise you can send a different invoice.'
            : 'I already received this exact file, so I did not add it again. Send Done to check your register.';
    await sendText(message.from, duplicateReply);
    return;
  }
  const objectPath = deriveObjectKey('invoices', mediaId, pathKeyHex);
  const { WHATSAPP_MAX_INVOICES_PER_DAY } = getEnv();
  const reservation = await reserveDailyIntake(db, {
    providerMessageId, workspaceId, senderHmac: mHmac,
    limit: WHATSAPP_MAX_INVOICES_PER_DAY,
  });
  if (!reservation.allowed) {
    logger.info('[WhatsApp] media intake blocked: daily invoice limit reached', { shortId, limit: WHATSAPP_MAX_INVOICES_PER_DAY });
    await sendText(message.from, `You've reached today's limit of ${WHATSAPP_MAX_INVOICES_PER_DAY} invoices. Please try again tomorrow — anything you already sent is safe and still processing.`);
    return;
  }
  await beginStorageIntake(db, { providerMessageId, objectPath, kind: 'invoice' });
  const enc = encryptFile(downloaded.buffer, masterKeyHex, { context: objectPath });
  const { error: uploadError } = await db.storage.from(invoiceBucket)
    .upload(objectPath, enc.container, {
      contentType: 'application/octet-stream', upsert: true,
      metadata: { original_mime: validated.mime, original_extension: validated.extension },
    });
  if (uploadError) throw new Error(`STORAGE_UPLOAD_ERROR: ${uploadError.message}`);
    await markStorageIntake(db, objectPath, 'uploaded');
    const recorded = await recordMediaIntake(db, {
    workspaceId, clientId, objectPath,
    originalFileName: media?.filename ?? null, contentType: downloaded.contentType,
    fileSha256Hex: sha256Hex, keyVersion: enc.keyVersion, sizeBytes: enc.sizeBytes,
  });
  await commitDailyIntake(db, providerMessageId, reservation.scopeKey);
  await markStorageIntake(db, objectPath, 'linked');
  const waiting = await countWaitingUploads(db, mHmac).catch(() => 0);
  const deduped = recorded.invoicePreexisting || recorded.jobPreexisting;
  logger.info('[WhatsApp] media intake recorded', {
    shortId, size: enc.sizeBytes,
    invoice: recorded.invoiceId.slice(0, 8), job: recorded.jobId.slice(0, 8),
    deduped,
  });
  // Acknowledge a freshly-filed invoice so a clean upload is never silent
  // (rule 5: send is best-effort — a delivery failure never rewinds the
  // accepted intake). A webhook redelivery dedups to the same invoice/job and
  // stays quiet, so Meta retries never double-ping the sender. §user: a burst of
  // files uploaded together arrives as separate near-simultaneous webhooks, so a
  // plain session read would let EVERY worker ack — the claim is an atomic
  // conditional update (sessions.claimUploadAck) so exactly ONE webhook in the
  // burst sends the confirmation and the rest stay silent. The slot re-arms after
  // the coalesce window so a genuinely later batch still gets one ack.
  if (!deduped) {
    const wonAck = await claimUploadAck(db, mHmac, UPLOAD_ACK_COALESCE_MS);
    if (wonAck) {
      const held = waiting > 0 && clientId
        ? ` Note: ${waiting} earlier file(s) are still held for a client — send Change client to assign them.`
        : '';
      await sendText(message.from, `Invoice received and queued for processing.${held} Send more anytime, or type Done / Menu.`);
    }
  }
}

/**
 * A storage upload rejected specifically because the object's transport MIME is
 * not permitted on the bucket (the encrypted container is uploaded as
 * application/octet-stream). That is a bucket-configuration gap the operator
 * fixes by applying the invoices bucket migration — retrying the same upload
 * can never succeed, so it must NOT loop Meta redeliveries. Deliberately narrow:
 * only a MIME/"not allowed"-style rejection matches, so transient network and
 * permission errors stay retryable.
 */
function isStorageConfigFailure(error: unknown): boolean {
  const message = error instanceof Error ? error.message : String(error);
  if (!/STORAGE_UPLOAD_ERROR/.test(message)) return false;
  return /(mime|content[ -]?type|file type)/i.test(message)
    && /(not allowed|not supported|unsupported)/i.test(message);
}

// lib/whatsapp/intake.ts — durable webhook intake ledger + invoice/job lifecycle.
// webhook_events is the idempotency + recovery ledger:
//   received  = legacy/unclaimed row awaiting its first atomic claim
//   processing = exclusively claimed with a renewable, expiring lease
//   accepted  = onboarding state + (media: invoice + job rows) persisted
//   failed    = attempt failed before durable acceptance; Meta retry can reclaim.
// Only `accepted` suppresses reprocessing. Claim-token fencing prevents stale writes.
import type { SupabaseClient } from '@supabase/supabase-js'
import type { IntakeStatus } from '@/lib/statuses'
import { randomUUID } from 'node:crypto'
import { logger } from '../logger'
import { wakeWorker } from '../jobs/wake'

export type { IntakeStatus } from '@/lib/statuses'

export interface IntakeRow {
  id: string
  provider_message_id: string
  status: IntakeStatus
  attempts: number
  processed_at: string | null
  claim_expires_at: string | null
}

const INTAKE_CLAIM_LEASE_SECONDS = 180

type IntakeClaimRpcRow = {
  claimed: boolean
  created: boolean
  event_id: string
  provider_message_id: string
  status: IntakeStatus
  attempts: number
  processed_at: string | null
  claim_expires_at: string | null
}

type Db = SupabaseClient

/**
 * Atomic PostgreSQL claim. Only the winning invocation receives its UUID token.
 * Expired claims can be reclaimed; all later writes are fenced by that token.
 */
export async function claimIntakeRow(
  db: Db,
  providerMessageId: string,
): Promise<{ row: IntakeRow; created: boolean; claimed: boolean; claimToken: string | null }> {
  const claimToken = randomUUID()
  const { data, error } = await db.rpc('claim_whatsapp_webhook_event', {
    p_provider_message_id: providerMessageId,
    p_claim_token: claimToken,
    p_lease_seconds: INTAKE_CLAIM_LEASE_SECONDS,
  }) as unknown as { data: IntakeClaimRpcRow[] | null; error: { message: string } | null }
  if (error) throw new Error(`INTAKE_CLAIM_ERROR: ${error.message}`)
  const result = Array.isArray(data) ? data[0] : null
  if (!result?.event_id || typeof result.claimed !== 'boolean') {
    throw new Error('INTAKE_CLAIM_ERROR: database returned an invalid claim result')
  }
  return {
    row: {
      id: result.event_id,
      provider_message_id: result.provider_message_id,
      status: result.status,
      attempts: result.attempts,
      processed_at: result.processed_at,
      claim_expires_at: result.claim_expires_at,
    },
    created: result.created,
    claimed: result.claimed,
    claimToken: result.claimed ? claimToken : null,
  }
}

/** True only when a previous delivery durably accepted this message. */
export function isDurablyAccepted(row: IntakeRow): boolean {
  return row.status === 'accepted' && row.processed_at !== null
}

/**
 * Renew only an unexpired claim. Expired owners cannot resurrect their lease.
 */
export async function renewIntakeClaim(db: Db, providerMessageId: string, claimToken: string): Promise<boolean> {
  const { data, error } = await db.rpc('renew_whatsapp_webhook_event_claim', {
    p_provider_message_id: providerMessageId,
    p_claim_token: claimToken,
    p_lease_seconds: INTAKE_CLAIM_LEASE_SECONDS,
  }) as unknown as { data: boolean | null; error: { message: string } | null }
  if (error) throw new Error(`INTAKE_HEARTBEAT_ERROR: ${error.message}`)
  return data === true
}

/** Complete only while this invocation owns the current, unexpired lease. */
export async function markIntakeAccepted(
  db: Db, providerMessageId: string, claimToken: string, errorCode?: string,
): Promise<void> {
  const { data, error } = await db.rpc('complete_whatsapp_webhook_event', {
    p_provider_message_id: providerMessageId,
    p_claim_token: claimToken,
    p_error_code: errorCode ? errorCode.slice(0, 120) : null,
  }) as unknown as { data: boolean | null; error: { message: string } | null }
  if (error) throw new Error(`INTAKE_ACCEPT_ERROR: ${error.message}`)
  if (data !== true) throw new Error('INTAKE_CLAIM_LOST: event claim expired or belongs to another invocation')
}

/** Fail only while this invocation owns the current, unexpired lease. */
export async function markIntakeFailed(
  db: Db, providerMessageId: string, claimToken: string, code: string,
): Promise<void> {
  const { data, error } = await db.rpc('fail_whatsapp_webhook_event', {
    p_provider_message_id: providerMessageId,
    p_claim_token: claimToken,
    p_error_code: code.slice(0, 120),
  }) as unknown as { data: boolean | null; error: { message: string } | null }
  if (error) throw new Error(`INTAKE_FAIL_ERROR: ${error.message}`)
  if (data !== true) throw new Error('INTAKE_CLAIM_LOST: event claim expired or belongs to another invocation')
}

/**
 * Media lifecycle: invoice (received) + processing job (queued) in one
 * recoverable unit. STRICTLY IDEMPOTENT (Phase 3): a webhook redelivery that
 * arrives between the insert and the intake accept must not create a second
 * invoice row or a second job. Uniqueness is enforced in the database
 * (invoices.file_object_path, one job per invoice_id+job_type); on a 23505 we
 * read back the existing rows and reuse them.
 *
 * Never upsert here: overwriting an invoice that already progressed would
 * rewind its status, which the queue interprets as work to redo.
 */
export async function recordMediaIntake(
  db: Db,
  args: {
    workspaceId: string
    clientId: string | null
    objectPath: string
    originalFileName: string | null
    contentType: string
    fileSha256Hex?: string | null
    keyVersion?: string | null
    sizeBytes?: number | null
  },
): Promise<{ invoiceId: string; jobId: string; invoicePreexisting: boolean; jobPreexisting: boolean }> {
  const invoicePayload = {
    workspace_id: args.workspaceId,
    client_id: args.clientId,
    file_object_path: args.objectPath,
    original_file_name: args.originalFileName,
    content_type: args.contentType,
    file_sha256: args.fileSha256Hex ? `\\x${args.fileSha256Hex}` : null,
    file_encryption_key_version: args.keyVersion ?? null,
    size_bytes: args.sizeBytes ?? null,
    status: 'received',
  }
  const invoice = (await db.from('invoices').insert(invoicePayload).select('id').single() as unknown as {
    data: { id: string } | null
    error: { code?: string; message: string } | null
  })
  let invoiceId: string
  let invoicePreexisting = false
  if (!invoice.error && invoice.data) {
    invoiceId = invoice.data.id
  } else if (invoice.error?.code === '23505') {
    const existing = (await db.from('invoices').select('id')
      .eq('file_object_path', args.objectPath).limit(1).maybeSingle() as unknown as {
      data: { id: string } | null; error: { message: string } | null
    })
    if (existing.error || !existing.data) {
      throw new Error(`INVOICE_READ_ERROR: ${existing.error?.message ?? 'missing row after conflict'}`)
    }
    invoiceId = existing.data.id
    invoicePreexisting = true
  } else {
    throw new Error(`INVOICE_RECORD_ERROR: ${invoice.error?.message ?? 'missing id'}`)
  }

  const jobPayload = {
    workspace_id: args.workspaceId,
    invoice_id: invoiceId,
    job_type: 'invoice_extraction',
    status: 'queued',
  }
  const job = (await db.from('processing_jobs').insert(jobPayload).select('id').single() as unknown as {
    data: { id: string } | null
    error: { code?: string; message: string } | null
  })
  let jobId: string
  let jobPreexisting = false
  if (!job.error && job.data) {
    jobId = job.data.id
  } else if (job.error?.code === '23505') {
    const existing = (await db.from('processing_jobs').select('id')
      .eq('invoice_id', invoiceId).eq('job_type', 'invoice_extraction').limit(1).maybeSingle() as unknown as {
      data: { id: string } | null; error: { message: string } | null
    })
    if (existing.error || !existing.data) {
      throw new Error(`JOB_READ_ERROR: ${existing.error?.message ?? 'missing row after conflict'}`)
    }
    jobId = existing.data.id
    jobPreexisting = true
  } else {
    throw new Error(`JOB_RECORD_ERROR: ${job.error?.message ?? 'missing id'}`)
  }
  // Best-effort, NON-BLOCKING wake so a freshly-filed invoice is picked up in
  // seconds instead of waiting for the next cron tick (§P2-9). Only fired when a
  // NEW job was created (a webhook redelivery dedups and stays quiet). It never
  // throws and the cron remains the durable backstop.
  if (!jobPreexisting) wakeWorker()
  return { invoiceId, jobId, invoicePreexisting, jobPreexisting }
}

/**
 * Layer 1 of duplicate prevention: is the EXACT SAME FILE (identical SHA-256)
 * already filed as a LIVE invoice for this (workspace, client)? A byte-for-byte
 * resend under a NEW provider media id slips past webhook idempotency (which is
 * keyed on object path) and would otherwise become a second register row, so we
 * check the stored `invoices.file_sha256` before uploading/queueing and skip it.
 *
 * Deliberately FAIL-OPEN: a lookup error (or the column being unreadable) must
 * NEVER block a legitimate invoice — we return null and the normal intake path
 * proceeds, deduping downstream as before. Only a genuinely matched live row
 * (status <> 'deleted') short-circuits. Tenant-scoped: an organization matches
 * within its own client; an individual (null client_id) matches null-client rows.
 *
 * Returns the matching invoice's id, status and batch association, or null
 * when there is no live duplicate. This lets callers distinguish an already
 * exported invoice from one that is pending, failed, or still needs review.
 */
export interface LiveFileDuplicate {
  id: string
  status: 'received' | 'queued' | 'processing' | 'needs_review' | 'completed' | 'failed' | 'deleted'
  batchId: string | null
}

/**
 * Layer 1 of duplicate prevention: find the exact same live file in this
 * workspace/client. Return its processing state as well as its id so the
 * WhatsApp acknowledgement never claims an unexported invoice was "not lost".
 * Lookup errors remain fail-open: a real invoice must never be blocked.
 */
export async function findLiveInvoiceByFileHash(
  db: Db,
  args: { workspaceId: string; clientId: string | null; fileSha256Hex: string },
): Promise<LiveFileDuplicate | null> {
  if (!args.fileSha256Hex) return null
  try {
    let query = db
      .from('invoices')
      .select('id,status,batch_id')
      .eq('workspace_id', args.workspaceId)
      .eq('file_sha256', `\\x${args.fileSha256Hex}`)
      .neq('status', 'deleted')
      .limit(1)
    query = args.clientId
      ? (query.eq('client_id', args.clientId) as typeof query)
      : (query.is('client_id', null) as typeof query)
    const { data, error } = await (query as unknown as {
      maybeSingle: () => Promise<{
        data: { id: string; status: LiveFileDuplicate['status']; batch_id: string | null } | null
        error: { message: string } | null
      }>
    }).maybeSingle()
    if (error) {
      logger.warn('[Intake] exact-file duplicate lookup failed; proceeding (fail-open)', { code: error.message.slice(0, 80) })
      return null
    }
    return data ? { id: data.id, status: data.status, batchId: data.batch_id } : null
  } catch (e) {
    logger.warn('[Intake] exact-file duplicate lookup threw; proceeding (fail-open)', { error: e instanceof Error ? e.message : String(e) })
    return null
  }
}

// ---- Pending uploads ("receive first, ask later") -------------------------
// Media that arrives before the sender has a workspace — or before an
// organization client is explicitly confirmed — is stored privately and held
// here. Assignment binds workspace/client at confirmation time and creates the
// invoice + job rows. Nothing is ever silently reassigned (client safety rule).

/**
 * Hold a securely stored upload until workspace/client assignment resolves.
 * Upserts on object_path (deterministic per provider media id): a webhook
 * redelivery refreshes the hold instead of erroring, and NEVER resets the
 * status of an already-assigned file (status is not in the payload).
 */
export async function recordPendingUpload(
  db: Db,
  args: {
    senderHmac: string
    objectPath: string
    originalFileName: string | null
    contentType: string
    fileSha256Hex: string | null
    sizeBytes: number
    providerMediaId: string | null
    providerMessageId: string
    keyVersion?: string | null
  },
): Promise<string> {
  const row = (await db.from('pending_uploads').upsert({
    sender_hmac: args.senderHmac,
    object_path: args.objectPath,
    original_file_name: args.originalFileName,
    content_type: args.contentType,
    file_sha256: args.fileSha256Hex ? `\\x${args.fileSha256Hex}` : null,
    size_bytes: args.sizeBytes,
    file_encryption_key_version: args.keyVersion ?? null,
    provider_media_id: args.providerMediaId,
    provider_message_id: args.providerMessageId,
  }, { onConflict: 'object_path' }).select('id').single() as unknown as { data: { id: string } | null; error: { message: string } | null })
  if (row.error || !row.data) throw new Error(`PENDING_UPLOAD_RECORD_ERROR: ${row.error?.message ?? 'missing id'}`)
  return row.data.id
}

/** Number of held files still waiting for assignment (drives confirm prompts). */
export async function countWaitingUploads(db: Db, senderHmac: string): Promise<number> {
  const { count, error } = await db
    .from('pending_uploads')
    .select('id', { count: 'exact', head: true })
    .eq('sender_hmac', senderHmac)
    .eq('status', 'waiting_assignment')
  if (error) throw new Error(`PENDING_UPLOAD_COUNT_ERROR: ${error.message}`)
  return count ?? 0
}

/**
 * Assign every held file for this sender to (workspace, client) — called only
 * after explicit resolution (individual selection or confirmed client).
 * Each row: invoice + job created first, then the hold is marked assigned.
 */
export async function assignPendingUploads(
  db: Db,
  senderHmac: string,
  args: { workspaceId: string; clientId: string | null; limit: number },
): Promise<number> {
  const { data: rows, error } = await db
    .from('pending_uploads')
    .select('id,object_path,original_file_name,content_type,file_sha256,size_bytes,file_encryption_key_version,provider_message_id')
    .eq('sender_hmac', senderHmac)
    .eq('status', 'waiting_assignment')
    .order('created_at')
  if (error) throw new Error(`PENDING_UPLOAD_LIST_ERROR: ${error.message}`)
  let assigned = 0
  for (const row of (rows ?? []) as Array<{
    id: string; object_path: string; original_file_name: string | null;
    content_type: string; file_sha256: string | null;
    size_bytes: number | null; file_encryption_key_version: string | null;
    provider_message_id: string | null
  }>) {
    const providerMessageId = row.provider_message_id ?? `legacy-pending:${row.id}`
    const reservation = await reserveDailyIntake(db, {
      providerMessageId, workspaceId: args.workspaceId, senderHmac, limit: args.limit,
    })
    if (!reservation.allowed) {
      logger.warn('[Intake] held upload assignment deferred by daily quota', {
        pendingId: row.id.slice(0, 8), workspaceId: args.workspaceId.slice(0, 8),
      })
      continue
    }
    const shaHex = typeof row.file_sha256 === 'string'
      ? row.file_sha256.replace(/^\\x/i, '')
      : null
    const recorded = await recordMediaIntake(db, {
      workspaceId: args.workspaceId,
      clientId: args.clientId,
      objectPath: row.object_path,
      originalFileName: row.original_file_name,
      contentType: row.content_type,
      fileSha256Hex: shaHex,
      keyVersion: row.file_encryption_key_version,
      sizeBytes: row.size_bytes,
    })
    await commitDailyIntake(db, providerMessageId, reservation.scopeKey)
    await markStorageIntake(db, row.object_path, 'linked')
    if (reservation.scopeKey !== `sender:${senderHmac}`) {
      await releaseDailyIntakeScope(db, providerMessageId, `sender:${senderHmac}`)
    }
    const { error: updateError } = await db
      .from('pending_uploads')
      .update({ status: 'assigned', assigned_invoice_id: recorded.invoiceId, updated_at: new Date().toISOString() })
      .eq('id', row.id)
      .eq('status', 'waiting_assignment')
    if (updateError) throw new Error(`PENDING_UPLOAD_ASSIGN_ERROR: ${updateError.message}`)
    assigned += 1
  }
  return assigned
}

// All queue transitions happen inside SECURITY DEFINER functions (see

const positiveInt = (raw: string | undefined, fallback: number): number => {

  return Number.isFinite(parsed) && parsed > 0 ? parsed : fallback
}
const boundedConcurrency = (raw: string | undefined, fallback: number): number =>
  Math.min(25, positiveInt(raw, fallback))

export const JOB_LEASE_SECONDS = positiveInt(process.env.JOB_LEASE_SECONDS, 120)
const BACKOFF_BASE_SECONDS = positiveInt(process.env.JOB_BACKOFF_BASE_SECONDS, 15)
const BACKOFF_CAP_SECONDS = positiveInt(process.env.JOB_BACKOFF_CAP_SECONDS, 900)
// These limits are enforced again inside claim_processing_jobs, so overlapping
// worker ticks/instances cannot multiply the configured concurrency.
export const JOB_GLOBAL_CONCURRENCY = boundedConcurrency(process.env.JOB_GLOBAL_CONCURRENCY, 5)
export const JOB_WORKSPACE_CONCURRENCY = boundedConcurrency(process.env.JOB_WORKSPACE_CONCURRENCY, 2)

/** Exponential backoff on the attempt count: 15s, 30s, 60s … capped. */
export function backoffSeconds(attempts: number): number {
  const exp = Math.max(1, attempts) - 1
  return Math.min(BACKOFF_CAP_SECONDS, BACKOFF_BASE_SECONDS * 2 ** exp)
}

/**
 * Claim up to `limit` due jobs for this worker. FOR UPDATE SKIP LOCKED means
 * concurrent claimers get disjoint sets — never the same job twice. Only pass
 * job types a handler is actually registered for, otherwise nothing is claimed.
 */
export async function claimJobs(
  db: Db,
  args: { worker: string; jobTypes: JobType[]; limit?: number; leaseSeconds?: number },
): Promise<ClaimedJob[]> {
  if (args.jobTypes.length === 0) return []
  const { data, error } = await db.rpc('claim_processing_jobs', {
    p_worker: args.worker,
    p_lease_seconds: args.leaseSeconds ?? JOB_LEASE_SECONDS,
    p_limit: args.limit ?? 1,
    p_job_types: args.jobTypes,
    p_global_limit: JOB_GLOBAL_CONCURRENCY,
    p_workspace_limit: JOB_WORKSPACE_CONCURRENCY,
  })
  if (error) throw new Error(`JOB_CLAIM_ERROR: ${error.message}`)
  return ((data ?? []) as unknown as ClaimedJob[]).map((row) => ({
    ...row,
    // bytea/timestamptz come back as strings; keep the raw lease for logging.
    lease_expires_at: String(row.lease_expires_at ?? ''),
  }))
}

/**

 * decides which, because it alone knows the attempt count. `invoiceStatus` is
 * applied only when the job settles, in the same transaction — a job and its
 * invoice can never be observed as a torn pair.
 */
export async function failJob(
  db: Db,
  args: {
    jobId: string
    worker: string
    errorCode: string
    retryInSeconds?: number
    permanent?: boolean
    invoiceStatus?: Extract<InvoiceStatus, 'failed' | 'needs_review'>

    p_permanent: args.permanent ?? false,
    p_invoice_status: args.invoiceStatus ?? null,
  })
  if (error) throw new Error(`JOB_FAIL_ERROR: ${error.message}`)

export class JobLeaseLostError extends Error {
  override name = 'JobLeaseLostError'
}

interface LeaseHeartbeat {
  assertOwned(): Promise<void>
  stop(): void
}

/** Renew often and fail closed if ownership cannot be confirmed. */
function startHeartbeat(db: Db, job: ClaimedJob, worker: string, leaseSeconds: number): LeaseHeartbeat {
  const everyMs = Math.max(1_000, Math.floor((leaseSeconds * 1000) / 4))
  let timer: ReturnType<typeof setTimeout> | undefined
  let inFlight: Promise<void> | null = null
  let lost: JobLeaseLostError | null = null
  let stopped = false

  const renew = (): Promise<void> => {
    if (lost) return Promise.reject(lost)
    if (inFlight) return inFlight
    inFlight = (async () => {
      let lastError: unknown
      for (let attempt = 1; attempt <= 2; attempt += 1) {
        try {
          const owned = await renewLease(db, job.id, worker, leaseSeconds)
          if (!owned) throw new JobLeaseLostError('JOB_LEASE_RENEWAL_REFUSED')
          if (attempt > 1) {
            logger.info('[Worker] lease renewal recovered after transient error', {
              jobId: job.id, worker, attempt,
            })
          }
          return
        } catch (error) {
          if (error instanceof JobLeaseLostError) throw error
          lastError = error
          if (attempt === 1) {
            logger.warn('[Worker] lease renewal transient error; retrying', {
              jobId: job.id, worker, attempt,
              errorType: error instanceof Error ? error.name : 'unknown',
            })
            await new Promise((resolve) => setTimeout(resolve, 300))
          }
        }
      }
      throw lastError
    })().catch((error: unknown) => {
      // A renewal that was already in flight when the handler finished races the
      // job's own settlement (the row has just left 'processing'), so it is
      // refused. That is expected once stop() was called: not an ownership loss.
      if (stopped) throw error
      lost = error instanceof JobLeaseLostError
        ? error
        : new JobLeaseLostError('JOB_LEASE_RENEWAL_UNCONFIRMED')
      logger.error('[Worker] lease ownership lost or unconfirmed', {
        jobId: job.id, worker, reason: lost.message,
      })
      throw lost
    }).finally(() => {
      inFlight = null
    })
    return inFlight
  }

  const schedule = (): void => {
    if (stopped || lost) return
    timer = setTimeout(() => {
      void renew().then(schedule, () => undefined)
    }, everyMs)
  }
  schedule()

  return {
    assertOwned: renew,
    stop() {
      stopped = true
      if (timer) clearTimeout(timer)
    },
  }
}

/**
 * Process ONE already-claimed job to its fenced outcome, mutating the shared
 * tick `result` counters. Extracted so a batch can run several concurrently
 * (§P2-10). It NEVER rejects: every path (missing handler, lease loss, handler
 * throw, fenced completion) is recorded and swallowed, so Promise.all over a batch
 * can't let one bad job take the tick — or its siblings — down. Lease ownership is
 * still enforced per job: a lost renewal fences ONLY that job, never the batch.
 */
async function processClaimedJob(
  db: Db, job: ClaimedJob, worker: string, leaseSeconds: number, result: TickResult,
): Promise<void> {
  const handler = handlers.get(job.job_type)
  if (!handler) {
    // Defensive: registeredJobTypes() should prevent this. Requeue without
    // settling — a later deploy may register the handler.
    await failJob(db, {
      jobId: job.id, worker, errorCode: 'HANDLER_NOT_REGISTERED',
      retryInSeconds: backoffSeconds(job.attempts),
    }).catch(() => undefined)
    result.requeued += 1
    return
  }

  const heartbeat = startHeartbeat(db, job, worker, leaseSeconds)
  // §P2-12: time the handler itself so a slow job is attributable without any PII.
  const jobTiming = new StageTimer()
  try {
    await handler({ db, job, worker, leaseSeconds, assertLease: heartbeat.assertOwned })
    await heartbeat.assertOwned()
    heartbeat.stop() // stop renewing BEFORE the job settles
    const owned = await completeJob(db, job.id, worker)
    jobTiming.mark('handler')
    logger.info('[Worker] job timing', {
      jobId: job.id, type: job.job_type, worker,
      outcome: owned ? 'completed' : 'fenced', ...jobTiming.summary(),
    })
    if (owned) {
      result.completed += 1
      // §completion-push: THIS job is now settled, so the workspace can be judged
      // honestly quiet — a register hand could never pass that gate while it was
      // still the queue's own live row. Hand the finished file to the delivery
      // consumer the moment its last build lands, instead of leaving it owed until
      // the owner happens to message again. The consumer applies every rule
      // (quiescence, newest-version-per-scope, the owner's proactive opt-in and
      // Meta's 24h window), no-ops on one count query when any fails, and never
      // throws — a push problem cannot un-settle a completed job or fail the tick.
      await pushReadyRegistersForWorkspace(db, job.workspace_id)
    } else {
      result.fenced += 1
      logger.warn('[Worker] completion fenced after handler', { jobId: job.id, worker })
    }
  } catch (e: unknown) {
    heartbeat.stop() // never renew a lease while (or after) failing the job
    if (e instanceof JobLeaseLostError) {
      result.fenced += 1
      logger.warn('[Worker] handler stopped after lease loss', { jobId: job.id, worker })
      return
    }
    const detail = e instanceof Error ? `${e.name}: ${e.message}` : String(e)
    const code = detail.split(':')[0] || 'JOB_HANDLER_ERROR'
    const permanent = isPermanentJobFailure(e)
    const outcome = await failJob(db, {
      jobId: job.id,
      worker,
      errorCode: code,
      retryInSeconds: permanent ? 0 : backoffSeconds(job.attempts),
      permanent,
      // Applied by the RPC only when the job actually settles for good.
      invoiceStatus: 'failed',
    })
    if (outcome === 'requeued') result.requeued += 1
    else if (outcome === 'failed') result.failed += 1
    else result.fenced += 1
    logger.warn('[Worker] job failed', {
      jobId: job.id, type: job.job_type, attempt: job.attempts,
      outcome, error: code, detail: detail.slice(0, 200),
      durationMs: jobTiming.total(),
    })
  } finally {
    heartbeat.stop()
  }
}

/**
 * One queue pass: sweep stale leases, claim due work for registered types,
 * run each handler independently. Never throws — a broken handler cannot take
 * the tick (or the other jobs) down with it.
 */
export async function runWorkerTick(
  db: Db,
  args?: { worker?: string; limit?: number; leaseSeconds?: number },
): Promise<TickResult> {
  const worker = args?.worker ?? `w_${randomUUID()}`
  const leaseSeconds = args?.leaseSeconds ?? JOB_LEASE_SECONDS
  const result: TickResult = {
    worker, recovered: 0, claimed: 0, completed: 0, requeued: 0, failed: 0, fenced: 0, idle: true,
  }

  // Crash recovery first: work abandoned by a dead worker becomes claimable.
  try {
    const recovered = await recoverStaleJobs(db)
    result.recovered = recovered.requeued + recovered.failed
    if (result.recovered > 0) {
      logger.info('[Worker] stale leases recovered', {
        requeued: recovered.requeued, failed: recovered.failed,
      })
    }
  } catch (e: unknown) {
    logger.error('[Worker] recovery pass failed', {
      error: e instanceof Error ? e.message : String(e),
    })
  }

  // §delivery recovery: a 'sending' delivery whose lease lapsed (the worker died
  // mid-send, so it is uncertain whether Meta accepted it) moves to 'unknown' —
  // NEVER blindly resent. Opportunistic per tick; returns 0 (no-op) until the
  // outbox migration is applied, so this is inert on an un-migrated deployment.
  try {
    const swept = await recoverExpiredDeliveryLeases(db)
    if (swept > 0) logger.info('[Worker] expired delivery leases recovered to unknown', { swept })
  } catch (e: unknown) {
    logger.error('[Worker] delivery lease recovery failed', {
      error: e instanceof Error ? e.message : String(e),
    })
  }

  try {
    const replay = await replayPendingWebhookMessages(db, 1)
    if (replay.attempted > 0) {
      logger.info('[Worker] durable webhook inbox replay pass', replay)
    }
  } catch (e: unknown) {
    logger.error('[Worker] durable webhook inbox replay failed', {
      error: e instanceof Error ? e.message.split(':')[0] : 'UNKNOWN',
    })
  }

  try {
    const recovered = await recoverStaleIntakeReservations(db)
    if (recovered > 0) logger.info('[Worker] stale intake quota reservations recovered', { recovered })
  } catch (e: unknown) {
    logger.error('[Worker] intake quota reservation recovery failed', {
      error: e instanceof Error ? e.message.split(':')[0] : 'UNKNOWN',
    })
  }

  try {
    const reconciliation = await reconcileOrphanedIntakeObjects(db, 10)
    if (reconciliation.claimed > 0) {
      logger.info('[Worker] intake storage reconciliation pass', reconciliation)
    }
  } catch (e: unknown) {
    logger.error('[Worker] intake storage reconciliation failed', {
      error: e instanceof Error ? e.message.split(':')[0] : 'UNKNOWN',
    })
  }

  try {
    const [expiredPayloads, oldReservations] = await Promise.all([
      purgeExpiredWebhookPayloads(db),
      purgeOldIntakeQuotaHistory(db),
    ])
    if (expiredPayloads > 0 || oldReservations > 0) {
      logger.info('[Worker] transient intake data expired', { expiredPayloads, oldReservations })
    }
  } catch (e: unknown) {
    logger.error('[Worker] intake retention sweep failed', {
      error: e instanceof Error ? e.message.split(':')[0] : 'UNKNOWN',
    })
  }

  try {
    const retention = await processRetentionMaintenance(db, 10)
    if (retention.notices > 0 || retention.deletions > 0 || retention.failed > 0) {
      logger.info('[Worker] review retention maintenance', {
        notices: retention.notices,
        deletions: retention.deletions,
        failed: retention.failed,
      })
    }
  } catch (e: unknown) {
    logger.error('[Worker] review retention maintenance failed', {
      error: e instanceof Error ? e.message.split(':')[0] : 'UNKNOWN',
    })
  }

  const jobTypes = registeredJobTypes()
  if (jobTypes.length === 0) {
    logger.info('[Worker] no job handlers registered; queue untouched')
    return result
  }

  const deadline = Date.now() + budgetMs()
  const maxJobs = Math.min(Math.max(1, args?.limit ?? tickLimit()), 25)
  const concurrency = jobConcurrency()
  while (result.claimed < maxJobs) {
    if (Date.now() > deadline || (result.claimed > 0 && Date.now() > deadline - minJobWindowMs())) {
      logger.info('[Worker] tick budget reached before next claim', { worker, claimed: result.claimed })
      break
    }
    // Claim a whole batch (bounded by the tick's remaining budget AND the
    // concurrency cap) and run it CONCURRENTLY (§P2-10). claimJobs uses FOR UPDATE
    // SKIP LOCKED, so a concurrent sibling worker never grabs the same rows, and
    // each job still settles independently through the fenced RPCs — one slow or
    // failing handler cannot block or corrupt the others in the same batch.
    const want = Math.min(concurrency, maxJobs - result.claimed)
    const batch = await claimJobs(db, { worker, jobTypes, limit: want, leaseSeconds })
    if (batch.length === 0) break
    result.claimed += batch.length
    result.idle = false
    await Promise.all(batch.map((job) => processClaimedJob(db, job, worker, leaseSeconds, result)))
  }

  if (result.claimed > 0) {
    logger.info('[Worker] tick finished', {
      worker, claimed: result.claimed, completed: result.completed,
      requeued: result.requeued, failed: result.failed, fenced: result.fenced,
      recovered: result.recovered,
    })
  }

  // §delivery sweep: a finished register must not wait for the owner to message
  // again. After the queue work of this tick, hand over anything owed to an owner
  // who opted into proactive delivery (and only to them — the consumer re-checks
  // consent, the 24h window and that the workspace is now quiet). Best-effort: a
  // failure leaves every row pending/failed for the next tick or inbound.
  try {
    const sent = await sweepDueDeliveries(db)
    if (sent > 0) logger.info('[Worker] due registers delivered without an inbound', { sent })
  } catch (e: unknown) {
    logger.error('[Worker] delivery sweep failed', {
      error: e instanceof Error ? e.message : String(e),
    })
  }

  return result
}

// lib/exports/access.ts — pure authorization decision for workbook downloads.
//
// The register is encrypted at rest and can only be served through our own
// authenticated route. This module holds the SINGLE decision function so the
// rule set is small, auditable and unit-testable in-process (it imports no DB,
// no crypto and no `@/` aliases). The route gathers the facts and defers the
// verdict entirely to decideDownloadAccess — so a cross-workspace request, an
// expired link, or an anonymous caller can never reach the decrypt step.
export type DownloadDenial = {
  ok: false
  status: 401 | 403 | 404 | 410
  code: string
}
export type DownloadDecision = { ok: true } | DownloadDenial

export interface DownloadFacts {
  /** A verified signed-in Supabase user made the request. */
  authenticated: boolean
  /** The requested workbook row exists. */
  workbookExists: boolean
  /** The verified user is a member of the workbook's workspace. */
  memberOfWorkspace: boolean
  /** generated_workbooks.status, or null when the row was not found. */
  status: string | null
  /** generated_workbooks.expires_at (ISO), or null when unset. */
  expiresAt: string | null
  /** Current wall-clock in ms (injected so the check is deterministic). */
  nowMs: number
}

/**
 * Decide whether a download may proceed, in strict precedence order:
 *   401 anonymous  ->  404 no such workbook  ->  403 not a member (tenant
 *   isolation)  ->  404 deleted  ->  410 expired. Only an all-clear returns ok.
 * A non-member is answered 403 to make cross-workspace denial explicit; the
 * route logs no PII and returns only these stable codes.
 */
export function decideDownloadAccess(f: DownloadFacts): DownloadDecision {
  if (!f.authenticated) return { ok: false, status: 401, code: 'UNAUTHENTICATED' }
  if (!f.workbookExists) return { ok: false, status: 404, code: 'WORKBOOK_NOT_FOUND' }
  if (!f.memberOfWorkspace) return { ok: false, status: 403, code: 'NOT_MEMBER' }
  if (f.status === 'deleted') return { ok: false, status: 404, code: 'WORKBOOK_DELETED' }
  if (f.status === 'expired') return { ok: false, status: 410, code: 'WORKBOOK_EXPIRED' }
  if (f.expiresAt) {
    const t = Date.parse(f.expiresAt)
    if (Number.isFinite(t) && t <= f.nowMs) return { ok: false, status: 410, code: 'LINK_EXPIRED' }
  }
  return { ok: true }
}

// lib/exports/deliver.ts — hand a generated register back to its OWNER in-chat.
//
// The product promises "your Excel register will be sent here." The workbook is
// encrypted at rest, and the web download route requires a signed-in Supabase
// session a WhatsApp-only user does not have, so a link alone is a dead end. The
// only reliable delivery for a WhatsApp-first user is to decrypt the container IN
// MEMORY and send the .xlsx back into the owner's own chat as a document.
//
// This mirrors the download route's serve path exactly (same storage bucket, same
// decrypt + key-version check) but is authorized by the workspace owner's sender
// identity at webhook time rather than a browser cookie. It NEVER logs the file
// bytes, the object path, the key version, or the media id.
//
// Idempotency: a workbook is only pushed while its row is status='ready'. Once
// delivered we flip it to 'sent'; a later 'sent' row is reported as already
// delivered without a re-send, so repeated Review/Done taps never spam the file.
import type { SupabaseClient } from '@supabase/supabase-js'
import { bucketFor } from '@/lib/supabase/server'
import { decryptFile, resolveFileKeyConfig } from '@/lib/encryption/files'
import { uploadWhatsAppMedia, isPermanentMediaFailure } from '@/lib/whatsapp/media'
import { sendDocument, type SendOutcome } from '@/lib/whatsapp/send'
import { workbookFileName, categoryFileToken } from '@/lib/exports/naming'
import { registerSheetNameForCategory } from '@/lib/excel/generator'
import { logger } from '@/lib/logger'

type Db = SupabaseClient

const SPREADSHEET_MIME = 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'
const UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i

interface WorkbookRow {
  workspace_id: string
  batch_id: string | null
  status: string
  expires_at: string | null
  file_object_path: string
  file_encryption_key_version: string | null
  accounting_category: string | null
  period_start?: string | null
  version?: number | null
}

export interface DeliverResult {
  /** true when the file is (now or previously) in the owner's chat. */
  delivered: boolean
  /** 'skipped' = not eligible (missing/expired/not-ready) — not a failure. */
  outcome: SendOutcome | 'skipped'
  /** Meta's message id when THIS call confirmed a send, for outbox audit. */
  providerMessageId?: string | null
  /** Bounded, PII-free reason when it did not land — persisted as last_error. */
  providerDetail?: string | null
}

/**
 * Deliver one workbook (by id) to `to` as a WhatsApp document. The caller has
 * already tied `to` to the workspace that OWNS this workbook (the inbound sender
 * who onboarded it), so this is owner-scoped, not cross-tenant: the row's
 * workspace is re-read here and only that workspace's stored object is fetched.
 */
export async function deliverWorkbookToWhatsApp(
  db: Db,
  args: { workbookId: string; to: string; caption?: string; expectedWorkspaceId?: string },
): Promise<DeliverResult> {
  const { workbookId, to } = args
  if (!UUID_RE.test(workbookId)) return { delivered: false, outcome: 'skipped' }

  const { data: row, error: readErr } = await db
    .from('generated_workbooks')
    .select('workspace_id,batch_id,status,expires_at,file_object_path,file_encryption_key_version,accounting_category,period_start,version')
    .eq('id', workbookId)
    .maybeSingle()
  if (readErr || !row) {
    logger.warn('[Deliver] workbook row unavailable for delivery', { code: 'DELIVER_NO_ROW' })
    return { delivered: false, outcome: 'skipped' }
  }
  const wb = row as WorkbookRow
  // §delivery isolation: when the caller knows which workspace the recipient is
  // owed (the delivery outbox records it per row), refuse to serve a workbook that
  // belongs to a DIFFERENT tenant — a claim must never resolve to another
  // workspace's file even given a forged/guessed workbook id.
  if (args.expectedWorkspaceId && wb.workspace_id !== args.expectedWorkspaceId) {
    logger.warn('[Deliver] workspace mismatch; refusing cross-tenant send', { code: 'DELIVER_TENANT_MISMATCH', workbookId })
    return { delivered: false, outcome: 'skipped' }
  }
  const expired = !!wb.expires_at && new Date(wb.expires_at).getTime() <= Date.now()
  if (expired || !wb.file_object_path) return { delivered: false, outcome: 'skipped' }
  // Already pushed to this owner once — do not re-send; report as delivered.
  if (wb.status === 'sent') return { delivered: true, outcome: 'ok' }
  if (wb.status !== 'ready') return { delivered: false, outcome: 'skipped' }
  // §one-file-per-month: a workbook is only servable while it is still the LIVE
  // register for its own (batch, category). Publish retires prior versions, but a
  // rebuild that raced the retire (or two builds that read the same max version)
  // can leave two 'ready' rows for one month — and the owner would then receive the
  // same month twice, an older copy first. If a NEWER version exists for this exact
  // scope, this row is superseded: skip it, never send a stale register.
  if (wb.batch_id && wb.version != null) {
    const scope = db.from('generated_workbooks').select('id')
      .eq('workspace_id', wb.workspace_id).eq('batch_id', wb.batch_id)
      .gt('version', wb.version).in('status', ['ready', 'sent', 'expired'])
    // A legacy row may carry no category at all: match on IS NULL rather than
    // eq(null), which PostgREST renders as a filter that can never match.
    const { data: newer } = await (wb.accounting_category
      ? scope.eq('accounting_category', wb.accounting_category)
      : scope.is('accounting_category', null)).limit(1)
    if ((newer ?? []).length > 0) {
      logger.info('[Deliver] workbook superseded by a newer version; not sending a stale register', { workbookId })
      return { delivered: false, outcome: 'skipped', providerDetail: 'superseded by a newer version' }
    }
  }

  try {
    const { masterKeyHex } = resolveFileKeyConfig()
    const { data: blob, error: dlErr } = await db.storage
      .from(bucketFor('exports'))
      .download(wb.file_object_path)
    if (dlErr || !blob) {
      // A MISSING object is permanent for this row — a retry can never resurrect it
      // — so it must NOT be reported 'transient' (which parks it 'failed' with a
      // future next_attempt_at and re-loops forever on every inbound, burying the
      // real fault). '404'/'NoSuchKey' is surfaced as 'skipped', which the consumer
      // parks as a terminal 'unknown' for reconciliation; a network/5xx read stays
      // 'transient'. Log only a bounded provider status/code — never the path or PII.
      const status = (dlErr as { statusCode?: number | string } | null)?.statusCode
      const pcode = (dlErr as { code?: string } | null)?.code
      const missing = String(status) === '404' || pcode === 'NoSuchKey'
      logger.error('[Deliver] storage read failed', {
        code: missing ? 'DELIVER_OBJECT_MISSING' : 'DELIVER_STORAGE_ERROR',
        status: status ?? null, providerCode: pcode ?? null, workbookId,
      })
      return {
        delivered: false, outcome: missing ? 'skipped' : 'transient',
        providerDetail: `storage: ${missing ? 'object missing' : `status ${String(status ?? pcode ?? 'unknown')}`}`,
      }
    }
    const container = Buffer.from(await blob.arrayBuffer())
    // A tampered / wrong-key container throws (never emits plaintext); we surface
    // a safe code only.
    const plain = decryptFile(container, masterKeyHex, {
      context: wb.file_object_path,
      expectedKeyVersion: wb.file_encryption_key_version ?? undefined,
    })

    // The in-chat file is named by its REAL accounting month (Billora_YYYY-MM.xlsx,
    // §P1-5), read from the workbook's OWN validated period (§register-scope) — the
    // scope it was actually built for — and only if that is absent (a legacy row from
    // before the period was stamped) does it fall back to the batch's period_start.
    // An unresolvable period keeps the stable generic name; we never stamp a guessed
    // month on the download.
    let periodStart: string | null = wb.period_start ?? null
    if (!periodStart && wb.batch_id) {
      const { data: batchRow } = await db
        .from('invoice_batches').select('period_start')
        .eq('workspace_id', wb.workspace_id).eq('id', wb.batch_id).maybeSingle()
      periodStart = (batchRow as { period_start?: string | null } | null)?.period_start ?? null
    }
    const filename = workbookFileName(periodStart, categoryFileToken(registerSheetNameForCategory(wb.accounting_category)))

    let mediaId: string
    try {
      mediaId = await uploadWhatsAppMedia({ buffer: plain, mimeType: SPREADSHEET_MIME, filename })
    } catch (e) {
      // The message is Meta's own text plus a bounded graph code — no token, no path.
      const detail = e instanceof Error ? e.message.slice(0, 120) : 'media upload error'
      if (isPermanentMediaFailure(e)) {
        logger.error('[Deliver] media upload blocked (credential)', { code: 'DELIVER_UPLOAD_BLOCKED' })
        return { delivered: false, outcome: 'blocked', providerDetail: `upload: ${detail}` }
      }
      logger.warn('[Deliver] media upload transient', { code: 'DELIVER_UPLOAD_RETRY' })
      return { delivered: false, outcome: 'transient', providerDetail: `upload: ${detail}` }
    }

    // Capture Meta's message id so the outbox records exactly which send settled
    // the claim (audit + reconciliation); it is never logged in full.
    const messageIdOut: { current: string | null } = { current: null }
    // And WHY it did not, when it didn't: a blocked document send and a blocked text
    // reply are not the same fault, and without the provider code parked in the row
    // an operator cannot tell a dead token from a temporary Meta media block.
    const diagnosticsOut: { current: string | null } = { current: null }
    const outcome = await sendDocument(
      to,
      {
        mediaId,
        filename,
        caption: args.caption ?? 'Your Billora invoice register. Each invoice is filed under its own invoice month.',
      },
      { workbookId },
      messageIdOut,
      diagnosticsOut,
    )
    if (outcome !== 'ok') {
      return { delivered: false, outcome, providerMessageId: messageIdOut.current, providerDetail: diagnosticsOut.current }
    }

    // Confirmed delivered: retire the pending row so we never re-send it. A lost
    // race (already flipped) is fine — the file is in the chat either way.
    await db.from('generated_workbooks').update({ status: 'sent' }).eq('id', workbookId).eq('status', 'ready')
    logger.info('[Deliver] register delivered as document', { workbookId })
    return { delivered: true, outcome: 'ok', providerMessageId: messageIdOut.current }
  } catch (e) {
    logger.error('[Deliver] delivery failed', { code: 'DELIVER_FAILED' })
    return { delivered: false, outcome: 'transient', providerDetail: `fault: ${e instanceof Error ? e.name : 'unknown'}`.slice(0, 120) }
  }
}

// lib/exports/naming.ts — deterministic, per-month register filenames (§P1-5).
//
// A batch is one (client, calendar month), so every generated workbook covers a
// single accounting period. Accountants file these by month, so the download and
// the in-chat document must be named `Billora_YYYY-MM.xlsx` — derived ONLY from
// the batch's real period_start (never from an upload date or a guess). When no
// period is available we fall back to a stable generic name rather than inventing
// a month, so a mis-dated batch can never masquerade as a specific period.
//
// Pure + alias-free so the exact filename a client receives is unit-testable in
// process (the whole point of this module is a deterministic, human-facing string).

/** Fallback when a workbook carries no resolvable accounting month. */
export const GENERIC_REGISTER_FILENAME = 'Billora-register.xlsx'

const SHORT_MONTHS = [
  'Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec',
] as const

/**
 * Normalise any date-ish period_start ("2026-10-01", a full timestamptz, etc.)
 * to its `YYYY-MM` accounting token, or null when it is absent/malformed/out of
 * range. Never coerces a bad value into a plausible month.
 */
export function monthToken(periodStart: string | null | undefined): string | null {
  if (typeof periodStart !== 'string') return null
  const m = /^(\d{4})-(\d{2})/.exec(periodStart.trim())
  if (!m) return null
  const month = Number(m[2])
  if (!Number.isInteger(month) || month < 1 || month > 12) return null
  return `${m[1]}-${String(month).padStart(2, '0')}`
}

/** The human-facing filename for one month's register. §separate-files-per-category: a non-purchase category register gets its category decorated into the name; the purchase file keeps the exact historical `Billora_YYYY-MM.xlsx` name. */
export function workbookFileName(periodStart: string | null | undefined, categoryToken?: string | null): string {
  const tok = monthToken(periodStart)
  if (!tok) return GENERIC_REGISTER_FILENAME
  return categoryToken ? `Billora_${tok}_${categoryToken}.xlsx` : `Billora_${tok}.xlsx`
}

/** A short label ("Oct 2026") for the bulk summary, or null when undated. */
export function monthLabel(periodStart: string | null | undefined): string | null {
  const tok = monthToken(periodStart)
  if (!tok) return null
  const month = Number(tok.slice(5, 7))
  return `${SHORT_MONTHS[month - 1]} ${tok.slice(0, 4)}`
}

/**
 * §separate-files-per-category: a filename-safe token for an accounting category
 * derived from its register sheet label (the SAME label the generator writes), so
 * the download name, the object key and the sheet name can never disagree.
 * 'Purchase Register' -> null (the purchase file keeps the historical name),
 * 'Expense Register' -> 'expense', 'Fixed Asset Register' -> 'fixed-asset',
 * 'Capital Purchase Register' -> 'capital-purchase'.
 */

  if (registerSheetName === 'Purchase Register') return null

// lib/exports/tally-xml.ts — a TallyPrime "Purchase" voucher XML importer (§P4-17).
//
// SCOPE + HONESTY: this renders a TallyPrime-compatible XML *Import Data* envelope
// from the SAME fully-resolved `TallyImportRowInput` lines the approved Excel
// register already uses — so it can never introduce a value the workbook did not.
// It is a GENERATOR only: it is deliberately NOT wired into the accountant-facing
// delivery pipeline, because correctness here depends on the tenant's TallyPrime
// ledger masters (the supplier, Purchase, CGST/SGST/IGST and Round Off ledgers must
// exist or be created first) and on a real import test we cannot run from this
// repo. A human MUST import a generated file into a TallyPrime TEST company and
// confirm the vouchers land before this is ever trusted on live books.
//
// Why pure + alias-free: it touches no DB/crypto/ExcelJS (the input type is imported
// as an ERASED `import type`, so exceljs is never loaded at runtime), which keeps it
// unit-testable in-process and lets a future version extend it without touching OCR.
//
// Double-entry rule that keeps every voucher balanced (Tally rejects un-balanced
// vouchers, and we must NEVER fabricate to force a tie):
//   Debit Purchase(taxable) + Debit(CGST/SGST/IGST, each non-zero)
//     + a balancing "Round Off" ledger (gross_total - the sum of the above)
//     = Credit Supplier(gross_total).
// The printed gross total is authoritative; the Round Off leg is the arithmetical
// remainder (standard rounding treatment), never a guessed amount.
import type { TallyImportRowInput } from '../excel/generator'

/** Result of a render: the document plus integrity counts (never any row values). */
export interface TallyXmlResult {
  xml: string
  /** Vouchers actually emitted. */
  voucherCount: number
  /** Lines skipped for being incomplete (never emitted as a malformed voucher). */
  skippedCount: number
}

/** Escape the five XML predefined entities; a null becomes the empty string. */
function esc(value: string | null | undefined): string {
  if (value === null || value === undefined) return ''
  return String(value)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&apos;')
}

/** Fixed 2-dp amount with no thousands separator; Tally's own numeric rendering. */
function amt(n: number): string {
  // Round to the paisa first so float noise (e.g. 0.1+0.2) never leaks a long tail.
  const r = Math.round((Number.isFinite(n) ? n : 0) * 100) / 100
  return Object.is(r, -0) ? '0.00' : r.toFixed(2)
}

/** ISO `YYYY-MM-DD` -> Tally `YYYYMMDD`, or null when unparseable. */
function tallyDate(iso: string | null): string | null {
  if (typeof iso !== 'string') return null
  const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(iso.trim())
  if (!m) return null
  return `${m[1]}${m[2]}${m[3]}`
}

/** A finite number or null; blank/NaN never becomes a fabricated 0. */
function num(value: number | null): number | null {
  return typeof value === 'number' && Number.isFinite(value) ? value : null
}

/** One `<ALLLEDGERENTRIESLIST>` leg. ISDEEMEDPOSITIVE mirrors the sign of `amount`. */
function ledgerLeg(name: string, amount: number): string {
  const deemed = amount >= 0 ? 'Yes' : 'No'
  return [
    '      <ALLLEDGERENTRIESLIST>',
    `        <LEDGERNAME>${esc(name)}</LEDGERNAME>`,
    `        <ISDEEMEDPOSITIVE>${deemed}</ISDEEMEDPOSITIVE>`,
    `        <AMOUNT>${amt(amount)}</AMOUNT>`,
    '      </ALLLEDGERENTRIESLIST>',
  ].join('\n')
}

/**
 * Render one voucher, or null when the line is too incomplete to import safely.
 * A missing date, invoice number, supplier, purchase ledger, taxable value or gross
 * total is NOT silently defaulted — it is skipped and counted, so a half-read bill
 * can never become a confidently-wrong voucher in someone's books.
 */
function renderVoucher(d: TallyImportRowInput): { xml: string } | null {
  const date = tallyDate(d.invoice_date)
  const invoiceNumber = d.invoice_number && d.invoice_number.trim() !== '' ? d.invoice_number.trim() : null
  const supplier = d.vendor_name && d.vendor_name.trim() !== '' ? d.vendor_name.trim() : null
  const purchaseLedger = d.ledger_name && d.ledger_name.trim() !== '' ? d.ledger_name.trim() : null
  const taxable = num(d.taxable_value)
  const gross = num(d.gross_total)

  if (date === null || invoiceNumber === null || supplier === null || purchaseLedger === null || taxable === null || gross === null) {
    return null
  }

  const cgst = num(d.cgst_amount) ?? 0
  const sgst = num(d.sgst_amount) ?? 0
  const igst = num(d.igst_amount) ?? 0

  const legs: string[] = []
  // Debit: the purchase (taxable) ledger.
  legs.push(ledgerLeg(purchaseLedger, taxable))
  // Debit: each non-zero duty ledger. A genuine 0 is omitted (never a phantom leg).
  if (cgst !== 0) legs.push(ledgerLeg('CGST', cgst))
  if (sgst !== 0) legs.push(ledgerLeg('SGST', sgst))
  if (igst !== 0) legs.push(ledgerLeg('IGST', igst))
  // Balancing Round Off leg = the remainder so debits tie exactly to the printed gross.
  const roundOffLeg = Math.round((gross - (taxable + cgst + sgst + igst)) * 100) / 100
  if (roundOffLeg !== 0) legs.push(ledgerLeg('Round Off', roundOffLeg))
  // Credit: the supplier for the full printed total. By construction the legs sum
  // to zero (debits = gross = the single credit leg), which is the balance Tally
  // requires; the tests re-derive the sum from the emitted <AMOUNT> nodes.
  legs.push(ledgerLeg(supplier, -gross))

  const head = [
    '    <TALLYMESSAGE xmlns:UDF="TallyUDF">',
    '      <VOUCHER VCHTYPE="Purchase" ACTION="Create" OBJVIEW="Accounting Voucher View">',
    `        <DATE>${date}</DATE>`,
    `        <EFFECTIVEDATE>${date}</EFFECTIVEDATE>`,
    '        <VOUCHERTYPENAME>Purchase</VOUCHERTYPENAME>',
    '        <VOUCHERNUMBER></VOUCHERNUMBER>',
    `        <REFERENCE>${esc(invoiceNumber)}</REFERENCE>`,
    `        <PARTYLEDGERNAME>${esc(supplier)}</PARTYLEDGERNAME>`,
    '        <NARRATION>Billora register import</NARRATION>',
  ].join('\n')
  const tail = [
    '      </VOUCHER>',
    '    </TALLYMESSAGE>',
  ].join('\n')

  return { xml: `${head}\n${legs.join('\n')}\n${tail}` }
}

/**
 * Build a TallyPrime XML *Import Data* document of Purchase vouchers from approved
 * register lines. Incomplete lines are skipped (never emitted malformed); balanced
 * lines are wrapped in the standard ENVELOPE/IMPORTDATA/REQUESTDATA structure.
 */
export function buildTallyPrimePurchaseXml(lines: readonly TallyImportRowInput[]): TallyXmlResult {
  const vouchers: string[] = []
  let skippedCount = 0
  for (const line of lines) {
    const v = renderVoucher(line)
    if (v === null) { skippedCount += 1; continue }
    vouchers.push(v.xml)
  }

  if (vouchers.length === 0) {
    return { xml: '', voucherCount: 0, skippedCount }
  }

  const xml = [
    '<?xml version="1.0" encoding="UTF-8"?>',
    '<ENVELOPE>',
    '  <HEADER>',
    '    <TALLYREQUEST>Import Data</TALLYREQUEST>',
    '  </HEADER>',
    '  <BODY>',
    '    <IMPORTDATA>',
    '      <REQUESTDESC>',
    '        <REPORTNAME>Vouchers</REPORTNAME>',
    '      </REQUESTDESC>',
    '      <REQUESTDATA>',
    vouchers.join('\n'),
    '      </REQUESTDATA>',
    '    </IMPORTDATA>',
    '  </BODY>',
    '</ENVELOPE>',
    '',
  ].join('\n')

  return { xml, voucherCount: vouchers.length, skippedCount }
}

// lib/encryption/files.ts — envelope encryption for original invoice files.
//
// Design (approved Stage 1):
//   * One fresh random 256-bit DEK per file. The file bytes are encrypted with
//     AES-256-GCM under the DEK; the DEK is then wrapped with AES-256-GCM under
//     the master key (ENCRYPTION_MASTER_KEY). Ciphertext AND metadata are
//     authenticated (the key version + object context are bound as AAD), so a
//     byte of tampering or a swapped path fails closed.
//   * The container is versioned and self-describing so rotation is detectable
//     and decrypt needs no out-of-band IV/tag.
//   * Object paths are a deterministic HMAC under a DEDICATED path key
//     (STORAGE_PATH_KEY), so webhook redelivery resolves to the same object and
//     the invoices.file_object_path unique constraint keeps intake idempotent.
//   * NEVER a plaintext fallback: a missing/misconfigured key is a hard error.
//
// All operations are server-side only. This module reads no DB and stores no
// plaintext to disk; callers pass buffers and keep everything in memory.
import {
  createCipheriv, createDecipheriv, createHmac, randomBytes, timingSafeEqual,
} from 'node:crypto'

const ALG = 'aes-256-gcm'
const IV_LEN = 12
const TAG_LEN = 16
const DEK_LEN = 32
const KEY_VER_LEN = 8
const HEX_64 = /^[0-9a-fA-F]{64}$/

// Container header (all big-endian, fixed layout):
//   [0]      MAGIC 0xBF
//   [1]      SCHEMA_VER
//   [2..10]  KEY_VER (8 bytes, derived from the master key)
//   [10..22] DEK-wrap IV
//   [22..38] DEK-wrap auth tag
//   [38..70] wrapped DEK (32 bytes; GCM ciphertext == plaintext length)
//   [70..82] content IV
//   [82..98] content auth tag
//   [98..n]  content ciphertext
const MAGIC = 0xbf
const SCHEMA_VER = 0x01
const OFF_SCHEMA = 1
const OFF_KEYVER = 2
const OFF_IVK = 10
const OFF_TAGK = 22
const OFF_WDEK = 38
const OFF_IVC = 70
const OFF_TAGC = 82
const OFF_CT = 98

/** A configured key is missing or malformed (wrong length/charset). */
export class EncryptionConfigError extends Error {
  override name = 'EncryptionConfigError'
}

/** A stored container is corrupt, tampered, or not decryptable with this key. */
export class MediaDecryptFailed extends Error {
  override name = 'MediaDecryptFailed'
}

/** Decrypt failure that a retry can never fix (auth/tag/format), for job classification. */
export function isMediaDecryptFailure(error: unknown): boolean {
  return error instanceof Error && error.name === 'MediaDecryptFailed'
}

/**
 * A missing/malformed key is an operator-configuration error, not a transient
 * fault: no retry can succeed until the environment is fixed and the server
 * restarts. Callers use this to keep such a failure OUT of a webhook retry loop
 * (which would also re-download the same media for nothing).
 */
export function isEncryptionConfigFailure(error: unknown): boolean {
  return error instanceof Error && error.name === 'EncryptionConfigError'
}

/**
 * Non-throwing readiness probe for the file-encryption keys, for health /
 * startup diagnostics. Reports presence + well-formedness only; it never
 * returns or logs key material.
 */
export function hasFileKeyConfig(): { masterKeyPresent: boolean; pathKeyPresent: boolean } {
  return {
    masterKeyPresent: HEX_64.test(process.env.ENCRYPTION_MASTER_KEY ?? ''),
    pathKeyPresent: HEX_64.test(process.env.STORAGE_PATH_KEY ?? ''),
  }
}

function requireKey(hex: string | undefined, label: string): Buffer {
  if (!hex || !HEX_64.test(hex)) {
    throw new EncryptionConfigError(
      `${label} must be a 32-byte hex string (64 hex chars). Generate: node -e "console.log(require('crypto').randomBytes(32).toString('hex'))"`,
    )
  }
  return Buffer.from(hex, 'hex')
}

/** Stable 8-byte key fingerprint derived from the master key (rotation detection). */
export function deriveKeyVersion(masterKeyHex: string): string {
  const key = requireKey(masterKeyHex, 'ENCRYPTION_MASTER_KEY')
  return createHmac('sha256', key).update('billora-file-keyver').digest().subarray(0, KEY_VER_LEN)
    .toString('hex')
}

function keyVersionBytes(masterKeyHex: string): Buffer {
  const key = Buffer.from(masterKeyHex, 'hex')
  return createHmac('sha256', key).update('billora-file-keyver').digest().subarray(0, KEY_VER_LEN)
}

// Domain-separated AAD binds schema version + key version + object context, so
// both the DEK wrap and the content are authenticated against the intended path.
function baseAad(keyVerBytes: Buffer, context: string): Buffer {
  return Buffer.concat([
    Buffer.from([SCHEMA_VER]),
    keyVerBytes,
    Buffer.from(context, 'utf8'),
  ])
}

export interface EncryptedFile {
  /** Self-describing ciphertext container ready for object storage upload. */
  container: Buffer
  /** 8-byte hex key fingerprint; store on the invoice for rotation auditing. */
  keyVersion: string
  /** Plaintext byte length (accounting; the SHA-256 stays over plaintext). */
  sizeBytes: number
}

/**
 * Envelope-encrypt a file. `context` must be the same value at decrypt time
 * (we use the deterministic object path), binding the container to its slot.
 */
export function encryptFile(
  plain: Buffer, masterKeyHex: string, args: { context: string },
): EncryptedFile {
  const masterKey = requireKey(masterKeyHex, 'ENCRYPTION_MASTER_KEY')
  if (!plain || plain.length === 0) throw new EncryptionConfigError('encryptFile: empty plaintext')
  const keyVer = keyVersionBytes(masterKeyHex)
  const aad = baseAad(keyVer, args.context)

  const dek = randomBytes(DEK_LEN)

  // Wrap the DEK under the master key.
  const ivK = randomBytes(IV_LEN)
  const wrapCipher = createCipheriv(ALG, masterKey, ivK)
  wrapCipher.setAAD(Buffer.concat([aad, Buffer.from('WRAP', 'utf8')]))
  const wrappedDEK = Buffer.concat([wrapCipher.update(dek), wrapCipher.final()])
  const tagK = wrapCipher.getAuthTag()

  // Encrypt the file bytes under the DEK.
  const ivC = randomBytes(IV_LEN)
  const contentCipher = createCipheriv(ALG, dek, ivC)
  contentCipher.setAAD(Buffer.concat([aad, Buffer.from('DATA', 'utf8')]))
  const ciphertext = Buffer.concat([contentCipher.update(plain), contentCipher.final()])
  const tagC = contentCipher.getAuthTag()

  // Zero the DEK from this buffer reference best-effort (Node buffers aren't
  // guaranteed zeroable, but we avoid holding the key longer than this scope).
  dek.fill(0)

  const header = Buffer.alloc(OFF_CT)
  header[0] = MAGIC
  header[OFF_SCHEMA] = SCHEMA_VER
  keyVer.copy(header, OFF_KEYVER)
  ivK.copy(header, OFF_IVK)
  tagK.copy(header, OFF_TAGK)
  wrappedDEK.copy(header, OFF_WDEK)
  ivC.copy(header, OFF_IVC)
  tagC.copy(header, OFF_TAGC)

  const container = Buffer.concat([header, ciphertext])
  return { container, keyVersion: keyVer.toString('hex'), sizeBytes: plain.length }
}

/**
 * Decrypt a container produced by `encryptFile`. Any structural or
 * authentication failure throws MediaDecryptFailed — never returns plaintext
 * for a tampered/unauthentic object, and never falls back to returning input.
 */
export function decryptFile(
  container: Buffer, masterKeyHex: string, args: { context: string; expectedKeyVersion?: string },
): Buffer {
  const masterKey = requireKey(masterKeyHex, 'ENCRYPTION_MASTER_KEY')
  if (!Buffer.isBuffer(container) || container.length < OFF_CT) {
    throw new MediaDecryptFailed('container truncated or not a buffer')
  }
  if (container[0] !== MAGIC) throw new MediaDecryptFailed('bad magic')
  if (container[OFF_SCHEMA] !== SCHEMA_VER) {
    throw new MediaDecryptFailed(`unsupported schema version ${container[OFF_SCHEMA]}`)
  }
  const keyVer = container.subarray(OFF_KEYVER, OFF_KEYVER + KEY_VER_LEN)
  if (args.expectedKeyVersion && keyVer.toString('hex') !== args.expectedKeyVersion) {
    // Reject on version mismatch BEFORE attempting decrypt; fail closed.
    throw new MediaDecryptFailed('key-version-mismatch')
  }
  const aad = baseAad(keyVer, args.context)

  const ivK = container.subarray(OFF_IVK, OFF_IVK + IV_LEN)
  const tagK = container.subarray(OFF_TAGK, OFF_TAGK + TAG_LEN)
  const wrappedDEK = container.subarray(OFF_WDEK, OFF_WDEK + DEK_LEN)
  const ivC = container.subarray(OFF_IVC, OFF_IVC + IV_LEN)
  const tagC = container.subarray(OFF_TAGC, OFF_TAGC + TAG_LEN)
  const ciphertext = container.subarray(OFF_CT)

  let dek: Buffer
  try {
    const unwrap = createDecipheriv(ALG, masterKey, ivK)
    unwrap.setAAD(Buffer.concat([aad, Buffer.from('WRAP', 'utf8')]))
    unwrap.setAuthTag(tagK)
    dek = Buffer.concat([unwrap.update(wrappedDEK), unwrap.final()])
  } catch {
    throw new MediaDecryptFailed('dek-unwrap-failed')
  }

  try {
    const content = createDecipheriv(ALG, dek, ivC)
    content.setAAD(Buffer.concat([aad, Buffer.from('DATA', 'utf8')]))
    content.setAuthTag(tagC)
    const plain = Buffer.concat([content.update(ciphertext), content.final()])
    dek.fill(0)
    return plain
  } catch {
    dek.fill(0)
    throw new MediaDecryptFailed('content-auth-failed')
  }
}

/** Deterministic, non-guessable object path for a provider media id under a scope. */
export function deriveObjectKey(
  scope: 'invoices' | 'pending', mediaId: string, pathKeyHex: string,
): string {
  requireKey(pathKeyHex, 'STORAGE_PATH_KEY')
  if (!mediaId) throw new EncryptionConfigError('deriveObjectKey: empty media id')
  const h = createHmac('sha256', pathKeyHex).update(`${scope}:${mediaId}`).digest('hex')
  return `whatsapp/${scope}/${h}.bin`
}

/**
 * Deterministic, non-guessable object path for a generated workbook, keyed by
 * batch id under the same dedicated path key. Re-running a batch regenerates
 * the same `.xlsx` slot (upsert) instead of accumulating duplicates, while an
 * unrelated batch can never collide.
 *
 * §separate-files-per-category: an optional category token widens the HMAC seed
 * so a batch's multiple per-category registers occupy DISTINCT object slots and
 * never overwrite one another. With no token the seed is the historical
 * `exports:<batch>` string, so the purchase register (and every pre-split object)
 * keeps its EXACT prior key — full backward compatibility.
 */
export function deriveWorkbookKey(batchId: string, pathKeyHex: string, categoryToken?: string | null): string {
  requireKey(pathKeyHex, 'STORAGE_PATH_KEY')
  if (!batchId) throw new EncryptionConfigError('deriveWorkbookKey: empty batch id')
  const seed = categoryToken ? `exports:${batchId}:${categoryToken}` : `exports:${batchId}`
  const h = createHmac('sha256', pathKeyHex).update(seed).digest('hex')
  return `whatsapp/exports/${h}.xlsx`
}

/** Constant-time compare for hex key versions (used by the legacy migrator). */
export function keyVersionMatches(a: string, b: string): boolean {
  const ab = Buffer.from(a, 'hex')
  const bb = Buffer.from(b, 'hex')
  return ab.length === bb.length && timingSafeEqual(ab, bb)
}

/**
 * Read + validate the file-encryption key configuration from the environment.
 * Fail-closed: any malformed/missing key throws EncryptionConfigError so intake
 * never stores plaintext by accident.
 */
export function resolveFileKeyConfig(): { masterKeyHex: string; pathKeyHex: string } {
  const masterKeyHex = process.env.ENCRYPTION_MASTER_KEY ?? ''
  const pathKeyHex = process.env.STORAGE_PATH_KEY ?? ''
  if (!HEX_64.test(masterKeyHex)) {
    throw new EncryptionConfigError('ENCRYPTION_MASTER_KEY is missing or not a 64-hex-char key')
  }
  if (!HEX_64.test(pathKeyHex)) {
    throw new EncryptionConfigError('STORAGE_PATH_KEY is missing or not a 64-hex-char key')
  }
  return { masterKeyHex, pathKeyHex }
}

// lib/media/file-type.ts — authoritative, dependency-free media typing for intake.
//
// WhatsApp's declared mime_type / filename / extension are HINTS, never proof:
// a caller can claim image/png while sending bytes of any other shape. We sniff
// the ACTUAL leading bytes and only accept formats whose full path works
// (validate -> encrypt -> private store -> decrypt -> vision -> ledger). Formats
// that need a conversion we do not perform (HEIC/HEIF) are DETECTED so the user
// gets a precise reply, but never accepted — we do not advertise what we cannot
// process. This module imports nothing aliased so it is unit-testable in
// isolation (no logger, no env, no DB, no network).
//
// Limits guard oversized input, decompression bombs and resource exhaustion. We
// never trust a declared size alone: the real downloaded byte length is capped,
// and image dimensions / PDF page counts are parsed from the bytes themselves.

export type MediaKind = 'pdf' | 'jpeg' | 'png' | 'webp' | 'heif' | 'gif' | 'tiff' | 'unknown'

// Hard caps. MAX_FILE_BYTES stays under the 20 MiB bucket limit so a stored
// object is never rejected for size after we already accepted the upload.
export const MAX_FILE_BYTES = 15 * 1024 * 1024 // 15 MiB
export const MAX_IMAGE_PIXELS = 40_000_000 // ~40 MP — well past any real invoice
export const MAX_PDF_PAGES = 20
export const MIN_FILE_BYTES = 32 // anything smaller is not a real document

// Formats whose complete pipeline works. HEIC/HEIF are intentionally absent.
const ACCEPTED: Partial<Record<MediaKind, { mime: string; ext: string }>> = {
  pdf: { mime: 'application/pdf', ext: 'pdf' },
  jpeg: { mime: 'image/jpeg', ext: 'jpg' },
  png: { mime: 'image/png', ext: 'png' },
  webp: { mime: 'image/webp', ext: 'webp' },
}

const GENERIC_UNSUPPORTED_REPLY =
  'Sorry, I can only process invoice PDF, JPG, PNG or WebP files. Please resend as one of those — or type Help.'

/** latin1 lets us compare raw leading bytes without any decoding surprises. */
function ascii(buf: Buffer, start: number, end: number): string {
  return buf.subarray(start, end).toString('latin1')
}

/** Sniff the real container from magic bytes. Never trusts declared metadata. */
export function sniffMediaKind(buf: Buffer): MediaKind {
  if (!Buffer.isBuffer(buf) || buf.length < 4) return 'unknown'
  // %PDF-
  if (buf[0] === 0x25 && ascii(buf, 0, 5) === '%PDF-') return 'pdf'
  // JPEG: FF D8 FF
  if (buf[0] === 0xff && buf[1] === 0xd8 && buf[2] === 0xff) return 'jpeg'
  // PNG: 89 50 4E 47 0D 0A 1A 0A
  if (buf.length >= 8
    && buf[0] === 0x89 && buf[1] === 0x50 && buf[2] === 0x4e && buf[3] === 0x47
    && buf[4] === 0x0d && buf[5] === 0x0a && buf[6] === 0x1a && buf[7] === 0x0a) return 'png'
  // WEBP: 'RIFF' <size> 'WEBP'
  if (buf.length >= 12 && ascii(buf, 0, 4) === 'RIFF' && ascii(buf, 8, 12) === 'WEBP') return 'webp'
  // HEIF/HEIC: ISO-BMFF 'ftyp' at byte 4 with a HEIC/AVC brand at 8.
  if (buf.length >= 12 && ascii(buf, 4, 8) === 'ftyp') {
    const brand = ascii(buf, 8, 12).toLowerCase()
    if (/^(heic|heix|heim|heis|hevc|hevx|mif1|msf1|avci|avcs|avcp)$/.test(brand)) return 'heif'
  }
  // GIF: 'GIF8'
  if (ascii(buf, 0, 4) === 'GIF8') return 'gif'
  // TIFF: 'II*\0' (little) or 'MM\0*' (big)
  if ((buf[0] === 0x49 && buf[1] === 0x49 && buf[2] === 0x2a && buf[3] === 0x00)
    || (buf[0] === 0x4d && buf[1] === 0x4d && buf[2] === 0x00 && buf[3] === 0x2a)) return 'tiff'
  return 'unknown'
}

/** True only for kinds that can actually be extracted end to end. */
export function isSupportedMediaKind(kind: MediaKind): boolean {
  return ACCEPTED[kind] !== undefined
}

/** Read canvas dimensions from PNG / JPEG / WebP headers. Defensive: null on any anomaly. */
export function readImageDimensions(kind: MediaKind, buf: Buffer): { width: number; height: number } | null {
  try {
    if (kind === 'png') {
      if (buf.length < 24 || ascii(buf, 12, 16) !== 'IHDR') return null
      const width = buf.readUInt32BE(16)
      const height = buf.readUInt32BE(20)
      return width > 0 && height > 0 ? { width, height } : null
    }
    if (kind === 'jpeg') {
      let off = 2
      while (off + 9 < buf.length) {
        if (buf[off] !== 0xff) { off += 1; continue }
        const marker = buf[off + 1]
        // Standalone markers carry no length segment.
        if (marker === 0x01 || (marker >= 0xd0 && marker <= 0xd9)) { off += 2; continue }
        // Start-Of-Frame (baseline/progressive): dims live right after the header.
        if (marker >= 0xc0 && marker <= 0xcf && marker !== 0xc4 && marker !== 0xc8 && marker !== 0xcc) {
          const height = buf.readUInt16BE(off + 5)
          const width = buf.readUInt16BE(off + 7)
          return width > 0 && height > 0 ? { width, height } : null
        }
        const segLen = buf.readUInt16BE(off + 2)
        if (segLen < 2) return null
        off += 2 + segLen
      }
      return null
    }
    if (kind === 'webp') {
      const fourcc = ascii(buf, 12, 16)
      if (fourcc === 'VP8X' && buf.length >= 30) {
        const width = (buf[24] | (buf[25] << 8) | (buf[26] << 16)) + 1
        const height = (buf[27] | (buf[28] << 8) | (buf[29] << 16)) + 1
        return { width, height }
      }
      if (fourcc === 'VP8L' && buf.length >= 25 && buf[20] === 0x2f) {
        const bits = buf.readUInt32LE(21)
        const width = (bits & 0x3fff) + 1
        const height = ((bits >>> 14) & 0x3fff) + 1
        return { width, height }
      }
      if (fourcc === 'VP8 ' && buf.length >= 30) {
        // Locate the 3-byte sync code (0x9d 0x01 0x2a) that precedes the size.
        for (let i = 20; i < 40 && i + 7 < buf.length; i++) {
          if (buf[i] === 0x9d && buf[i + 1] === 0x01 && buf[i + 2] === 0x2a) {
            const width = buf.readUInt16LE(i + 3) & 0x3fff
            const height = buf.readUInt16LE(i + 5) & 0x3fff
            return width > 0 && height > 0 ? { width, height } : null
          }
        }
        return null
      }
    }
    return null
  } catch {
    return null
  }
}

/**
 * Best-effort PDF page count by scanning the object structure for `/Type /Page`
 * (excluding `/Pages`). Linearized/object-stream PDFs can under-report, so a
 * null result (or 0) is treated as "unknown" and is NOT a rejection — the byte
 * cap is the hard guard; this is a defense-in-depth check for flat PDFs.
 */
export function estimatePdfPageCount(buf: Buffer): number | null {
  if (buf.length < 8 || ascii(buf, 0, 5) !== '%PDF-') return null
  const text = buf.toString('latin1')
  let count = 0
  const re = /\/Type\s*\/Page(?![s\/])/g
  while (re.exec(text) !== null) count++
  return count > 0 ? count : null
}

export interface ValidatedMedia {
  kind: MediaKind
  /** Authoritative MIME from the SNIFFED bytes (never the declared hint). */
  mime: string
  /** Validated safe extension for the sniffed kind. */
  extension: string
  width: number | null
  height: number | null
  pages: number | null
}

/** A file whose real format is not one we can process. Retry can never help. */
export class UnsupportedMediaError extends Error {
  override name = 'UnsupportedMediaError'
  constructor(message: string, readonly userReply: string) {
    super(message)
  }
}

/** A file that exceeds a byte / pixel / page limit. Retry can never help. */
export class OversizeMediaError extends Error {
  override name = 'OversizeMediaError'
  constructor(message: string, readonly userReply: string) {
    super(message)
  }
}

/**
 * Validate downloaded bytes: sniff the real type, reject unsupported/mismatched
 * content, and enforce size, pixel and PDF-page limits. Declared MIME/filename
 * are ignored except as a hint for the reply. Throws UnsupportedMediaError or
 * OversizeMediaError (both permanent) on rejection; returns the authoritative
 * validated descriptor on success.
 */
export function validateInvoiceMedia(args: {
  buffer: Buffer
  declaredMime?: string | null
  declaredFilename?: string | null
}): ValidatedMedia {
  const { buffer } = args
  if (!Buffer.isBuffer(buffer) || buffer.length < MIN_FILE_BYTES) {
    throw new UnsupportedMediaError('MEDIA_TOO_SMALL', GENERIC_UNSUPPORTED_REPLY)
  }
  const kind = sniffMediaKind(buffer)
  if (kind === 'heif') {
    throw new UnsupportedMediaError(
      'MEDIA_UNSUPPORTED_HEIF',
      "I can't read HEIC/HEIF photos. Please resend the invoice as a PDF, JPG or PNG screenshot.",
    )
  }
  const accepted = ACCEPTED[kind]
  if (!accepted) {
    throw new UnsupportedMediaError(`MEDIA_UNSUPPORTED_${kind.toUpperCase()}`, GENERIC_UNSUPPORTED_REPLY)
  }
  if (buffer.length > MAX_FILE_BYTES) {
    throw new OversizeMediaError(
      'MEDIA_TOO_LARGE_BYTES',
      `That file is too large for me to process (over ${Math.floor(MAX_FILE_BYTES / (1024 * 1024))} MB). Please resend a smaller PDF, JPG or PNG.`,
    )
  }

  let width: number | null = null
  let height: number | null = null
  let pages: number | null = null
  if (kind === 'pdf') {
    pages = estimatePdfPageCount(buffer)
    if (pages !== null && pages > MAX_PDF_PAGES) {
      throw new OversizeMediaError(
        'MEDIA_TOO_MANY_PAGES',
        `That PDF has more than ${MAX_PDF_PAGES} pages. Please send only the invoice page(s).`,
      )
    }
  } else {
    const dims = readImageDimensions(kind, buffer)
    if (dims) {
      width = dims.width
      height = dims.height
      if (width * height > MAX_IMAGE_PIXELS) {
        throw new OversizeMediaError(
          'MEDIA_TOO_MANY_PIXELS',
          'That image is too large (too many pixels). Please resend a lower-resolution photo or a PDF.',
        )
      }
    }
  }
  return { kind, mime: accepted.mime, extension: accepted.ext, width, height, pages }
}

/** A format/size rejection that no retry can fix — settle the intake as final. */
export function isPermanentMediaFormatFailure(error: unknown): boolean {
  return error instanceof Error
    && (error.name === 'UnsupportedMediaError' || error.name === 'OversizeMediaError')
}

// lib/supabase/server.ts — Privileged service role execution pool.
import { createClient } from '@supabase/supabase-js'
import { getEnv } from '@/lib/env'

// Server-side only — uses the secret key. NEVER import from client components.
let cached: ReturnType<typeof createClient> | null = null

export function getSupabaseAdmin() {
  if (cached) return cached
  const env = getEnv()
  if (!env.SUPABASE_SECRET_KEY) {
    throw new Error('Missing Supabase env: set SUPABASE_SECRET_KEY, then restart the server.')
  }
  cached = createClient(env.NEXT_PUBLIC_SUPABASE_URL, env.SUPABASE_SECRET_KEY, {
    auth: { persistSession: false },
  })
  return cached
}

/**
 * Resolve a logical bucket to its configured slug. Intake/exports code must
 * never hardcode a bucket name — this reads SUPABASE_INVOICES_BUCKET /
 * SUPABASE_EXPORTS_BUCKET (defaults match the provisioned private buckets).
 */
export function bucketFor(kind: 'invoices' | 'exports'): string {
  const env = getEnv()
  return kind === 'invoices' ? env.SUPABASE_INVOICES_BUCKET : env.SUPABASE_EXPORTS_BUCKET
}

  get(_t, prop) {
