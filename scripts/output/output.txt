export async function GET(req: Request) {
  const { searchParams } = new URL(req.url);
  const mode = searchParams.get('hub.mode');
  const receivedToken = searchParams.get('hub.verify_token');
  const expectedToken = process.env.WHATSAPP_VERIFY_TOKEN || '';
  // Minimal operational log: outcome only, no token material or comparison diagnostics.
  const verified = mode === 'subscribe' && !!expectedToken && !!receivedToken
    && safeTokenEqual(receivedToken, expectedToken);
  logger.info('[WhatsApp] verify attempt', { mode: mode ?? null, verified });
  if (verified) {
    return new Response(searchParams.get('hub.challenge') || '', {
      status: 200, headers: { 'Content-Type': 'text/plain' },
    });
  }
  return new Response('Forbidden', { status: 403 });
}

/**
 * Humanize a single editor field's validation failure (from the shared
 * parseReviewCorrections) so the reviewer learns exactly WHICH field and WHY it
 * was rejected â€” never a silent no-op. The codes mirror lib/invoices/review.ts.
 */
function describeFieldErrors(label: string, errors: Array<{ field: string; code: string }>): string {
  const reason = (code: string): string =>
    code === 'TOO_LONG' ? 'is too long'
      : code === 'INVALID_AMOUNT' ? 'is not a readable amount'
        : code === 'NEGATIVE_NOT_ALLOWED' ? 'cannot be negative'
          : code === 'INVALID_TYPE' ? 'was not a valid value'
            : 'was not accepted';
  return `I could not save that change â€” ${label} ${reason(errors[0]?.code ?? '')}.`;
}

        logger.warn('[WhatsApp] review confirm list send failed', { sender: label, outcome });
      }
      track(outcome);
    } catch (e) {
      logger.warn('[WhatsApp] review confirm list unavailable', {
        sender: label, error: e instanceof Error ? e.message : String(e),
      });
    }
  };
  const countNeedsReview = async (): Promise<number> => {
    if (!workspaceId) return 0;
    const { count } = await db.from('invoices').select('id', { count: 'exact', head: true })
      .eq('workspace_id', workspaceId).eq('status', 'needs_review');
    return count ?? 0;
  };
  // "Done" is a real finalization, not just an ack. It re-affirms a register job
  // for every eligible batch, then reads the CURRENT state and sends a TRUTHFUL
  // result. It never claims the Excel is ready until a verified generated_workbooks
  // row exists (build_register inserts that row only after an encrypted upload
  // succeeded), so a ready claim always implies the stored file is available.
  // While extraction/register jobs are still running it says "still working" and
  // defers the final result to the register job's delivery. Repeated Done is
  // idempotent: scheduling is deduped and the reads never mutate processing state.
  // Â§immediate-register: register generation is an async worker job, so a Done
  // tapped right after approval often finds the workbook still compiling. Wait a
  // SHORT, bounded window for it to land so the common "approve then Done" path can
  // still ATTACH the Excel in this very reply, instead of only reaching the chat on
  // the next inbound. If a register is still running after the window we keep the
  // truthful "still working" reply and the outbox flush delivers on the next inbound.
  // Â§never-partial: the window ends on QUIESCENCE, not on "a workbook exists". Mid-burst
  // there is always some live register, but it is a PARTIAL snapshot of the month â€” and
  // the delivery consumer deliberately holds a flush while any job is live, so returning
  // early here attached nothing and taught the owner to type Done twice. A workspace with
  // extraction or register work still running is never "ready" for this reply.
  const waitForRegisterReady = async (scope: DeliveryScope, first: FinalizeCounts): Promise<FinalizeCounts> => {
    if (!workspaceId) return first;
    if (first.outstanding === 0 && first.compiling === 0) return first;
    // Nudge the worker so the queued register gets a tick inside this window
    // (ensureBatchRegisters only wakes when it newly scheduled one).
    wakeWorker();
    let latest = first;
    const deadline = Date.now() + REGISTER_READY_WAIT_MS;
    while (Date.now() < deadline) {
      await new Promise((r) => setTimeout(r, 1200));
      try {
        latest = await collectFinalizeState(db, workspaceId, scope);
      } catch {
        return latest;
      }
      if (latest.outstanding === 0 && latest.compiling === 0) return latest;
    }
    return latest;
  };
  // Â§truthful-promise: the Done reply may only promise an unprompted handover when
  // the SANCTIONED opt-in path can actually deliver it â€” consent + the separate
  // sealing key + inside Meta's 24h window (canProactivelyPush), with the sealed
  // number still opening. `pushOffer` is the honest middle state: the operator
  // turned the feature on but this owner has not opted in yet, so naming the
  // "Proactive on" keyword is real help instead of a dead end. Never leaks the
  // phone or the sealed value â€” only two booleans leave here.
  const proactivePushState = async (): Promise<{ pushOk: boolean; pushOffer: boolean }> => {
    if (!proactiveDeliveryEnabled() || !proactiveKeyConfigured()) {
      return { pushOk: false, pushOffer: false };
    }
    try {
      const id = await loadProactiveIdentity(db, hmac);
      if (!id) return { pushOk: false, pushOffer: false };
      const pushOk = canProactivelyPush({ consent: id.consent, lastActivityMs: id.lastActivityMs })
        && openWaId(id.sealed) != null;
      return { pushOk, pushOffer: !id.consent };
    } catch {
      return { pushOk: false, pushOffer: false };
    }
  };
  const deliverDone = async (): Promise<void> => {
    if (!workspaceId) { await deliverText('Nothing to finalize yet â€” send Hi to start, or just send an invoice.'); return; }
    // Serve ONLY this sender's tenant: individuals see client-less batches,
    // org senders see their selected client's. An older month still sitting at
    // 'ready' for another client can then never ride along on this reply.
    const scope = { clientId: clientId ?? null };
    try {
      await ensureBatchRegisters(db, workspaceId, scope);
      const counts = await waitForRegisterReady(scope, await collectFinalizeState(db, workspaceId, scope));
      const delivered = await deliverOwedRegisters();
      const base = (process.env.PUBLIC_APP_URL ?? '').replace(/\/$/, '');
      // Deterministic link: the LATEST accounting period among the scoped ready
      // registers â€” never an arbitrary first row (the wrong-month leak path).
      const primary = pickPrimaryWorkbook(counts.readyMonths ?? [])?.workbookId ?? null;
      const link = base && primary
        ? `${base}/api/exports/${primary}/download`
        : null;
      const { pushOk, pushOffer } = await proactivePushState();
      await deliverText(composeDoneMessage(counts, { link, delivered: delivered > 0, pushOk, pushOffer }));
    } catch (e) {
      // Finalization is read/schedule only; a failure here must never lose work or
      // rewind intake. Fall back to an honest "queued" message (no ready claim).
      logger.warn('[WhatsApp] Done finalization deferred', {
        sender: label, error: e instanceof Error ? e.message : String(e),
      });
      await deliverText('Done â€” your uploads are queued. I\'ll send your register here once processing finishes.');
    }
  };
  const sendStateMessage = async (): Promise<void> => {
    if (result.note === 'individual-selected') {
      const held = assignedNow > 0 ? ` I also filed the ${assignedNow} invoice file(s) you sent earlier.` : '';
      await deliverText(`Your workspace is ready. Send invoice PDFs or photos here.${held}`);
    } else if (result.note.startsWith('org-name:')) {
      const clients = await listClients(db, workspaceId as string);
      if (clients.length > 0) { track(await sendClientList(to, clients)); }
      else {
        await deliverText('Organization workspace created. No clients yet â€” send the first client name.');
      }
    } else if (result.note === 'reprompt-client-list' || result.note === 're-list-clients'
      || result.note === 'change-client-requested') {
      if (workspaceId) track(await sendClientList(to, await listClients(db, workspaceId)));
    } else if (result.note.startsWith('client-name:')) {
      await deliverText('Client saved. Send invoice PDFs or photos here. Optional: send "My GSTIN <number>" so I never mistake your own GSTIN for a supplier\'s.');
    } else if (result.note === 'client-selected') {
      await deliverText('Client selected. Send invoice PDFs or photos here. Optional: send "My GSTIN <number>" so I never mistake your own GSTIN for a supplier\'s.');
    } else if (result.note === 'client-selection-rejected') {
      await deliverText('That client is not available. Choose from the list below.');
      if (workspaceId) track(await sendClientList(to, await listClients(db, workspaceId)));
    } else if (result.note === 'gstin-set' || result.note === 'gstin-clear') {
      // Own-GSTIN capture reply (Â§P0-1b). The set/clear note is normalised by the
      // side-effects block, and the write already happened before this send.
      await deliverText(gstinReplyBody ?? 'Done.');
    } else if (result.note === 'proactive-enable' || result.note === 'proactive-disable') {
      // Proactive opt-in/opt-out reply (Â§P1-6). Consent was persisted (or softly
      // declined) in the side-effects block before this send.
      await deliverText(proactiveReplyBody ?? 'Done.');
    } else if (result.note === 'awaiting-pending-confirm' || result.note === 'pending-confirm-stale') {
      // noop: handled by the confirm branch below / generic text.
    } else if (result.note === 'pending-confirm-yes') {
      await deliverText(assignedNow > 0
        ? `Applied to ${assignedNow} file(s) â€” they're queued for processing. Send more invoices anytime.`
        : 'Applied. Nothing was held for you. Send invoice PDFs or photos here.');
    } else if (result.note === 'welcome-back') {
      const waiting = await countWaitingUploads(db, hmac);
      await deliverText(waiting > 0
        ? `Welcome back to Billora. Note: ${waiting} earlier file(s) are still held for a client â€” send Change client to assign them, or just send more invoices.`
        : 'Welcome back to Billora â€” send invoice PDFs or photos here. Send Menu for options.');
    } else if (result.note === 'done') {
      await deliverDone();
    } else if (result.note === 'status-requested') {
      const waiting = await countWaitingUploads(db, hmac).catch(() => 0);
      await deliverText(!workspaceId
        ? "You're not set up yet â€” send Hi to begin, or just send an invoice and I'll hold it safely."
        : `You're ready. Send invoice PDFs or photos anytime.${waiting > 0 ? ` ${waiting} file(s) are held for a client â€” send Change client to assign them.` : ''} Send Done when this batch is finished, or Menu for options.`);
    } else if (result.note === 'review-requested') {
      const waiting = await countWaitingUploads(db, hmac).catch(() => 0);
      let body: string;
      if (!workspaceId) {
        body = "You're not set up yet â€” send Hi to begin, or just send an invoice.";
      } else {
        try {
          // Same tenant scope as Done: a Review reply can only ever carry this
          // sender's own client's ready registers, latest period first.
          const scope = { clientId: clientId ?? null };
          const counts = await collectFinalizeState(db, workspaceId, scope);
          const delivered = await deliverOwedRegisters();
          const base = (process.env.PUBLIC_APP_URL ?? '').replace(/\/$/, '');

            ? `${base}/api/exports/${primary}/download`
            : null;

        } catch {

            : 'Nothing needs your attention right now â€” keep sending invoices.';

      // recipient GSTIN recurring on >=3 of their bills is a strong candidate â€”
      // but a machine must NEVER silently rewrite a fact that drives the
      // supplier==buyer guard. Surface a plain-English SUGGESTION and let the
      // reviewer confirm it (reply `My GSTIN <value>`). Best-effort: a missing
      // key, read or decrypt failure leaves the truthful reply untouched.
      if (workspaceId && clientId) {
        try {
          const { masterKeyHex: suggestKey } = resolveFileKeyConfig();
          const hint = await suggestOwnGstin(db, { workspaceId, clientId, masterKeyHex: suggestKey, minCount: 3 });
          if (hint.suggestion) {
            body += `\n\nAlso: ${hint.suggestion} looks like it might be your own GSTIN â€” it appears as the recipient on ${hint.count} of your invoices. If that's right, reply "My GSTIN ${hint.suggestion}" and I'll never file it as a supplier. If not, just ignore this.`;
          }
        } catch { /* suggestion is a bonus; never break the review reply */ }
      }
      const held = waiting > 0 ? ` Also: ${waiting} file(s) are held for a client â€” send Change client to assign them.` : '';
      await deliverText(body + held);
      // Â§P1-7: one confirmable page (with an "Approve all" control + a "Next" row
      // when more remain) instead of a single unbounded 10-row list. Best-effort.
      // Â§user #2: ONE combined Review & edit screen. The list rows now carry

      // separate edit picker is gone. `listPendingReviewPage` widens its eligibility
      // to cover EVERY needs_review invoice (including unbatched / numberless ones
      // whose only problem is a missing date or number), so nothing becomes
      // unreachable now that the second list is removed. Best-effort: a read/send

      if (workspaceId) {
        await sendReviewPage(1);
      }
    } else if (result.note === 'review-page') {
      // Â§P1-7: page through a review list longer than the 10-row cap. No re-summary
      // (the reviewer already saw it on the first Review) â€” just the requested page.
      await sendReviewPage(result.page ?? 2);
    } else if (result.note === 'review-approve-all') {
      // Â§P1-7: bulk approval already ran (with re-validation) in the side-effects
      // block; deliver the honest counts, then re-show any row that stayed flagged.
      await deliverText(approveAllSummary ?? 'Done.');
      await sendReviewPage(1);
    } else if (result.note === 'review-confirm') {
      await deliverText(composeConfirmResultMessage(confirmOutcome ?? { ok: false, error: 'FAULT' }));
    } else if (result.note === 'review-confirm-invalid') {
      await deliverText('That confirmation is no longer valid â€” send Review to see the current list.');
    } else if (result.note.startsWith('review-edit')) {
      // One text reply for the whole editor turn (open / correct / approve /
      // cancel / help). Already computed during side effects, BEFORE persist.
      await deliverText(editReplyBody ?? REVIEW_EDIT_HELP);
    } else if (result.outbound.kind === 'menu_list') {
      await deliverMenu(result.outbound.workspace);
    } else if (result.outbound.kind === 'text') {
      await deliverText(result.outbound.body);
    } else if (result.outbound.kind === 'account_type_buttons') {
      track(await sendAccountTypeButtons(to));
    }

    if (result.nextState === 'awaiting_pending_confirm' && pendingClientId && workspaceId) {
      const waiting = await countWaitingUploads(db, hmac);
      const clients = await listClients(db, workspaceId);
      const chosen = clients.find((c) => c.id === pendingClientId);

    }
  };
  await sendStateMessage();
  if (sendOutcome !== 'ok') {
    logger.warn('[WhatsApp] outbound send deferred; processing state persisted', {
      sender: label, outcome: sendOutcome,
    });
  }
  return { workspaceId: result.workspaceId ?? workspaceId, clientId: result.clientId ?? clientId };
}

export async function POST(request: Request) {
  try {
    const body = await readRequestBodyLimited(request, WEBHOOK_BODY_MAX_BYTES);
    if (!body) {
      return NextResponse.json({ status: 'SKIPPED_EMPTY_BODY' }, { status: 200 });
    }
    const rawText = body.toString('utf8');
    if (!rawText || rawText.trim().length === 0) {
      return NextResponse.json({ status: 'SKIPPED_EMPTY_BODY' }, { status: 200 });
    }
    // Signature gate: X-Hub-Signature-256 over the RAW body, verified before any
    // intake row is created. Production fails closed when the app secret is
    // unconfigured; dev without a secret proceeds (local/smoke testing only).
    const appSecret = process.env.WHATSAPP_APP_SECRET || '';
    if (!appSecret && process.env.NODE_ENV === 'production') {
      logger.error('[WhatsApp] WHATSAPP_APP_SECRET unconfigured in production; delivery rejected');
      return new Response('Forbidden', { status: 401 });
    }
    if (appSecret && !verifyWebhookSignature(rawText, request.headers.get('x-hub-signature-256'), appSecret)) {
      logger.warn('[WhatsApp] webhook signature invalid; delivery rejected');
      return new Response('Forbidden', { status: 401 });
    }
    let payload: { entry?: Array<{ changes?: Array<{ value?: {
      messages?: IncomingMessage[]; statuses?: DeliveryStatus[];
    } }> }> };
    try { payload = JSON.parse(rawText); }
    catch {
      logger.info('[WhatsApp] non-JSON body skipped', {
        preview: rawText.slice(0, 200), contentType: request.headers.get('content-type'),
      });
      return NextResponse.json({ status: 'SKIPPED_NON_JSON_BODY' }, { status: 200 });
    }
    const changes = (payload.entry ?? []).flatMap((entry) => entry.changes ?? []);
    const values = changes.map((change) => change.value).filter((value) => value !== undefined);
    const messages = values.flatMap((value) => value?.messages ?? []);
    const statuses = values.flatMap((value) => value?.statuses ?? []);
    for (const delivery of statuses) {
      logger.info('[WhatsApp] delivery status received', {
        providerMessageIdSuffix: delivery.id?.slice(-8) ?? null,
        status: delivery.status ?? 'unknown',
        timestamp: delivery.timestamp ?? null,
        errorCodes: (delivery.errors ?? []).map((error) => error.code).filter((code) => code !== undefined),
      });
    }
    if (messages.length === 0) {
      // Delivery receipts/statuses carry no inbound message to process.
      logger.info('[WhatsApp] non-message delivery skipped', {
        hasMessages: messages.length > 0, statusCount: statuses.length,
      });
      return NextResponse.json({ status: 'SKIPPED_STATUS_UPDATE' }, { status: 200 });
    }
    const db = getSupabaseAdmin();
    // Persist every normalized message before claiming or doing synchronous
    // intake work. The worker can replay these encrypted envelopes if this
    // invocation dies and Meta does not redeliver the webhook.
    for (const message of messages) {
      if (!message?.id) continue;
      await storeReplayableWebhookMessage(db, message.id, { ...message, id: message.id });
    }
    // Process EVERY message in the delivery: a 503 makes Meta redeliver the
    // whole payload and the intake ledger skips messages already accepted.
    let retryable = false;
    for (const message of messages) {
      if (!message?.id) continue;
      if (await handleSingleMessage(db, message) === 'retryable') retryable = true;
    }

    async function readRequestBodyLimited(request: Request, maxBytes: number): Promise<Buffer | null> {
      const declaredLength = Number(request.headers.get('content-length'))
      if (Number.isFinite(declaredLength) && declaredLength > maxBytes) {
        throw new Error('WEBHOOK_BODY_TOO_LARGE')
      }
      if (!request.body) return null

      const reader = request.body.getReader()
      const chunks: Buffer[] = []
      let total = 0
      try {
        while (true) {
          const { done, value } = await reader.read()
          if (done) break
          total += value.byteLength
          if (total > maxBytes) {
            await reader.cancel('webhook body exceeds limit')
            throw new Error('WEBHOOK_BODY_TOO_LARGE')
          }
          chunks.push(Buffer.from(value))
        }
      } finally {
        reader.releaseLock()
      }
      return total === 0 ? null : Buffer.concat(chunks, total)
    }
    if (retryable) {
      return NextResponse.json({ status: 'RETRYABLE_PROCESSING_ERROR' }, { status: 503 });
    }
    return NextResponse.json({ status: 'ACCEPTED_FOR_QUEUE' }, { status: 200 });
  } catch (error: unknown) {
    const detail = error instanceof Error ? `${error.name}: ${error.message}` : String(error);
    logger.error('[Gateway Interface Router] runtime anomaly', { error: detail });
    const code = /Missing Supabase env|Invalid environment/i.test(detail) ? 'CONFIG_ERROR'
      : /webhook_events|relation .* does not exist/i.test(detail) ? 'DB_TABLE_MISSING'
      : 'SYSTEM_ROUTING_FAULT';
    const retryable = /CONFIG_ERROR|DB_TABLE_MISSING/.test(code) ? 503 : 500;
    return NextResponse.json({ status: code, detail }, { status: retryable });
  }
}

type MessageOutcome = 'processed' | 'retryable'

/**
 * Claim -> process -> durably accept exactly one provider message. Only an
 * `accepted` row suppresses reprocessing; `received`/`failed` rows reprocess
 * on Meta redelivery. Retries stay idempotent end to end.
 */
async function handleSingleMessage(db: Db, message: IncomingMessage): Promise<MessageOutcome> {
  const providerMessageId: string = message.id as string;
  const shortId = providerMessageId.slice(-8);
  logger.info('[WhatsApp] webhook received', {
    hasMessages: true, shortId, messageType: message.type ?? null, statuses: null,
  });
  let intake: Awaited<ReturnType<typeof claimIntakeRow>>;
  try {
    intake = await claimIntakeRow(db, providerMessageId);
  } catch (e) {
    const detail = e instanceof Error ? e.message : String(e);
    logger.error('[WhatsApp] intake claim failed', { shortId, error: detail });
    return 'retryable';
  }
  if (!intake.claimed) {
    if (isDurablyAccepted(intake.row)) {
      logger.info('[WhatsApp] duplicate accepted event skipped', { shortId });
      return 'processed';
    }
    // Another invocation owns the live lease. A 503 asks Meta to retry; returning
    // 200 could strand this event if its current owner crashes before acceptance.
    logger.info('[WhatsApp] duplicate delivery deferred; active claim exists', {
      shortId, priorStatus: intake.row.status, leaseExpiresAt: intake.row.claim_expires_at,
    });
    return 'retryable';
  }
  const claimToken = intake.claimToken;
  if (!claimToken) {
    logger.error('[WhatsApp] acquired intake claim has no owner token', { shortId });
    return 'retryable';
  }
  if (!intake.created) logger.info('[WhatsApp] retry acquired exclusive claim', {
    shortId, priorStatus: intake.row.status, attempt: intake.row.attempts,
  });
  let claimLost = false;
  const heartbeat = setInterval(() => {
    void renewIntakeClaim(db, providerMessageId, claimToken).then((renewed) => {
      if (!renewed) {
        claimLost = true;
        logger.error('[WhatsApp] intake claim heartbeat rejected; owner is stale', { shortId });
      }
    }).catch((heartbeatError) => {
      logger.warn('[WhatsApp] intake claim heartbeat failed', {
        shortId, error: heartbeatError instanceof Error ? heartbeatError.message : 'unknown',
      });
    });
  }, 45_000);
  heartbeat.unref?.();
  // A crash leaves a leased processing row; Meta redelivery can reclaim it after expiry.
  try {
    await processMessage(db, message, providerMessageId, shortId);
    if (claimLost) throw new Error('INTAKE_CLAIM_LOST: heartbeat rejected this invocation');
    await markIntakeAccepted(db, providerMessageId, claimToken);
  } catch (e) {
    const detail = e instanceof Error ? `${e.name}: ${e.message}` : String(e);
    const code = (e instanceof Error ? e.message : detail).split(':')[0] || 'PROCESSING_ERROR';
    if (isPermanentMediaFailure(e)) {
      // Expired/blocked credential: 503 would loop Meta redeliveries forever
      // and a retry can never succeed. Accept the event as final (audit code
      // recorded), alert the operator, and ask the sender to resend once the
      // token is reissued. Delivery itself is best-effort (rule 5).
      logger.error('[WhatsApp] MEDIA_CREDENTIAL_BLOCKED: media unrecoverable with current token; operator must reissue WHATSAPP_ACCESS_TOKEN', {
        shortId, error: code, detail: detail.slice(0, 200),
      });
      try {
        await markIntakeAccepted(db, providerMessageId, claimToken, detail.slice(0, 120));
      } catch (claimError) {
        logger.warn('[WhatsApp] permanent media failure could not settle stale claim', {
          shortId, error: claimError instanceof Error ? claimError.message : 'unknown',
        });
        return 'retryable';
      }
      if (message.from) {
        await sendText(message.from, 'I could not retrieve that file right now (a system credential expired). Please resend the invoice in a few minutes.').catch(() => undefined);
      }
      return 'processed';
    }
    if (isEncryptionConfigFailure(e)) {
      // Encryption keys are a deploy-time contract: a missing/malformed key is
      // an operator-configuration error that NO retry can fix. Returning 503
      // would make Meta redeliver the message in a loop, so accept this
      // delivery as final with a stable audit code (webhook_events stays the
      // idempotency ledger: a genuine later redelivery of an ACCEPTED event is
      // skipped, so the sender below is notified at most once per message),
      // alert the operator, and tell the sender the truth: nothing was saved.
      logger.error('[WhatsApp] ENCRYPTION_CONFIG: file encryption is not configured; nothing was stored â€” operator must set ENCRYPTION_MASTER_KEY + STORAGE_PATH_KEY and restart', {
        shortId, code: 'ENCRYPTION_CONFIG',
      });
      try {
        await markIntakeAccepted(db, providerMessageId, claimToken, 'ENCRYPTION_CONFIG');
      } catch (claimError) {
        logger.warn('[WhatsApp] encryption config failure could not settle stale claim', {
          shortId, error: claimError instanceof Error ? claimError.message : 'unknown',
        });
        return 'retryable';
      }
      if (message.from) {
        await sendText(message.from, 'I could not file that invoice â€” a required system setting is missing on our side, so nothing was saved. The account owner needs to finish setup; please resend afterwards.').catch(() => undefined);
      }
      return 'processed';
    }
    if (isStorageConfigFailure(e)) {
      // The private bucket is not yet configured to accept the encrypted
      // container's transport type (application/octet-stream). This is an
      // operator/schema-delivery gap that no retry can fix: returning 503 would
      // loop Meta redeliveries and re-download the same media forever. Accept as
      // final with a stable audit code, alert the operator, and tell the sender
      // the truth â€” nothing was saved.
      logger.error('[WhatsApp] STORAGE_CONFIG: private storage rejected the encrypted upload; operator must apply the invoices bucket migration', {
        shortId, code: 'STORAGE_CONFIG',
      });
      try {
        await markIntakeAccepted(db, providerMessageId, claimToken, 'STORAGE_CONFIG');
      } catch (claimError) {
        logger.warn('[WhatsApp] storage config failure could not settle stale claim', {
          shortId, error: claimError instanceof Error ? claimError.message : 'unknown',
        });
        return 'retryable';
      }
      if (message.from) {
        await sendText(message.from, 'I could not file that invoice â€” our private storage is mid-setup, so nothing was saved. The account owner needs to finish a configuration step; please resend afterwards.').catch(() => undefined);
      }
      return 'processed';
    }
    await markIntakeFailed(db, providerMessageId, claimToken, code).catch(() => undefined);
    logger.error('[WhatsApp] processing failed, retryable', { shortId, error: code, detail: detail.slice(0, 200) });
    return 'retryable';
  } finally {
    clearInterval(heartbeat);
  }
  logger.info('[WhatsApp] accepted for queue', { shortId });
  return 'processed';
}

/**
 * Best-effort resend of a reply that failed delivery earlier (rule 5 decouples
 * outbound from inbound). Fires at the top of a NEW inbound message and only
 * sends â€” it never re-runs intake or onboarding side effects, so a completed
 * business operation cannot be duplicated. Cleared on success; left stored on
 * any non-ok so the next message retries. Never throws.
 */
async function resendPendingReply(db: Db, hmac: string, to: string): Promise<void> {
  try {
    const { data } = await db.from('whatsapp_sessions').select('pending_reply_body')
      .eq('sender_hmac', hmac).maybeSingle();
    const rawBody = (data?.pending_reply_body as string | null) ?? null;
    if (!rawBody) return;
    // A register handoff carries one or more leading [[deliver:<uuid>]] tokens â€”
    // one per ready workbook that finished since the last inbound (a multi-month
    // burst can queue several). Deliver EACH decrypted file into the chat; the web
    // link is unreachable for a WhatsApp-only owner (no login exists). Strip every
    // internal token so any text fallback shows only human copy, and fall back to
    // the text link only if document delivery is not possible.
    const workbookIds = [...rawBody.matchAll(/\[\[deliver:([0-9a-f-]{36})\]\]/gi)].map((m) => m[1]);
    const body = rawBody.replace(/\s*\[\[deliver:[0-9a-f-]{36}\]\]/gi, '').trim();
    if (workbookIds.length > 0) {
      const outstanding: string[] = [];
      let deliveredCount = 0;
      for (const workbookId of workbookIds) {
        const r = await deliverWorkbookToWhatsApp(db, { workbookId, to });
        if (r.delivered) { deliveredCount += 1; continue }
        // NOT yet in the chat but could succeed later: keep THIS handoff so the
        // next inbound retries the document. Falling through here would send the
        // reassuring "I'll send the Excel" text and, on a successful text send,
        // clear the slot â€” stranding a 'ready' workbook forever with the file
        // never arriving. A 'skipped' (gone/expired/not-ready) workbook has nothing
        // to retry, so it is simply dropped from the set below.
        if (r.outcome === 'transient' || r.outcome === 'blocked') outstanding.push(workbookId);
      }
      if (outstanding.length > 0) {
        // Re-arm the slot with ONLY the still-outstanding tokens and stay silent;
        // the current message is still handled normally, and these retry next time.
        const kept = outstanding.map((id) => `[[deliver:${id}]]`).join('\n');
        await updateSessionDeliveryState(db, hmac, { pending_reply_body: `${kept}\n${body}`.trim(), last_reply_error_code: null });
        logger.warn('[WhatsApp] some deferred registers still pending; retrying next inbound', { outstanding: outstanding.length });
        return;
      }
      // Every register landed or is gone. When at least one file arrived, the
      // document IS the reply â€” clear the handoff and say nothing further.
      await updateSessionDeliveryState(db, hmac, { pending_reply_body: null, last_reply_error_code: null });
      if (deliveredCount > 0) {
        logger.info('[WhatsApp] deferred register(s) delivered as document', { count: deliveredCount });
        return;
      }
      // All were 'skipped' (nothing landed): fall through to the one-time text.
    }
    const o = await sendText(to, body, workbookIds[0] ? { workbookId: workbookIds[0] } : undefined);
    if (o === 'ok') {
      await updateSessionDeliveryState(db, hmac, { pending_reply_body: null, last_reply_error_code: null });
      logger.info('[WhatsApp] deferred reply resent');
    }
  } catch { /* delivery is best-effort; ignore and try again next message */ }
}

/**
 * Onboarding + upload decision engine for one message. Throws only when the
 * work is NOT durably recorded (caller marks intake failed -> Meta retries).
 * Media lifecycle: validate type -> store privately + SHA-256 -> invoice
 * (received) + job (queued) â€” or a pending_uploads hold when workspace/client
 * is unresolved ("receive first, ask later"). Held paths are deterministic
 * per provider media id, so webhook redelivery never duplicates a hold.
 */
async function processMessage(
  db: Db, message: IncomingMessage, providerMessageId: string, shortId: string,
): Promise<void> {
  const replyId = message.interactive?.button_reply?.id || message.interactive?.list_reply?.id || null;
  const textBody = message.text?.body ?? null;
  // Decoupled outbound retry: if a previous reply failed delivery, resend it
  // now (send only â€” no business operation re-runs), before normal processing.
  if (message.from) {
    try { await resendPendingReply(db, senderHmac(message.from), message.from); } catch { /* best effort */ }
    // Durable outbox flush: send every workbook this owner is currently owed via
    // the atomic-claim state machine (replaces the racy single-slot handoff). Send
    // only â€” never re-runs intake/onboarding. Best-effort; a failure leaves the
    // rows pending for the next inbound or the worker recovery sweep.
    try { await flushPendingDeliveries(db, senderHmac(message.from), message.from); } catch { /* best effort */ }
  }
  let onboardingWorkspaceId: string | null = null;
  let onboardingClientId: string | null = null;
  if ((message.type === 'text' || message.type === 'interactive') && message.from) {
    const hmac = senderHmac(message.from);
    const label = senderLabel(hmac);
    const stored = await loadSession(db, hmac);
    let snapshot: SessionSnapshot = isExpired(stored.updated_at)
      ? expiredSnapshot(stored, hmac)
      : { senderHmac: hmac, state: stored.state as ConversationState,
          workspaceId: stored.workspace_id, clientId: stored.client_id,
          batchId: stored.batch_id, pendingClientId: stored.pending_client_id ?? null,
          reviewEditInvoiceId: stored.review_edit_invoice_id ?? null,
          lastProviderMessageId: stored.last_provider_message_id,
          expired: false };
    // Session row lost its linkage but the identity is onboarded: backfill so
    // returning users never repeat setup.
    if (!snapshot.workspaceId) {
      const linked = await db.from('whatsapp_identities').select('workspace_id')
        .eq('sender_hmac', hmac).maybeSingle();
      const linkedWs = (linked.data?.workspace_id as string | null) ?? null;
      if (linkedWs) snapshot = { ...snapshot, workspaceId: linkedWs };
    }
    const result = transition({ session: snapshot, replyId, text: textBody, providerMessageId });
    logger.info('[WhatsApp] onboarding transition', { sender: label, note: result.note });
    const out = await handleOnboardingSideEffects(db, hmac, label, message.from, providerMessageId, snapshot, result);
    onboardingWorkspaceId = out.workspaceId;
    onboardingClientId = out.clientId;
  }
  const media = message.type === 'image' ? message.image
    : message.type === 'document' ? message.document : undefined;
  const mediaId = media?.id;
  if (!mediaId) {
    if (message.type === 'image' || message.type === 'document') {
      logger.info('[WhatsApp] media message without media id', { shortId });
      throw new Error('MEDIA_MISSING_ID: retryable media without provider id');
    }
    return;
  }
  if (!message.from) throw new Error('MEDIA_MISSING_SENDER: media without from address');
  // Encryption keys are required before ANY file handling, so validate them
  // FIRST: a misconfigured deploy must not pay for a media download it can
  // never store (fail-closed: plaintext is never written, retries never help).
  const { masterKeyHex, pathKeyHex } = resolveFileKeyConfig();
  logger.info('[WhatsApp] media intake start', { shortId });
  const mHmac = senderHmac(message.from);
  const mSession = await loadSession(db, mHmac);

  // Download, then validate the ACTUAL bytes. WhatsApp's declared mime /
  // filename / extension are hints only â€” the magic-byte sniff is authoritative,
  // so a mislabeled, unsupported or oversized file is rejected here: nothing is
  // stored, no DB error is raised, and the intake settles accepted (a permanent
  // format/size failure must NEVER trigger a Meta retry / re-download loop).
  let downloaded: DownloadedMedia;
  try {
    downloaded = await downloadWhatsAppMedia({
      mediaId, accessToken: process.env.WHATSAPP_ACCESS_TOKEN || '', declaredMime: media?.mime_type,
    });
  } catch (e) {
    if (isPermanentMediaFormatFailure(e)) {
      logger.info('[WhatsApp] media rejected by declared size', { shortId, reason: (e as Error).message });
      await sendText(message.from, (e as { userReply: string }).userReply);
      return;
    }
    throw e;
  }
  let validated: ValidatedMedia;
  try {
    validated = validateInvoiceMedia({
      buffer: downloaded.buffer, declaredMime: downloaded.contentType, declaredFilename: media?.filename ?? null,
    });
  } catch (e) {
    if (isPermanentMediaFormatFailure(e)) {
      logger.info('[WhatsApp] unsupported or oversized media rejected', { shortId, reason: (e as Error).message });
      await sendText(message.from, (e as { userReply: string }).userReply);
      return;
    }
    throw e;
  }
  // Persist the VALIDATED original MIME + extension (never the provider hint) so
  // extraction routes on the true format. The stored blob is always the opaque
  // octet-stream container; the original bytes stay intact inside it.
  downloaded.contentType = validated.mime;
  downloaded.extension = validated.extension;
  const sha256Hex = createHash('sha256').update(downloaded.buffer).digest('hex');
  const workspaceId = onboardingWorkspaceId ?? mSession.workspace_id;
  const clientId = onboardingClientId ?? mSession.client_id;

  // Envelope-encrypt the ORIGINAL bytes before they ever touch storage (keys
  // were validated above, before the download). The SHA-256 above stays over
  // plaintext for dedup + integrity; the stored blob is the self-describing
  // container.
  const invoiceBucket = bucketFor('invoices');

  // Deterministic private hold path: redelivery of the same provider media id
  // resolves to the same opaque object + upserts the same row â€” never a
  // duplicate hold, and never collides with an unrelated file.
  const holdMedia = async (): Promise<string | null> => {
    const pendingPath = deriveObjectKey('pending', mediaId, pathKeyHex);
    const { WHATSAPP_MAX_INVOICES_PER_DAY } = getEnv();
    const reservation = await reserveDailyIntake(db, {
      providerMessageId,
      workspaceId,
      senderHmac: mHmac,
      limit: WHATSAPP_MAX_INVOICES_PER_DAY,
    });
    if (!reservation.allowed) {
      logger.info('[WhatsApp] held media blocked: daily intake limit reached', {
        shortId, limit: WHATSAPP_MAX_INVOICES_PER_DAY,
      });
      await sendText(message.from!, `You've reached today's limit of ${WHATSAPP_MAX_INVOICES_PER_DAY} invoices. Please try again tomorrow.`);
      return null;
    }
    await beginStorageIntake(db, { providerMessageId, objectPath: pendingPath, kind: 'pending' });
    const enc = encryptFile(downloaded.buffer, masterKeyHex, { context: pendingPath });
    const { error: pendingUploadError } = await db.storage.from(invoiceBucket)
      .upload(pendingPath, enc.container, {
        contentType: 'application/octet-stream', upsert: true,
        metadata: { original_mime: validated.mime, original_extension: validated.extension },
      });
    if (pendingUploadError) throw new Error(`STORAGE_UPLOAD_ERROR: ${pendingUploadError.message}`);
    await markStorageIntake(db, pendingPath, 'uploaded');
    await recordPendingUpload(db, {
      senderHmac: mHmac, objectPath: pendingPath, originalFileName: media?.filename ?? null,
      contentType: downloaded.contentType, fileSha256Hex: sha256Hex,
      sizeBytes: enc.sizeBytes, providerMediaId: mediaId, providerMessageId, keyVersion: enc.keyVersion,
    });
    await commitDailyIntake(db, providerMessageId, reservation.scopeKey);
    await markStorageIntake(db, pendingPath, 'linked');
    return pendingPath;
  };

  if (!workspaceId) {
    // Account type unknown: preserve the file, then continue/ask setup.
    if (!(await holdMedia())) return;
    if (mSession.state === 'awaiting_account_type') {
      await sendText(message.from, 'I saved that invoice safely. Finish setup above â€” pick Individual or Organization â€” and I will file it automatically.');
    } else {
      await sendText(message.from, 'I saved that invoice safely. Before I file it: are you onboarding as an Individual or an Organization / CA firm?');
      await sendAccountTypeButtons(message.from);
    }
    return;
  }

  const accountType = await loadAccountType(db, mHmac);
  if (accountType === 'organization' && !clientId) {
    // Client unresolved: hold and ask. NEVER guess a client.
    if (!(await holdMedia())) return;
    await saveSession(db, mHmac, mSession.last_provider_message_id, {
      state: 'awaiting_client_select', workspace_id: workspaceId, client_id: clientId,
      batch_id: mSession.batch_id, pending_client_id: mSession.pending_client_id,
      last_provider_message_id: providerMessageId,
      expires_at: new Date(Date.now() + 24 * 60 * 60 * 1000).toISOString(),
    }).catch(() => undefined); // best effort: the file is durably held either way
    await sendText(message.from, 'I saved that invoice â€” but I must not guess which client it belongs to. Choose a client (or add one) below.');
    await sendClientList(message.from, await listClients(db, workspaceId));
    return;
  }

  // Assigned path is deterministic too (per provider media id): combined with
  // the invoices.file_object_path unique constraint, a redelivery can never
  // store or queue the same file twice, whatever the retry timing.
  //
  // Exact-file deduplication is status-aware. Keep the duplicate guard, but do
  // not tell customers that nothing was lost when the original row is failed,
  // awaiting review, or not yet included in a register.
  const duplicateFileInvoice = await findLiveInvoiceByFileHash(db, {
    workspaceId, clientId, fileSha256Hex: sha256Hex,
  });
  if (duplicateFileInvoice) {
    logger.info('[WhatsApp] exact-file duplicate skipped at intake', {
      shortId, invoiceId: duplicateFileInvoice.id.slice(0, 8),
      status: duplicateFileInvoice.status, hasBatch: !!duplicateFileInvoice.batchId,
    });
    const duplicateReply = duplicateFileInvoice.status === 'needs_review'
      ? 'I already received this exact file, so I did not create a duplicate. It still needs your review â€” send Review to correct or approve it.'
      : duplicateFileInvoice.status === 'failed'
        ? 'I already received this exact file, but its previous processing attempt failed. I have not created a duplicate. Please contact support with this message so the existing record can be recovered safely.'
        : duplicateFileInvoice.status === 'received' || duplicateFileInvoice.status === 'queued' || duplicateFileInvoice.status === 'processing'
          ? 'I already received this exact file and it is still in the processing queue. I did not create a duplicate. Send Done later to check the register.'
          : duplicateFileInvoice.status === 'completed'
            ? 'I already received and processed this exact file, so I did not add it again. If it is missing from your Excel register, send Done to refresh the register; otherwise you can send a different invoice.'
            : 'I already received this exact file, so I did not add it again. Send Done to check your register.';
    await sendText(message.from, duplicateReply);
    return;
  }
  const objectPath = deriveObjectKey('invoices', mediaId, pathKeyHex);
  const { WHATSAPP_MAX_INVOICES_PER_DAY } = getEnv();
  const reservation = await reserveDailyIntake(db, {
    providerMessageId, workspaceId, senderHmac: mHmac,
    limit: WHATSAPP_MAX_INVOICES_PER_DAY,
  });
  if (!reservation.allowed) {
    logger.info('[WhatsApp] media intake blocked: daily invoice limit reached', { shortId, limit: WHATSAPP_MAX_INVOICES_PER_DAY });
    await sendText(message.from, `You've reached today's limit of ${WHATSAPP_MAX_INVOICES_PER_DAY} invoices. Please try again tomorrow â€” anything you already sent is safe and still processing.`);
    return;
  }
  await beginStorageIntake(db, { providerMessageId, objectPath, kind: 'invoice' });
  const enc = encryptFile(downloaded.buffer, masterKeyHex, { context: objectPath });
  const { error: uploadError } = await db.storage.from(invoiceBucket)
    .upload(objectPath, enc.container, {
      contentType: 'application/octet-stream', upsert: true,
      metadata: { original_mime: validated.mime, original_extension: validated.extension },
    });
  if (uploadError) throw new Error(`STORAGE_UPLOAD_ERROR: ${uploadError.message}`);
    await markStorageIntake(db, objectPath, 'uploaded');
    const recorded = await recordMediaIntake(db, {
    workspaceId, clientId, objectPath,
    originalFileName: media?.filename ?? null, contentType: downloaded.contentType,
    fileSha256Hex: sha256Hex, keyVersion: enc.keyVersion, sizeBytes: enc.sizeBytes,
  });
  await commitDailyIntake(db, providerMessageId, reservation.scopeKey);
  await markStorageIntake(db, objectPath, 'linked');
  const waiting = await countWaitingUploads(db, mHmac).catch(() => 0);
  const deduped = recorded.invoicePreexisting || recorded.jobPreexisting;
  logger.info('[WhatsApp] media intake recorded', {
    shortId, size: enc.sizeBytes,
    invoice: recorded.invoiceId.slice(0, 8), job: recorded.jobId.slice(0, 8),
    deduped,
  });
  // Acknowledge a freshly-filed invoice so a clean upload is never silent
  // (rule 5: send is best-effort â€” a delivery failure never rewinds the
  // accepted intake). A webhook redelivery dedups to the same invoice/job and
  // stays quiet, so Meta retries never double-ping the sender. Â§user: a burst of
  // files uploaded together arrives as separate near-simultaneous webhooks, so a
  // plain session read would let EVERY worker ack â€” the claim is an atomic
  // conditional update (sessions.claimUploadAck) so exactly ONE webhook in the
  // burst sends the confirmation and the rest stay silent. The slot re-arms after
  // the coalesce window so a genuinely later batch still gets one ack.
  if (!deduped) {
    const wonAck = await claimUploadAck(db, mHmac, UPLOAD_ACK_COALESCE_MS);
    if (wonAck) {
      const held = waiting > 0 && clientId
        ? ` Note: ${waiting} earlier file(s) are still held for a client â€” send Change client to assign them.`
        : '';
      await sendText(message.from, `Invoice received and queued for processing.${held} Send more anytime, or type Done / Menu.`);
    }
  }
}

/**
 * A storage upload rejected specifically because the object's transport MIME is
 * not permitted on the bucket (the encrypted container is uploaded as
 * application/octet-stream). That is a bucket-configuration gap the operator
 * fixes by applying the invoices bucket migration â€” retrying the same upload
 * can never succeed, so it must NOT loop Meta redeliveries. Deliberately narrow:
 * only a MIME/"not allowed"-style rejection matches, so transient network and
 * permission errors stay retryable.
 */
function isStorageConfigFailure(error: unknown): boolean {
  const message = error instanceof Error ? error.message : String(error);
  if (!/STORAGE_UPLOAD_ERROR/.test(message)) return false;
  return /(mime|content[ -]?type|file type)/i.test(message)
    && /(not allowed|not supported|unsupported)/i.test(message);
}

export interface IntakeRow {
  id: string
  provider_message_id: string
  status: IntakeStatus
  attempts: number
  processed_at: string | null
  claim_expires_at: string | null
}

const INTAKE_CLAIM_LEASE_SECONDS = 180

type IntakeClaimRpcRow = {
  claimed: boolean
  created: boolean
  event_id: string
  provider_message_id: string
  status: IntakeStatus
  attempts: number
  processed_at: string | null
  claim_expires_at: string | null
}

type Db = SupabaseClient

/**
 * Atomic PostgreSQL claim. Only the winning invocation receives its UUID token.
 * Expired claims can be reclaimed; all later writes are fenced by that token.
 */
export async function claimIntakeRow(
  db: Db,
  providerMessageId: string,
): Promise<{ row: IntakeRow; created: boolean; claimed: boolean; claimToken: string | null }> {
  const claimToken = randomUUID()
  const { data, error } = await db.rpc('claim_whatsapp_webhook_event', {
    p_provider_message_id: providerMessageId,
    p_claim_token: claimToken,
    p_lease_seconds: INTAKE_CLAIM_LEASE_SECONDS,
  }) as unknown as { data: IntakeClaimRpcRow[] | null; error: { message: string } | null }
  if (error) throw new Error(`INTAKE_CLAIM_ERROR: ${error.message}`)
  const result = Array.isArray(data) ? data[0] : null
  if (!result?.event_id || typeof result.claimed !== 'boolean') {
    throw new Error('INTAKE_CLAIM_ERROR: database returned an invalid claim result')
  }
  return {
    row: {
      id: result.event_id,
      provider_message_id: result.provider_message_id,
      status: result.status,
      attempts: result.attempts,
      processed_at: result.processed_at,
      claim_expires_at: result.claim_expires_at,
    },
    created: result.created,
    claimed: result.claimed,
    claimToken: result.claimed ? claimToken : null,
  }
}

/** True only when a previous delivery durably accepted this message. */
export function isDurablyAccepted(row: IntakeRow): boolean {
  return row.status === 'accepted' && row.processed_at !== null
}

/**
 * Renew only an unexpired claim. Expired owners cannot resurrect their lease.
 */
export async function renewIntakeClaim(db: Db, providerMessageId: string, claimToken: string): Promise<boolean> {
  const { data, error } = await db.rpc('renew_whatsapp_webhook_event_claim', {
    p_provider_message_id: providerMessageId,
    p_claim_token: claimToken,
    p_lease_seconds: INTAKE_CLAIM_LEASE_SECONDS,
  }) as unknown as { data: boolean | null; error: { message: string } | null }
  if (error) throw new Error(`INTAKE_HEARTBEAT_ERROR: ${error.message}`)
  return data === true
}

/** Complete only while this invocation owns the current, unexpired lease. */
export async function markIntakeAccepted(
  db: Db, providerMessageId: string, claimToken: string, errorCode?: string,

  if (data !== true) throw new Error('INTAKE_CLAIM_LOST: event claim expired or belongs to another invocation')
}

/**
 * Media lifecycle: invoice (received) + processing job (queued) in one
 * recoverable unit. STRICTLY IDEMPOTENT (Phase 3): a webhook redelivery that
 * arrives between the insert and the intake accept must not create a second
 * invoice row or a second job. Uniqueness is enforced in the database
 * (invoices.file_object_path, one job per invoice_id+job_type); on a 23505 we
 * read back the existing rows and reuse them.
 *
 * Never upsert here: overwriting an invoice that already progressed would
 * rewind its status, which the queue interprets as work to redo.
 */
export async function recordMediaIntake(
  db: Db,
  args: {
    workspaceId: string
    clientId: string | null
    objectPath: string
    originalFileName: string | null
    contentType: string
    fileSha256Hex?: string | null
    keyVersion?: string | null
    sizeBytes?: number | null
  },
): Promise<{ invoiceId: string; jobId: string; invoicePreexisting: boolean; jobPreexisting: boolean }> {
  const invoicePayload = {
    workspace_id: args.workspaceId,
    client_id: args.clientId,
    file_object_path: args.objectPath,
    original_file_name: args.originalFileName,
    content_type: args.contentType,
    file_sha256: args.fileSha256Hex ? `\\x${args.fileSha256Hex}` : null,
    file_encryption_key_version: args.keyVersion ?? null,
    size_bytes: args.sizeBytes ?? null,
    status: 'received',
  }
  const invoice = (await db.from('invoices').insert(invoicePayload).select('id').single() as unknown as {
    data: { id: string } | null
    error: { code?: string; message: string } | null
  })
  let invoiceId: string
  let invoicePreexisting = false
  if (!invoice.error && invoice.data) {
    invoiceId = invoice.data.id
  } else if (invoice.error?.code === '23505') {
    const existing = (await db.from('invoices').select('id')
      .eq('file_object_path', args.objectPath).limit(1).maybeSingle() as unknown as {
      data: { id: string } | null; error: { message: string } | null
    })
    if (existing.error || !existing.data) {
      throw new Error(`INVOICE_READ_ERROR: ${existing.error?.message ?? 'missing row after conflict'}`)
    }
    invoiceId = existing.data.id
    invoicePreexisting = true
  } else {
    throw new Error(`INVOICE_RECORD_ERROR: ${invoice.error?.message ?? 'missing id'}`)
  }

  const jobPayload = {
    workspace_id: args.workspaceId,
    invoice_id: invoiceId,
    job_type: 'invoice_extraction',
    status: 'queued',
  }
  const job = (await db.from('processing_jobs').insert(jobPayload).select('id').single() as unknown as {
    data: { id: string } | null
    error: { code?: string; message: string } | null
  })
  let jobId: string
  let jobPreexisting = false
  if (!job.error && job.data) {
    jobId = job.data.id
  } else if (job.error?.code === '23505') {
    const existing = (await db.from('processing_jobs').select('id')
      .eq('invoice_id', invoiceId).eq('job_type', 'invoice_extraction').limit(1).maybeSingle() as unknown as {
      data: { id: string } | null; error: { message: string } | null
    })
    if (existing.error || !existing.data) {
      throw new Error(`JOB_READ_ERROR: ${existing.error?.message ?? 'missing row after conflict'}`)
    }
    jobId = existing.data.id
    jobPreexisting = true
  } else {
    throw new Error(`JOB_RECORD_ERROR: ${job.error?.message ?? 'missing id'}`)
  }
  // Best-effort, NON-BLOCKING wake so a freshly-filed invoice is picked up in
  // seconds instead of waiting for the next cron tick (Â§P2-9). Only fired when a
  // NEW job was created (a webhook redelivery dedups and stays quiet). It never
  // throws and the cron remains the durable backstop.
  if (!jobPreexisting) wakeWorker()
  return { invoiceId, jobId, invoicePreexisting, jobPreexisting }
}

/**
 * Layer 1 of duplicate prevention: is the EXACT SAME FILE (identical SHA-256)
 * already filed as a LIVE invoice for this (workspace, client)? A byte-for-byte
 * resend under a NEW provider media id slips past webhook idempotency (which is
 * keyed on object path) and would otherwise become a second register row, so we
 * check the stored `invoices.file_sha256` before uploading/queueing and skip it.
 *
 * Deliberately FAIL-OPEN: a lookup error (or the column being unreadable) must
 * NEVER block a legitimate invoice â€” we return null and the normal intake path
 * proceeds, deduping downstream as before. Only a genuinely matched live row
 * (status <> 'deleted') short-circuits. Tenant-scoped: an organization matches
 * within its own client; an individual (null client_id) matches null-client rows.
 *
 * Returns the matching invoice's id, status and batch association, or null
 * when there is no live duplicate. This lets callers distinguish an already
 * exported invoice from one that is pending, failed, or still needs review.
 */
export interface LiveFileDuplicate {
  id: string
  status: 'received' | 'queued' | 'processing' | 'needs_review' | 'completed' | 'failed' | 'deleted'
  batchId: string | null
}

/**
 * Layer 1 of duplicate prevention: find the exact same live file in this
 * workspace/client. Return its processing state as well as its id so the
 * WhatsApp acknowledgement never claims an unexported invoice was "not lost".
 * Lookup errors remain fail-open: a real invoice must never be blocked.
 */
export async function findLiveInvoiceByFileHash(
  db: Db,
  args: { workspaceId: string; clientId: string | null; fileSha256Hex: string },
): Promise<LiveFileDuplicate | null> {
  if (!args.fileSha256Hex) return null
  try {
    let query = db
      .from('invoices')
      .select('id,status,batch_id')
      .eq('workspace_id', args.workspaceId)
      .eq('file_sha256', `\\x${args.fileSha256Hex}`)
      .neq('status', 'deleted')
      .limit(1)
    query = args.clientId
      ? (query.eq('client_id', args.clientId) as typeof query)
      : (query.is('client_id', null) as typeof query)
    const { data, error } = await (query as unknown as {
      maybeSingle: () => Promise<{
        data: { id: string; status: LiveFileDuplicate['status']; batch_id: string | null } | null
        error: { message: string } | null
      }>
    }).maybeSingle()
    if (error) {
      logger.warn('[Intake] exact-file duplicate lookup failed; proceeding (fail-open)', { code: error.message.slice(0, 80) })
      return null
    }
    return data ? { id: data.id, status: data.status, batchId: data.batch_id } : null
  } catch (e) {
    logger.warn('[Intake] exact-file duplicate lookup threw; proceeding (fail-open)', { error: e instanceof Error ? e.message : String(e) })
    return null
  }
}

// ---- Pending uploads ("receive first, ask later") -------------------------
// Media that arrives before the sender has a workspace â€” or before an
// organization client is explicitly confirmed â€” is stored privately and held
// here. Assignment binds workspace/client at confirmation time and creates the
// invoice + job rows. Nothing is ever silently reassigned (client safety rule).

/**
 * Hold a securely stored upload until workspace/client assignment resolves.
 * Upserts on object_path (deterministic per provider media id): a webhook
 * redelivery refreshes the hold instead of erroring, and NEVER resets the
 * status of an already-assigned file (status is not in the payload).
 */
export async function recordPendingUpload(
  db: Db,
  args: {
    senderHmac: string
    objectPath: string
    originalFileName: string | null
    contentType: string
    fileSha256Hex: string | null
    sizeBytes: number
    providerMediaId: string | null
    providerMessageId: string
    keyVersion?: string | null
  },
): Promise<string> {
  const row = (await db.from('pending_uploads').upsert({
    sender_hmac: args.senderHmac,
    object_path: args.objectPath,
    original_file_name: args.originalFileName,
    content_type: args.contentType,
    file_sha256: args.fileSha256Hex ? `\\x${args.fileSha256Hex}` : null,
    size_bytes: args.sizeBytes,
    file_encryption_key_version: args.keyVersion ?? null,
    provider_media_id: args.providerMediaId,
    provider_message_id: args.providerMessageId,
  }, { onConflict: 'object_path' }).select('id').single() as unknown as { data: { id: string } | null; error: { message: string } | null })
  if (row.error || !row.data) throw new Error(`PENDING_UPLOAD_RECORD_ERROR: ${row.error?.message ?? 'missing id'}`)
  return row.data.id
}

/** Number of held files still waiting for assignment (drives confirm prompts). */
export async function countWaitingUploads(db: Db, senderHmac: string): Promise<number> {
  const { count, error } = await db
    .from('pending_uploads')
    .select('id', { count: 'exact', head: true })
    .eq('sender_hmac', senderHmac)
    .eq('status', 'waiting_assignment')
  if (error) throw new Error(`PENDING_UPLOAD_COUNT_ERROR: ${error.message}`)
  return count ?? 0
}

/**
 * Assign every held file for this sender to (workspace, client) â€” called only
 * after explicit resolution (individual selection or confirmed client).
 * Each row: invoice + job created first, then the hold is marked assigned.
 */
export async function assignPendingUploads(
  db: Db,
  senderHmac: string,
  args: { workspaceId: string; clientId: string | null; limit: number },
): Promise<number> {
  const { data: rows, error } = await db
    .from('pending_uploads')
    .select('id,object_path,original_file_name,content_type,file_sha256,size_bytes,file_encryption_key_version,provider_message_id')
    .eq('sender_hmac', senderHmac)
    .eq('status', 'waiting_assignment')
    .order('created_at')
  if (error) throw new Error(`PENDING_UPLOAD_LIST_ERROR: ${error.message}`)
  let assigned = 0
  for (const row of (rows ?? []) as Array<{
    id: string; object_path: string; original_file_name: string | null;
    content_type: string; file_sha256: string | null;
    size_bytes: number | null; file_encryption_key_version: string | null;
    provider_message_id: string | null
  }>) {
    const providerMessageId = row.provider_message_id ?? `legacy-pending:${row.id}`
    const reservation = await reserveDailyIntake(db, {
      providerMessageId, workspaceId: args.workspaceId, senderHmac, limit: args.limit,
    })
    if (!reservation.allowed) {
      logger.warn('[Intake] held upload assignment deferred by daily quota', {
        pendingId: row.id.slice(0, 8), workspaceId: args.workspaceId.slice(0, 8),
      })
      continue
    }
    const shaHex = typeof row.file_sha256 === 'string'
      ? row.file_sha256.replace(/^\\x/i, '')
      : null
    const recorded = await recordMediaIntake(db, {
      workspaceId: args.workspaceId,
      clientId: args.clientId,
      objectPath: row.object_path,
      originalFileName: row.original_file_name,
      contentType: row.content_type,
      fileSha256Hex: shaHex,
      keyVersion: row.file_encryption_key_version,
      sizeBytes: row.size_bytes,
    })
    await commitDailyIntake(db, providerMessageId, reservation.scopeKey)
    await markStorageIntake(db, row.object_path, 'linked')
    if (reservation.scopeKey !== `sender:${senderHmac}`) {
      await releaseDailyIntakeScope(db, providerMessageId, `sender:${senderHmac}`)
    }
    const { error: updateError } = await db
      .from('pending_uploads')
      .update({ status: 'assigned', assigned_invoice_id: recorded.invoiceId, updated_at: new Date().toISOString() })
      .eq('id', row.id)
      .eq('status', 'waiting_assignment')
    if (updateError) throw new Error(`PENDING_UPLOAD_ASSIGN_ERROR: ${updateError.message}`)
    assigned += 1
  }
  return assigned
}

export async function extractInvoice(ctx: JobContext): Promise<void> {
  const { db, job, assertLease } = ctx
  if (!job.invoice_id) throw new PermanentJobError('EXTRACT_MISSING_TARGET: invoice job without invoice_id')
  const invoiceId = job.invoice_id
  const workspaceId = job.workspace_id

  // Idempotent short-circuit: an already-approved record means the work is done;
  // re-affirm invoice completion + batch scheduling and finish without re-calling
  // the provider (never burn Gemini quota on a retry/duplicate claim).
  const { data: existing } = await db
    .from('extracted_records')
    .select('review_status')
    .eq('workspace_id', workspaceId).eq('invoice_id', invoiceId).maybeSingle()
  if (existing && existing.review_status === 'approved') {
    await resolveInvoiceBatch(db, { workspaceId, invoiceId, jobId: job.id, assertLease })
    logger.info('[Extract] already approved; skipping provider call', { invoiceId, jobId: job.id })
    return
  }

  // Load the invoice scoped to this job's workspace (tenant isolation).
  const { data: invoiceData, error: invErr } = await db
    .from('invoices')
    .select('id,workspace_id,client_id,batch_id,file_object_path,content_type,file_encryption_key_version,status,received_at')
    .eq('workspace_id', workspaceId).eq('id', invoiceId).maybeSingle()
  if (invErr) throw new Error(`INVOICE_READ_ERROR: ${invErr.message}`)
  const invoice = invoiceData as InvoiceRow | undefined
  if (!invoice) throw new PermanentJobError('INVOICE_NOT_FOUND')
  if (!invoice.file_object_path || !invoice.content_type) {
    throw new PermanentJobError('INVOICE_INCOMPLETE: no stored object/content type')
  }

  // The client's OWN (buyer) GSTIN, when set, is the fact that lets extraction
  // reject a buyer GSTIN mislabelled as the supplier. Read it tenant-scoped. We
  // select('*') and fail SOFT: if the additive own_gstin migration has not been
  // applied yet the key is simply absent and we continue with null â€” this
  // enhancement must never be able to break extraction the way an unknown-column
  // reference would (see the PGRST204 session-save regression).
  let buyerGstin: string | null = null
  if (invoice.client_id) {
    const { data: clientRow, error: clientErr } = await db
      .from('clients')
      .select('*')
      .eq('workspace_id', workspaceId)
      .eq('id', invoice.client_id)
      .maybeSingle()
    if (clientErr) {
      logger.warn('[Extract] client own_gstin read failed; continuing without buyer cross-check', {
        invoiceId, kind: clientErr.code ?? 'unknown',
      })
    } else {
      buyerGstin = (clientRow as { own_gstin?: string | null } | null)?.own_gstin ?? null
    }
  }

  const { masterKeyHex } = resolveFileKeyConfig()

  // Â§P2-12 per-stage timing. Started before the download so 'decrypt' also accounts
  // for the storage fetch. Stage names + integer ms only â€” never invoice content.
  const timing = new StageTimer()

  // Download the encrypted container and decrypt it IN MEMORY only.
  const { data: blob, error: dlErr } = await db.storage.from(bucketFor('invoices'))
    .download(invoice.file_object_path)
  if (dlErr || !blob) throw new Error(`MEDIA_DOWNLOAD_ERROR: ${dlErr?.message ?? 'empty'}`)
  const container = Buffer.from(await blob.arrayBuffer())

  let plain: Buffer
  try {
    plain = decryptFile(container, masterKeyHex, {
      context: invoice.file_object_path,
      expectedKeyVersion: invoice.file_encryption_key_version ?? undefined,
    })
  } catch (e) {
    // A corrupt/tampered/expired container can never succeed on retry.
    if (isMediaDecryptFailure(e)) throw new PermanentJobError(`MEDIA_DECRYPT_FAILED: ${(e as Error).message}`)
    throw e
  }
  timing.mark('decrypt')

  // Extraction now runs through the dedicated OCR engine (lib/ocr/invoice-extractor):
  // sharp + tesseract.js for images, a PDF text layer / bounded rendered-page OCR
  // for PDFs, and an optional privacy-GATED Gemini vision fallback. The handler
  // keeps ownership of decryption, encryption-at-rest, tenant-scoped writes,
  // review routing, batch scheduling and the in-memory wipe. Nothing is fabricated.
  const env = getEnv()
  try {
    // AI stays strictly behind the operator gate and is NEVER default-on: the
    // engine only reaches Gemini when allowAI is true AND a paid tier is set.
    const operatorAllowsAI = env.EXTRACTION_ALLOW_AI === true && env.EXTRACTION_AI_TIER === 'paid'
    let allowAI = false
    if (operatorAllowsAI) {
      // AI_CONSENT_READ_ERROR: consent?.gemini_consent_at != null && consent?.gemini_consent_version === GEMINI_WORKSPACE_CONSENT_VERSION
      const consent = await getWorkspaceGeminiConsentState(db, workspaceId)
      allowAI = consent?.enabled === true
        && consent.consentVersion === GEMINI_WORKSPACE_CONSENT_VERSION
      if (!allowAI) {
        logger.info('[Extract] Gemini withheld because workspace consent is absent or outdated', {
          workspaceId: workspaceId.slice(0, 8),
        })
      }
    }
    if (allowAI && env.EXTRACTION_PROVIDER !== 'gemini') {
      throw new PermanentJobError(`PROVIDER_UNIMPLEMENTED: ${env.EXTRACTION_PROVIDER}`)
    }

    if (allowAI) {
      const consent = await getWorkspaceGeminiConsentState(db, workspaceId)
      if (!consent?.enabled || consent.consentVersion !== GEMINI_WORKSPACE_CONSENT_VERSION) {
        throw new PermanentJobError('GEMINI_CONSENT_DISABLED')
      }
    }

    // The engine returns a raw candidate + provenance. A permanent failure (bad
    // media, unsupported type) never spins retries; a transient OCR/vision fault does.
    const result: InvoiceExtractionResult = await runExtraction({
      buffer: plain,
      mimeType: invoice.content_type,
      minConfidence: env.EXTRACTION_MIN_CONFIDENCE,
      amountToleranceMinorUnits: env.EXTRACTION_AMOUNT_TOLERANCE_MINOR_UNITS,
      allowAI,
      apiKey: allowAI ? (env.GEMINI_API_KEY || '') : undefined,
      model: env.EXTRACTION_VISION_MODEL || env.GEMINI_MODEL,
      buyerGstin,
      parallelVision: env.EXTRACTION_VISION_PARALLEL === true,
      imageQualityGate: imageQualityGateFromEnv(env),
    }).catch((e: unknown): never => {
      if (e instanceof InvoiceExtractionError) {
        if (!e.retryable) throw new PermanentJobError(`EXTRACTION_FAILED: ${e.message}`)
        throw e // transient OCR / vision fault -> the worker requeues
      }
      // A transient provider fault that reached us unwrapped must still requeue,
      // never settle; a permanent provider fault (bad key, unparseable output,
      // unsupported type) can never succeed on retry, so it settles immediately
      // instead of burning the queue's bounded attempt budget.
      if (e instanceof VisionTransientError) throw e
      if (e instanceof VisionPermanentError) throw new PermanentJobError(`EXTRACTION_FAILED: ${e.message}`)
      throw e
    })
    timing.mark('extract')

    // Re-evaluate the returned candidate independently so the handler owns the
    // canonical row and the date-resolved batching gate (the engine never returns
    // a canonicalised ledger row, only the fields it located).
    const evaluation = evaluateLedger(result.candidate, {
      minConfidence: env.EXTRACTION_MIN_CONFIDENCE,
      amountToleranceMinorUnits: env.EXTRACTION_AMOUNT_TOLERANCE_MINOR_UNITS,
      buyerGstin,
    })
    const confidenceSource: ExtractionSource = result.source

    // A critical-field conflict the engine detected is unreconciled disagreement
    // between the local and AI readings. A machine must not silently pick a winner,
    // so never auto-approve over it: downgrade to needs_review and make the reasons
    // name the exact fields, so the persisted status describes the selected candidate.
    const conflictFields = result.conflictFields ?? []
    if (conflictFields.length > 0) {
      evaluation.review = 'needs_review'
      evaluation.reasons = evaluation.reasons.filter((reason) => !reason.startsWith('money-field-conflict:'))
      const conflictReasons = conflictFields.map((field) => {
        const values = result.candidate.field_conflicts?.[field]
        return values
          ? `money-field-conflict:${field}:local=${values.local}:ai=${values.ai}`
          : `candidate-conflict:${field}`
      })
      evaluation.reasons = [...new Set([...evaluation.reasons, ...conflictReasons])]
      logger.info('[Extract] sources conflict on critical fields; forcing review', {
        invoiceId, jobId: job.id, source: confidenceSource, conflictFields,
      })
    }

    // Â§P3-13 fold the image-quality verdict into the authoritative row. A failing
    // gate can only ADD reasons and push the row to review (never approve it), so it
    // can never weaken auto-approve precision. Kept separate from the conflict rule.
    const imageQualityReasons = result.imageQualityReasons ?? []
    if (imageQualityReasons.length > 0) {
      if (evaluation.review === 'approved') evaluation.review = 'needs_review'
      evaluation.reasons = [...evaluation.reasons, ...imageQualityReasons]
      logger.info('[Extract] image quality gate forced review', {
        invoiceId, jobId: job.id, reasons: imageQualityReasons,
      })
    }

    // Â§P3-14 a multi-invoice document: the engine read TWO+ invoice numbers off one
    // file and returned a single group's row, dropping the others. That row can
    // never be auto-approved â€” a human must split the file. Like the quality gate
    // this only ADDS a reason and pushes to review, so precision is never weakened.
    if (result.multiInvoiceDetected) {
      if (evaluation.review === 'approved') evaluation.review = 'needs_review'
      if (!evaluation.reasons.includes('multi-invoice-document')) {
        evaluation.reasons = [...evaluation.reasons, 'multi-invoice-document']
      }
      logger.info('[Extract] multi-invoice document forced review', {
        invoiceId, jobId: job.id, pagesProcessed: result.pagesProcessed,
      })
    }

    // Â§P3-16 a printed CREDIT/DEBIT NOTE reverses an earlier invoice, so a machine
    // must never auto-approve it. Like the quality / multi-invoice gates this only
    // ADDS a reason and pushes to review (never approves), so auto-approve precision
    // is untouched. Because the reason is folded HERE (not inside the shared
    // evaluateLedger), a human correcting the note in applyInvoiceReview can still
    // approve it â€” the block is on silent machine approval, not on human review.
    const documentType = evaluation.row.document_type ?? 'invoice'
    if (documentType !== 'invoice') {
      if (evaluation.review === 'approved') evaluation.review = 'needs_review'
      if (!evaluation.reasons.includes('credit-debit-note')) {
        evaluation.reasons = [...evaluation.reasons, 'credit-debit-note']
      }
      logger.info('[Extract] credit/debit note forced review', {
        invoiceId, jobId: job.id, documentType,
      })
    }

    // Â§Part-2 classify the document's TYPE (what it IS) separately from its
    // accounting CATEGORY (how it books), from the document's OWN printed headings
    // â€” never a filename and never a single generic keyword. Only a POSITIVE
    // signal adds a review reason here: two competing titles, or a document that
    // is clearly NOT invoice-shaped (a receipt / payment voucher / delivery
    // challan / proforma). An ABSENT heading never blocks, so the historical
    // purchase-register default and its recall are preserved. Like the conflict /
    // quality / multi-invoice / note gates this can ONLY add a reason and push to
    // review (never approve). The full metadata is carried as an encrypted sibling
    // (`_classification`), never overwriting the original OCR row (Â§2.3).
    const classification = classifyDocument({
      structuralNoteType: documentType,
      headingText: headingRegion(result.rawText ?? ''),
      fullText: result.rawText ?? '',
    })
    if (classification.needsReview) {
      if (evaluation.review === 'approved') evaluation.review = 'needs_review'
      for (const reason of classification.reviewReasons) {
        if (!evaluation.reasons.includes(reason)) {
          evaluation.reasons = [...evaluation.reasons, reason]
        }
      }
      logger.info('[Extract] classification forced review', {
        invoiceId, jobId: job.id,
        documentType: classification.documentType,
        accountingCategory: classification.accountingCategory,
        reasonCodes: classification.reviewReasons,
      })
    }

    // Â§Part-5 Layer 2 (invoice-identity duplicate detection). Distinct from the
    // exact-file Layer 1: a re-scan of the SAME physical bill (a photo and a PDF
    // of one invoice) has different bytes yet the same identity. We build a blind
    // index (HMAC of the canonical invoice number + supplier + date under the
    // file master key â€” never the plaintext) and look for a PRIOR live invoice in
    // this (workspace, client) sharing it. A hit only ADDS a reason and pushes to
    // review (NEVER auto-drops, NEVER approves, NEVER deletes either row) â€” a
    // human confirms. The key is computed fail-soft and persisted as a separate
    // best-effort write below, so extraction is completely unaffected when the
    // additive `identity_key` migration has not been applied yet.
    let identityKeyHex: string | null = null
    try {
      identityKeyHex = computeIdentityKey(
        {
          invoiceNumber: evaluation.row.invoice_number,
          supplierGstin: evaluation.row.supplier_gstin,
          vendorName: evaluation.row.vendor_name,
          invoiceDate: evaluation.row.invoice_date,
        },
        masterKeyHex,
      )
    } catch {
      identityKeyHex = null
    }
    if (identityKeyHex) {
      const priorId = await findPriorDuplicateIdentity(db, {
        workspaceId, clientId: invoice.client_id, identityKeyHex, excludeInvoiceId: invoiceId,
      })
      if (priorId) {
        if (evaluation.review === 'approved') evaluation.review = 'needs_review'
        if (!evaluation.reasons.includes(DUPLICATE_IDENTITY_REASON)) {
          evaluation.reasons = [...evaluation.reasons, DUPLICATE_IDENTITY_REASON]
        }
        logger.info('[Extract] possible duplicate identity flagged for review', {
          invoiceId, jobId: job.id, priorInvoiceId: priorId.slice(0, 8),
        })
        // Â§3.5 record a durable, machine-'pending' duplicate-review DECISION row so a
        // human can later resolve the pair (confirmed_duplicate / distinct / resolved).
        // Kept SEPARATE and best-effort exactly like the identity_key write below: until
        // the additive `duplicate_reviews` migration is applied this upsert FAILS SOFT
        // (logged, swallowed) and detection still works purely through the encrypted
        // `_reviewReasons` path â€” no invoice is blocked, lost, or auto-approved. The
        // machine only ever writes 'pending'; it NEVER self-resolves a duplicate.
        // match_evidence is the SAME non-reversible blind index, never plaintext.
        try {
          const { error: drErr } = await db.from('duplicate_reviews').upsert({
            workspace_id: workspaceId,
            incoming_invoice_id: invoiceId,
            canonical_invoice_id: priorId,
            match_layer: 'identity',
            match_evidence: identityKeyToBytea(identityKeyHex),
            review_status: 'pending',
            updated_at: new Date().toISOString(),
          }, { onConflict: 'workspace_id,incoming_invoice_id,canonical_invoice_id' })
          if (drErr) {
            logger.warn('[Extract] duplicate_reviews write skipped (migration not applied?); decision record inactive', {
              invoiceId, jobId: job.id, code: drErr.code ?? 'unknown',
            })
          }
        } catch {
          // A missing table must NEVER fail a real bill's extraction.
        }
      }
    }

    // Say WHY nothing could read the file: a silent empty review row was previously
    // indistinguishable from a model failure in the logs.
    if (confidenceSource === 'human' && evaluation.review === 'needs_review') {
      logger.warn('[Extract] no extractor available: local OCR missed and AI is gated off', {
        invoiceId, jobId: job.id,
        mime: invoice.content_type,
        aiGate: env.EXTRACTION_ALLOW_AI !== true ? 'EXTRACTION_ALLOW_AI is not true' : 'EXTRACTION_AI_TIER is not paid',
        reasonCodes: evaluation.reasons.map((reason) => reason.split(':')[0]).slice(0, 4),
      })
    }

    const keyVersion = deriveKeyVersion(masterKeyHex)
    // Carry the machine reason codes inside the encrypted payload as a sibling
    // key (`_reviewReasons`) so the Review list and in-chat editor can explain
    // WHY a row was flagged, in plain English (Â§P0-3). No new column; readers
    // index known fields, it never reaches the register export, and it is never
    // logged. An approved row carries an empty list.
    // `_recipientGstin` is likewise an informational sibling (underscore-prefixed
    // so it is never mistaken for a candidate field) that lets P0-1c learn a
    // client's own GSTIN from the buyer token recurring across their bills.
    const payload = encryptExtractedPayload(
      JSON.stringify({
        ...evaluation.row,
        _reviewReasons: evaluation.reasons,
        _recipientGstin: result.candidate.recipient_gstin ?? null,
        _classification: classification,
      }),
      masterKeyHex,
    )

    await assertLease()
    const { error: recErr } = await db.from('extracted_records').upsert({
      workspace_id: workspaceId,
      invoice_id: invoiceId,
      encrypted_payload: `\\x${payload.toString('hex')}`,
      encryption_key_version: keyVersion,
      review_status: evaluation.review,
      confidence_source: confidenceSource,
      updated_at: new Date().toISOString(),
    }, { onConflict: 'invoice_id' })
    if (recErr) throw new Error(`RECORD_WRITE_ERROR: ${recErr.message}`)
    logger.info('[Extract] result saved', {
      invoiceId, jobId: job.id, review: evaluation.review,
      source: confidenceSource, confidence: evaluation.confidence,
      repairs: evaluation.repairs ?? [],
      presentFieldNames: [
        'invoice_date', 'invoice_number', 'vendor_name', 'supplier_gstin',
        'taxable_value', 'cgst_amount', 'sgst_amount', 'igst_amount', 'total_amount',
      ].filter((field) => {
        const value = result.candidate[field as keyof typeof result.candidate]
        return value !== undefined && value !== null && value !== ''
      }),
      reasonCodes: evaluation.reasons.map((reason) => reason.split(':')[0]).slice(0, 8),
    })
    timing.mark('persist')

    // Opt-in local debugging: print the complete field -> value map that was
    // extracted. Emitted at DEBUG level, which lib/logger drops when
    // NODE_ENV=production, so this can never leak invoice values to a prod log
    // sink even if the flag is left enabled. Enable locally with
    // EXTRACTION_DEBUG_VALUES=true in .env and restart `next dev`.
    if (getEnv().EXTRACTION_DEBUG_VALUES) {
      const present = (o: Record<string, unknown>) => Object.fromEntries(
        Object.entries(o).filter(([, v]) => v !== undefined && v !== null && v !== ''),
      )
      logger.debug('[Extract.debug] full extraction values (LOCAL ONLY)', {
        invoiceId,
        jobId: job.id,
        source: confidenceSource,
        review: evaluation.review,
        confidence: evaluation.confidence,
        rawCandidate: present(result.candidate as unknown as Record<string, unknown>),
        normalizedRow: present(evaluation.row as unknown as Record<string, unknown>),
        reasonCodes: evaluation.reasons.map((reason) => reason.split(':')[0]),
      })
    }

    // Route the invoice status. The invoices.status CHECK allows 'completed' but
    // NOT 'approved' ('approved' is an extracted_records.review_status value, not
    // an invoice lifecycle state), so a clean extraction lands as 'completed' and
    // anything unresolved as 'needs_review'.
    const invoiceStatus = evaluation.review === 'approved' ? 'completed' : 'needs_review'
    await assertLease()
    const { error: stErr } = await db.from('invoices').update({ status: invoiceStatus })
      .eq('workspace_id', workspaceId).eq('id', invoiceId)
    if (stErr) throw new Error(`INVOICE_STATUS_ERROR: ${stErr.message}`)

    // Â§Part-5: persist the identity blind index as a SEPARATE best-effort write.
    // Kept apart from the status update on purpose â€” if the additive
    // `identity_key` migration has not been applied, this update fails SOFT (a
    // logged warning) and the invoice is still correctly routed to review above;
    // the whole feature stays inactive until the operator applies the migration.
    if (identityKeyHex) {
      const { error: idErr } = await db.from('invoices')
        .update({ identity_key: identityKeyToBytea(identityKeyHex) })
        .eq('workspace_id', workspaceId).eq('id', invoiceId)
      if (idErr) {
        logger.warn('[Extract] identity_key write skipped (migration not applied?); duplicate detection inactive', {
          invoiceId, jobId: job.id, code: idErr.code ?? 'unknown',
        })
      }
    }

    // Batch + register scheduling is NOT restricted to approved invoices: a
    // needs_review invoice with a reliably parsed date still belongs in its real
    // (client, month) batch so the register can account for outstanding review
    // items. When the date is missing/ambiguous we NEVER guess the upload month â€”
    // the invoice stays unbatched and visible in the review workflow instead.
    if (evaluation.invoiceDateResolved) {
      const batchId = await resolveInvoiceBatch(db, {
        workspaceId, invoiceId, jobId: job.id, row: evaluation.row, invoice, assertLease,
      })
      logger.info('[Extract] batch assignment complete', { invoiceId, jobId: job.id, batchId })
    } else {
      logger.info('[Extract] no reliable invoice date; left unbatched for review', {
        invoiceId, jobId: job.id, reasonCodes: evaluation.reasons.map((reason) => reason.split(':')[0]).slice(0, 8),
      })
    }
    if (evaluation.review === 'needs_review') {
      await queueReviewNotice(db, workspaceId, job.id, invoiceId, assertLease)
    }
    timing.mark('batch')
    // Â§P2-12 one structured line per job with the per-stage breakdown (metadata only).
    logger.info('[Extract] stage timing', { invoiceId, jobId: job.id, ...timing.summary() })
    logger.info('[Extract] complete', {
      invoiceId, jobId: job.id, review: evaluation.review,
      source: confidenceSource, reasonCodes: evaluation.reasons.map((reason) => reason.split(':')[0]).slice(0, 8),
      mimeType: invoice.content_type, pagesProcessed: result.pagesProcessed,
    })
  } finally {
    plain.fill(0) // wipe plaintext bytes on every exit path (approve, review, throw)
  }
}

// Â§P3-13 translate the operator's image-quality knobs into the engine gate options.
// The gate is ON unless explicitly disabled; only thresholds the operator actually
// set are forwarded, so the engine applies its own sane defaults for the rest.
function imageQualityGateFromEnv(env: ReturnType<typeof getEnv>): {
  enabled: boolean
  blurVarianceAtMost?: number
  minLongEdge?: number
  minPixels?: number
} {
  return {
    enabled: env.IMAGE_QUALITY_GATE !== false,
    ...(typeof env.IMAGE_BLUR_VARIANCE_MAX === 'number' ? { blurVarianceAtMost: env.IMAGE_BLUR_VARIANCE_MAX } : {}),
    ...(typeof env.IMAGE_MIN_LONG_EDGE === 'number' ? { minLongEdge: env.IMAGE_MIN_LONG_EDGE } : {}),
    ...(typeof env.IMAGE_MIN_PIXELS === 'number' ? { minPixels: env.IMAGE_MIN_PIXELS } : {}),
  }
}

/**
 * Â§Part-5 Layer 2 lookup: is there a PRIOR LIVE invoice in this (workspace,
 * client) whose identity blind index equals `identityKeyHex`? Excludes the
 * invoice currently being extracted and any soft-deleted row. Tenant-scoped, and
 * a null client (individual) is matched with an `is client_id null` filter so two
 * organizations never see each other's keys.
 *
 * Deliberately FAIL-SOFT: if the additive `identity_key` column has not been
 * created yet (migration unapplied) the read errors â€” we log and return null so
 * extraction proceeds with duplicate detection simply inactive, exactly the way
 * the optional `own_gstin` read degrades. A lookup must never break a real bill.
 */
async function findPriorDuplicateIdentity(
  db: Db,
  args: { workspaceId: string; clientId: string | null; identityKeyHex: string; excludeInvoiceId: string },
): Promise<string | null> {
  try {
    let query = db
      .from('invoices')
      .select('id')
      .eq('workspace_id', args.workspaceId)
      .eq('identity_key', identityKeyToBytea(args.identityKeyHex))
      .neq('id', args.excludeInvoiceId)
      .neq('status', 'deleted')
      .limit(1)
    query = args.clientId
      ? (query.eq('client_id', args.clientId) as typeof query)
      : (query.is('client_id', null) as typeof query)
    const { data, error } = await (query as unknown as {
      maybeSingle: () => Promise<{ data: { id: string } | null; error: { message: string } | null }>
    }).maybeSingle()
    if (error) {
      logger.warn('[Extract] identity duplicate lookup failed; continuing (fail-soft)', {
        code: error.message.slice(0, 80),
      })
      return null
    }
    return data?.id ?? null
  } catch (e) {
    logger.warn('[Extract] identity duplicate lookup threw; continuing (fail-soft)', {
      error: e instanceof Error ? e.message : String(e),
    })
    return null
  }
}

async function queueReviewNotice(
  db: Db,
  workspaceId: string,
  jobId: string,
  invoiceId: string,
  assertLease: () => Promise<void>,
): Promise<void> {
  const { data, error } = await db.from('whatsapp_identities').select('sender_hmac')
    .eq('workspace_id', workspaceId).limit(1)
  if (error) {
    logger.warn('[Extract] review notice deferred: sender lookup failed', { invoiceId, jobId, code: error.code ?? null })
    return
  }
  const senderHmac = (data?.[0]?.sender_hmac as string | undefined) ?? null
  if (!senderHmac) {
    logger.info('[Extract] review notice unavailable: no WhatsApp sender linked', { invoiceId, jobId })
    return
  }
  // whatsapp_sessions has ONE pending_reply_body slot. If a register delivery
  // ([[deliver:<id>]]) is already waiting there, overwriting it with this notice
  // would silently drop the Excel hand-off; the register message already carries
  // the "N invoice(s) need review" note, so leave it untouched.
  const { data: slot } = await db.from('whatsapp_sessions').select('pending_reply_body')
    .eq('sender_hmac', senderHmac).maybeSingle()
  const pending = (slot?.pending_reply_body as string | null | undefined) ?? ''
  if (pending.startsWith('[[deliver:')) {
    logger.info('[Extract] review notice skipped: register delivery already queued', { invoiceId, jobId })
    return
  }
  await assertLease()
  const queued = await updateSessionDeliveryState(db, senderHmac, {
    pending_reply_body: 'One or more invoices need review and are not included in an Excel register until corrected. Reply Review or choose Menu > Review to see what needs attention.',
    last_reply_error_code: null,
  })
  logger.info(queued
    ? '[Extract] review notice queued for next inbound message'
    : '[Extract] review notice deferred: delivery state could not be saved', { invoiceId, jobId })
}
